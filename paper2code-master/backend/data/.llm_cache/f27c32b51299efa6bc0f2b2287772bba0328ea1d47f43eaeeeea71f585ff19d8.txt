import math
from typing import Optional, Tuple

import torch
import torch.nn as nn
import torch.nn.functional as F
import yaml
from omegaconf import OmegaConf


def _load_config(config_path: str = "config/config.yaml") -> dict:
    """Load configuration from YAML file."""
    with open(config_path, "r", encoding="utf-8") as f:
        config = yaml.safe_load(f)
    return config


class InterpretableMultiHeadAttention(nn.Module):
    """
    Interpretable Multi-Head Attention from TFT paper.
    Uses shared values across heads and additive aggregation of attention heads.
    """

    def __init__(self, d_model: int, num_heads: int, dropout: float = 0.1):
        super().__init__()
        assert d_model % num_heads == 0, "d_model must be divisible by num_heads"
        self.d_model = d_model
        self.num_heads = num_heads
        self.d_head = d_model // num_heads

        # Shared value projection across all heads
        self.v_proj = nn.Linear(d_model, d_model)
        # Query and key projections per head
        self.q_proj = nn.Linear(d_model, d_model)
        self.k_proj = nn.Linear(d_model, d_model)

        self.out_proj = nn.Linear(d_model, d_model)
        self.dropout = nn.Dropout(dropout)
        self.scale = math.sqrt(self.d_head)

    def forward(
        self,
        query: torch.Tensor,
        key: torch.Tensor,
        value: torch.Tensor,
        mask: Optional[torch.Tensor] = None,
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Args:
            query: (batch, seq_len_q, d_model)
            key: (batch, seq_len_k, d_model)
            value: (batch, seq_len_k, d_model)
            mask: (batch, seq_len_q, seq_len_k) or None
        Returns:
            output: (batch, seq_len_q, d_model)
            attention_weights: (batch, num_heads, seq_len_q, seq_len_k)
        """
        batch_size, seq_len_q, _ = query.shape
        _, seq_len_k, _ = key.shape

        # Project queries, keys, and shared values
        Q = self.q_proj(query)  # (batch, seq_len_q, d_model)
        K = self.k_proj(key)    # (batch, seq_len_k, d_model)
        V = self.v_proj(value)  # (batch, seq_len_k, d_model)

        # Reshape to (batch, num_heads, seq_len, d_head)
        Q = Q.view(batch_size, seq_len_q, self.num_heads, self.d_head).transpose(1, 2)
        K = K.view(batch_size, seq_len_k, self.num_heads, self.d_head).transpose(1, 2)
        V = V.view(batch_size, seq_len_k, self.num_heads, self.d_head).transpose(1, 2)

        # Scaled dot-product attention
        scores = torch.matmul(Q, K.transpose(-2, -1)) / self.scale  # (batch, num_heads, seq_len_q, seq_len_k)

        if mask is not None:
            scores = scores.masked_fill(mask == 0, float("-inf"))

        attention_weights = F.softmax(scores, dim=-1)
        attention_weights = self.dropout(attention_weights)

        # Apply attention to shared values
        context = torch.matmul(attention_weights, V)  # (batch, num_heads, seq_len_q, d_head)

        # Additive aggregation of attention heads
        context = context.mean(dim=1)  # (batch, seq_len_q, d_head)

        # Project back to d_model
        output = self.out_proj(context)  # (batch, seq_len_q, d_model)

        return output, attention_weights


class Transformer(nn.Module):
    """
    Temporal Fusion Transformer model for interpretable multihorizon time series forecasting.
    Implements the core Transformer architecture with interpretable multi-head attention.
    """

    def __init__(self, config: Optional[dict] = None):
        super().__init__()
        if config is None:
            config = _load_config()

        # Extract model hyperparameters from config with safe defaults
        model_cfg = config.get("model", {})
        dropout_ratio = model_cfg.get("dropout_ratio", [0.1, 0.3, 0.1, 0.3])
        if isinstance(dropout_ratio, list) and len(dropout_ratio) > 0:
            dropout = float(dropout_ratio[0])
        else:
            dropout = 0.1

        # Default values derived from TFT paper
        # d_model = 64 (hidden state size used in the paper)
        # num_heads = 4 (number of attention heads)
        # num_quantiles = 3 (0.1, 0.5, 0.9)
        self.d_model = 64
        self.num_heads = 4
        self.num_quantiles = 3
        self.dropout = dropout

        # Input embedding layers
        self.input_embedding = nn.Linear(1, self.d_model)  # Target values embedding
        self.static_embedding = nn.Linear(1, self.d_model)  # Static covariates embedding
        self.known_input_embedding = nn.Linear(1, self.d_model)  # Known future inputs embedding

        # Positional encoding
        self.pos_encoding = PositionalEncoding(self.d_model, dropout)

        # Interpretable multi-head attention
        self.attention = InterpretableMultiHeadAttention(
            d_model=self.d_model,
            num_heads=self.num_heads,
            dropout=dropout,
        )

        # Feed-forward network
        self.ffn = nn.Sequential(
            nn.Linear(self.d_model, self.d_model * 4),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(self.d_model * 4, self.d_model),
            nn.Dropout(dropout),
        )

        # Layer normalization
        self.norm1 = nn.LayerNorm(self.d_model)
        self.norm2 = nn.LayerNorm(self.d_model)

        # Output projection for quantile forecasts
        self.output_projection = nn.Linear(self.d_model, self.num_quantiles)

        # Initialize weights
        self._init_weights()

    def _init_weights(self):
        """Initialize weights using kaiming_normal as specified in config."""
        for module in self.modules():
            if isinstance(module, nn.Linear):
                nn.init.kaiming_normal_(module.weight, mode="fan_in", nonlinearity="relu")
                if module.bias is not None:
                    nn.init.constant_(module.bias, 0)
            elif isinstance(module, nn.LayerNorm):
                nn.init.constant_(module.weight, 1)
                nn.init.constant_(module.bias, 0)

    def forward(
        self,
        past_targets: torch.Tensor,
        static_covariates: Optional[torch.Tensor] = None,
        known_future_inputs: Optional[torch.Tensor] = None,
        mask: Optional[torch.Tensor] = None,
    ) -> torch.Tensor:
        """
        Forward pass of the Transformer model.

        Args:
            past_targets: (batch, seq_len, 1) - historical target values
            static_covariates: (batch, 1, 1) or None - static features
            known_future_inputs: (batch, horizon, 1) or None - known future inputs
            mask: (batch, seq_len, seq_len) or None - attention mask

        Returns:
            predictions: (batch, horizon, num_quantiles) - quantile forecasts
        """
        batch_size, seq_len, _ = past_targets.shape

        # Embed inputs
        x = self.input_embedding(past_targets)  # (batch, seq_len, d_model)

        # Add static covariates if provided
        if static_covariates is not None:
            static_emb = self.static_embedding(static_covariates)  # (batch, 1, d_model)
            static_emb = static_emb.expand(-1, seq_len, -1)
            x = x + static_emb

        # Add positional encoding
        x = self.pos_encoding(x)

        # Self-attention with residual connection
        attn_out, attention_weights = self.attention(x, x, x, mask)
        x = self.norm1(x + self.dropout(attn_out))

        # Feed-forward with residual connection
        ffn_out = self.ffn(x)
        x = self.norm2(x + ffn_out)

        # Generate predictions for future horizons
        if known_future_inputs is not None:
            horizon = known_future_inputs.shape[1]
            # Use the last hidden state to generate forecasts
            last_hidden = x[:, -1, :]  # (batch, d_model)
            # Expand for each horizon step
            last_hidden = last_hidden.unsqueeze(1).expand(-1, horizon, -1)
            # Add known future inputs
            future_emb = self.known_input_embedding(known_future_inputs)
            combined = last_hidden + future_emb
        else:
            # If no known future inputs, use last hidden state repeated
            horizon = 1
            last_hidden = x[:, -1, :]  # (batch, d_model)
            combined = last_hidden.unsqueeze(1)

        # Project to quantile outputs
        predictions = self.output_projection(combined)  # (batch, horizon, num_quantiles)

        return predictions


class PositionalEncoding(nn.Module):
    """Sinusoidal positional encoding."""

    def __init__(self, d_model: int, dropout: float = 0.1, max_len: int = 5000):
        super().__init__()
        self.dropout = nn.Dropout(p=dropout)

        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
        div_term = torch.exp(torch.arange(0, d_model, 2).float() * (-math.log(10000.0) / d_model))
        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)
        pe = pe.unsqueeze(0).transpose(0, 1)
        self.register_buffer("pe", pe)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (batch, seq_len, d_model)
        Returns:
            (batch, seq_len, d_model)
        """
        x = x + self.pe[: x.size(1), :]
        return self.dropout(x)


def create_transformer_model(config_path: str = "config/config.yaml") -> Transformer:
    """Factory function to create a Transformer model from config."""
    config = _load_config(config_path)
    return Transformer(config)
