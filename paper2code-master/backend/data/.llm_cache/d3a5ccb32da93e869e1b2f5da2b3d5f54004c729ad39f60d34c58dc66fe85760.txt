import sys
import json
import numpy as np
import cv2

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        if img is None:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
    except Exception:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

    # OCR text provided externally
    ocr_text = """Transformer Encoder wth
XGBoost Classifier
Attention Mechanisms
Linear

Add
Norm
Lines of Code (LOC)
Feed Forward
McCabe (Cyclomatic)

forward

Forward

Tree
Selected Halstead Metrics
Nx
Multi-Head
Attention
Cross-attention
Self:
attention
Masked
Positional
Multi-Head Attention"""

    ocr_lower = ocr_text.lower()

    # Architecture keywords
    arch_keywords = [
        "cnn", "transformer", "attention", "encoder", "decoder",
        "block", "embedding", "classifier", "feature extractor",
        "alexnet", "resnet", "vgg", "googlenet"
    ]

    # Experiment result keywords
    exp_keywords = [
        "accuracy", "loss", "epoch", "precision", "recall", "f1",
        "curve", "performance", "comparison"
    ]

    # Dataset keywords
    dataset_keywords = [
        "sample", "example", "dataset", "input image", "training data"
    ]

    # Check for experiment result first (highest priority)
    exp_found = [kw for kw in exp_keywords if kw in ocr_lower]
    if exp_found:
        # Extract model names if present
        components = []
        for kw in ["alexnet", "resnet", "vgg", "googlenet", "transformer", "cnn", "xgboost"]:
            if kw in ocr_lower:
                components.append(kw.title())
        # Add metric names
        for kw in ["accuracy", "loss", "precision", "recall", "f1"]:
            if kw in ocr_lower:
                components.append(kw.title())
        if not components:
            components = []
        return {
            "type": "experiment_result",
            "description": "展示实验结果或性能对比数据",
            "important_components": components,
            "code_relevant_information": "需要实现实验评估模块，包含指标计算和结果可视化"
        }

    # Check for architecture
    arch_found = [kw for kw in arch_keywords if kw in ocr_lower]
    if arch_found:
        # Extract specific components
        components = []
        if "transformer" in ocr_lower:
            components.append("Transformer Encoder")
        if "attention" in ocr_lower:
            components.append("Attention Mechanism")
        if "multi-head" in ocr_lower:
            components.append("Multi-Head Attention")
        if "xgboost" in ocr_lower:
            components.append("XGBoost Classifier")
        if "classifier" in ocr_lower:
            components.append("Classifier")
        if "embedding" in ocr_lower:
            components.append("Embedding")
        if "feed forward" in ocr_lower:
            components.append("Feed Forward")
        if "softmax" in ocr_lower:
            components.append("Softmax")
        if "linear" in ocr_lower:
            components.append("Linear")
        if not components:
            components = []
        return {
            "type": "architecture",
            "description": "展示模型网络结构和数据流关系",
            "important_components": components,
            "code_relevant_information": "需要实现模型架构，包含特征提取、注意力机制和分类模块"
        }

    # Check for dataset
    dataset_found = [kw for kw in dataset_keywords if kw in ocr_lower]
    if dataset_found:
        return {
            "type": "dataset",
            "description": "展示数据集样本或输入示例",
            "important_components": [],
            "code_relevant_information": "需要实现数据加载和预处理模块"
        }

    # Default case
    return {
        "type": "other",
        "description": "无法确定图片类型",
        "important_components": [],
        "code_relevant_information": ""
    }

if __name__ == "__main__":
