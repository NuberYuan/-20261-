import sys
import json
import numpy as np
import cv2
from PIL import Image

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty, so rely on visual features
        # Get basic image properties
        height, width = img.shape[:2]
        
        # Convert to grayscale for analysis
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        
        # Check if image is mostly white/light (typical for architecture diagrams)
        white_ratio = np.sum(gray > 200) / (height * width)
        
        # Check color variance (experiment results often have colored lines)
        color_std = np.std(img.reshape(-1, 3), axis=0).mean()
        
        # Check if image has grid-like structure (common in plots)
        # Simple check: look for horizontal/vertical line patterns
        edges = cv2.Canny(gray, 50, 150)
        horizontal_lines = np.sum(edges, axis=1)
        vertical_lines = np.sum(edges, axis=0)
        
        # Detect if there are many horizontal/vertical lines (plot-like)
        h_line_ratio = np.sum(horizontal_lines > 100) / height
        v_line_ratio = np.sum(vertical_lines > 100) / width
        
        # Architecture diagrams typically:
        # - High white ratio (clean background)
        # - Low color variance (mostly black/white)
        # - Few horizontal/vertical grid lines
        is_architecture = white_ratio > 0.7 and color_std < 50 and h_line_ratio < 0.1 and v_line_ratio < 0.1
        
        # Experiment results typically:
        # - Have grid lines
        # - Have colored elements
        # - Medium white ratio
        is_experiment = (h_line_ratio > 0.15 or v_line_ratio > 0.15) and color_std > 30
        
        # Dataset images typically:
        # - Have multiple distinct regions
        # - Medium white ratio
        # - Varied colors
        is_dataset = not is_architecture and not is_experiment and color_std > 60
        
        # Since OCR is empty, rely on visual features
        if is_experiment:
            return {
                "type": "experiment_result",
                "description": "展示实验结果曲线或性能对比图表",
                "important_components": [],
                "code_relevant_information": "需要实现实验评估模块，包含指标计算和结果可视化"
            }
        elif is_architecture:
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": [],
                "code_relevant_information": "需要实现模型网络结构定义和参数初始化"
            }
        elif is_dataset:
            return {
                "type": "dataset",
                "description": "展示数据集样本或输入示例",
                "important_components": [],
                "code_relevant_information": "需要实现数据加载和预处理模块"
            }
        else:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
            
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
