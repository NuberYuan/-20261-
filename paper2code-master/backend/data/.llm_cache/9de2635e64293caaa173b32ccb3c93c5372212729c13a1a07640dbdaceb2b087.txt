import os
import sys
import random
import logging
from typing import Any, Dict, Optional

import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim

PROJECT_ROOT = os.path.abspath(os.path.join(os.path.dirname(__file__), ".."))
if PROJECT_ROOT not in sys.path:
    sys.path.insert(0, PROJECT_ROOT)

from data.dataset import get_dataloaders, load_config
from models.u_net import UNet

logger = logging.getLogger(__name__)


def _resolve_config(config: Optional[Any] = None) -> Dict[str, Any]:
    """Resolve config into a dict from a path, dict, or default config.yaml."""
    if config is None:
        config_path = os.path.join(PROJECT_ROOT, "config", "config.yaml")
        return load_config(config_path)
    if isinstance(config, str):
        return load_config(config)
    if isinstance(config, dict):
        return config
    config_path = os.path.join(PROJECT_ROOT, "config", "config.yaml")
    return load_config(config_path)


def _resolve_device(config: Optional[Any] = None, device: Optional[str] = None) -> str:
    """Resolve target device from config, falling back to CPU when CUDA is unavailable."""
    if device is not None:
        return device
    cfg = _resolve_config(config) if not isinstance(config, dict) else config
    requested_device = cfg.get("system", {}).get("device", "cuda:0")
    if (
        requested_device
        and "cuda" in str(requested_device).lower()
        and not torch.cuda.is_available()
    ):
        print("Warning: CUDA device requested but not available; falling back to CPU.")
        return "cpu"
    return requested_device or "cpu"


def init_weights(module: nn.Module) -> None:
    """Kaiming-normal initialization for Conv2d and standard init for BatchNorm2d."""
    if isinstance(module, nn.Conv2d):
        nn.init.kaiming_normal_(module.weight, mode="fan_out", nonlinearity="relu")
        if module.bias is not None:
            nn.init.constant_(module.bias, 0)
    elif isinstance(module, nn.BatchNorm2d):
        if module.weight is not None:
            nn.init.constant_(module.weight, 1)
        if module.bias is not None:
            nn.init.constant_(module.bias, 0)


def initialize_model(
    config: Optional[Any] = None,
    num_classes: int = 1,
    in_channels: int = 3,
    device: Optional[str] = None,
) -> nn.Module:
    """Build a randomly initialized UNet++ model from config.

    This follows training_modes.scratch: no pretrained weights are loaded.
    """
    cfg = _resolve_config(config)
    model_cfg = cfg.get("model", {}) if isinstance(cfg, dict) else {}
    resolved_num_classes = int(model_cfg.get("num_classes", num_classes))
    if resolved_num_classes < 1:
        resolved_num_classes = 1

    model = UNet(config=cfg, in_channels=in_channels, deep_supervision=False)
    model.apply(init_weights)
    target_device = _resolve_device(cfg, device)
    model.to(target_device)
    return model


class BCEWithDiceLoss(nn.Module):
    """Combined binary cross-entropy and soft Dice loss.

    The model outputs sigmoid probabilities for binary segmentation; therefore
    binary_cross_entropy is used without logits.
    """

    def __init__(self, bce_weight: float = 1.0, dice_weight: float = 1.0, smooth: float = 1e-6):
        super().__init__()
        self.bce_weight = float(bce_weight)
        self.dice_weight = float(dice_weight)
        self.smooth = float(smooth)

    def forward(self, preds: torch.Tensor, targets: torch.Tensor) -> torch.Tensor:
        preds = preds.float()
        targets = targets.float()

        bce_loss = F.binary_cross_entropy(preds, targets, reduction="mean")

        batch_size = preds.shape[0]
        preds_flat = preds.reshape(batch_size, -1)
        targets_flat = targets.reshape(batch_size, -1)

        intersection = (preds_flat * targets_flat).sum(dim=1)
        cardinality = preds_flat.sum(dim=1) + targets_flat.sum(dim=1)
        dice_score = (2.0 * intersection + self.smooth) / (cardinality + self.smooth)
        dice_loss = 1.0 - dice_score.mean()

        return self.bce_weight * bce_loss + self.dice_weight * dice_loss


def validate_model(
    model: nn.Module,
    dataloader: torch.utils.data.DataLoader,
    criterion: nn.Module,
    device: str,
    threshold: float = 0.5,
) -> Dict[str, float]:
    """Run validation and return loss, accuracy, and val_accuracy."""
    if dataloader is None:
        return {"loss": 0.0, "accuracy": 0.0, "val_accuracy": 0.0}

    model.eval()
    total_loss = 0.0
    total_correct = 0
    total_pixels = 0

    with torch.no_grad():
        for images, masks in dataloader:
            images = images.to(device)
            masks = masks.to(device)

            outputs = model(images)
            loss = criterion(outputs, masks)
            total_loss += loss.item()

            preds = (outputs > threshold).float()
            total_correct += (preds == masks).sum().item()
            total_pixels += masks.numel()

    avg_loss = total_loss / max(len(dataloader), 1)
    accuracy = total_correct / float(total_pixels) if total_pixels > 0 else 0.0
    model.train()
    return {"loss": avg_loss, "accuracy": accuracy, "val_accuracy": accuracy}


def save_model(
    model: nn.Module,
    path: str,
    optimizer: Optional[optim.Optimizer] = None,
    epoch: Optional[int] = None,
    metrics: Optional[Dict[str, Any]] = None,
    config: Optional[Any] = None,
) -> str:
    """Save model weights, optimizer state, and optional metadata.

    By default optimizer state is saved when checkpoint.save_optimizer_state is true.
    """
    if not path:
        raise ValueError("save_model requires a non-empty path.")

    dirname = os.path.dirname(path)
    if dirname:
        os.makedirs(dirname, exist_ok=True)

    checkpoint = {
        "model_state_dict": model.state_dict(),
        "model_class": model.__class__.__name__,
    }

    save_optimizer_state = True
    if config is not None:
        cfg = _resolve_config(config)
        checkpoint_cfg = cfg.get("checkpoint", {}) if isinstance(cfg, dict) else {}
        save_optimizer_state = bool(checkpoint_cfg.get("save_optimizer_state", True))

    if optimizer is not None and save_optimizer_state:
        checkpoint["optimizer_state_dict"] = optimizer.state_dict()

    if epoch is not None:
        checkpoint["epoch"] = int(epoch)

    if metrics is not None:
        checkpoint["metrics"] = {
            key: float(value) if isinstance(value, (int, float)) else value
            for key, value in metrics.items()
        }

    torch.save(checkpoint, path)
    return path


def train_model(
    model: Optional[nn.Module] = None,
    train_loader: Optional[torch.utils.data.DataLoader] = None,
    val_loader: Optional[torch.utils.data.DataLoader] = None,
    config: Optional[Any] = None,
    device: Optional[str] = None,
    checkpoint_dir: Optional[str] = None,
    run_name: Optional[str] = None,
    threshold: float = 0.5,
) -> Dict[str, Any]:
    """Train a model for one run.

    If val_loader is None, the validation branch is skipped entirely. This is the
    required behavior for reproduction_mode == 'strict'.
    """
    cfg = _resolve_config(config)

    if model is None or train_loader is None:
        print("train_model: model and train_loader are required; skipping training.")
        return {
            "history": {"train_loss": [], "val_loss": [], "val_accuracy": []},
            "best_score": None,
            "best_model_path": None,
            "final_model_path": None,
        }

    training_cfg = cfg.get("training", {}) if isinstance(cfg, dict) else {}
    checkpoint_cfg = cfg.get("checkpoint", {}) if isinstance(cfg, dict) else {}
    target_device = _resolve_device(cfg, device)

    if checkpoint_dir is None:
        log_dir = cfg.get("logging", {}).get("save_dir", "logs")
        checkpoint_dir = os.path.join(log_dir, "checkpoints")
    os.makedirs(checkpoint_dir, exist_ok=True)

    optimizer_cfg = training_cfg.get("optimizer", {}) or {}
    lr = float(optimizer_cfg.get("lr", 4))
    weight_decay = optimizer_cfg.get("weight_decay")
    weight_decay = float(weight_decay) if weight_decay is not None else 0.0

    optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=weight_decay)

    epochs = int(training_cfg.get("epochs", 30))
    scheduler_cfg = training_cfg.get("scheduler") or {}
    scheduler = None
    if isinstance(scheduler_cfg, dict):
        scheduler_name = scheduler_cfg.get("name")
        if scheduler_name:
            step_size = int(scheduler_cfg.get("step_size", 10))
            gamma = float(scheduler_cfg.get("gamma", 0.1))
            if "cos" in str(scheduler_name).lower():
                scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs)
            elif "step" in str(scheduler_name).lower():
                scheduler = optim.lr_scheduler.StepLR(optimizer, step_size=step_size, gamma=gamma)

    save_best = bool(checkpoint_cfg.get("save_best", True))
    save_final = bool(checkpoint_cfg.get("save_final", True))
    best_metric = checkpoint_cfg.get("best_metric", "val_accuracy")
    best_mode = checkpoint_cfg.get("best_mode", "max")

    best_score = float("-inf") if best_mode == "max" else float("inf")
    prefix = run_name or "unetplusplus"
    best_path = os.path.join(checkpoint_dir, f"{prefix}_best.pth")
    final_path = os.path.join(checkpoint_dir, f"{prefix}_final.pth")

    criterion = BCEWithDiceLoss()
    model.to(target_device)
    model.train()

    history = {"train_loss": [], "val_loss": [], "val_accuracy": []}
    final_val_metrics = None

    for epoch in range(1, epochs + 1):
        model.train()
        running_loss = 0.0

        for images, masks in train_loader:
            images = images.to(target_device)
            masks = masks.to(target_device)

            optimizer.zero_grad(set_to_none=True)
            outputs = model(images)
            loss = criterion(outputs, masks)
            loss.backward()
            optimizer.step()

            running_loss += loss.item()

        avg_train_loss = running_loss / max(len(train_loader), 1)
        history["train_loss"].append(avg_train_loss)

        if val_loader is not None:
            val_metrics = validate_model(model, val_loader, criterion, target_device, threshold)
            history["val_loss"].append(val_metrics["loss"])
            history["val_accuracy"].append(val_metrics["val_accuracy"])
            final_val_metrics = val_metrics

            current_score = val_metrics.get(best_metric, val_metrics.get("val_accuracy", 0.0))
            print(
                f"Epoch {epoch}/{epochs} | "
                f"train_loss={avg_train_loss:.6f} | "
                f"val_loss={val_metrics['loss']:.6f} | "
                f"val_accuracy={val_metrics['val_accuracy']:.6f}"
            )

            improved = False
            if best_mode == "max":
                improved = current_score > best_score
            else:
                improved = current_score < best_score

            if improved:
                best_score = current_score
                if save_best:
                    save_model(
                        model,
                        best_path,
                        optimizer=optimizer,
                        epoch=epoch,
                        metrics=val_metrics,
                        config=cfg,
                    )
        else:
            final_val_metrics = None
            print(f"Epoch {epoch}/{epochs} | train_loss={avg_train_loss:.6f}")

        if scheduler is not None:
            scheduler.step()

    if save_final:
        save_model(
            model,
            final_path,
            optimizer=optimizer,
            epoch=epochs,
            metrics=final_val_metrics,
            config=cfg,
        )

    return {
        "history": history,
        "best_score": best_score,
        "best_model_path": best_path if save_best else None,
        "final_model_path": final_path if save_final else None,
    }


def main(config_path: Optional[str] = None, run_index: Optional[int] = None) -> None:
    """Entry point for training pipeline.

    Supports both engineering and strict reproduction modes:
    - engineering: split train/val/test, best model selected by val_accuracy, then test.
    - strict: no validation; only final model saved, then test.
    """
    if config_path is None:
        config_path = os.path.join(PROJECT_ROOT, "config", "config.yaml")

    config = load_config(config_path)

    log_level = config.get("logging", {}).get("level", "INFO")
    logging.basicConfig(level=getattr(logging, str(log_level).upper(), logging.INFO))

    seed = int(config.get("system", {}).get("seed", 42))
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)

    reproduction_mode = config.get("reproduction_mode", "engineering")
    runs = config.get("experiment", {}).get("runs", [])
    if not runs:
        default_version = config.get("version_selector", {}).get("default", "color")
        runs = [
            {
                "run_name": "unetplusplus_scratch_color",
                "architecture": "UNet++",
                "dataset_version": default_version,
                "training_mode": "scratch",
            }
        ]

    if run_index is not None:
        runs = [runs[int(run_index)]]

    batch_size = int(config.get("training", {}).get("batch_size", 16))
    num_workers = int(config.get("system", {}).get("num_workers", 4))
    checkpoint_dir = os.path.join(
        config.get("logging", {}).get("save_dir", "logs"), "checkpoints"
    )
    device = _resolve_device(config)

    for run in runs:
        dataset_version = run.get(
            "dataset_version", config.get("version_selector", {}).get("default", "color")
        )
        run_name = run.get("run_name", "run")
        logger.info("Run: %s | version: %s | mode: %s", run_name, dataset_version, reproduction_mode)

        train_loader, val_loader, test_loader = get_dataloaders(
            config=config,
            version=dataset_version,
            batch_size=batch_size,
            num_workers=num_workers,
            root_dir=config.get("data", {}).get("root_dir"),
        )

        # In strict mode the validation loader is deliberately not used.
        val_loader_to_use = None if reproduction_mode == "strict" else val_loader

        model = initialize_model(config=config, device=device)

        train_model(
            model=model,
            train_loader=train_loader,
            val_loader=val_loader_to_use,
            config=config,
            device=device,
            checkpoint_dir=checkpoint_dir,
            run_name=run_name,
        )

        # Test set is used only for final evaluation, never for training/validation.
        if test_loader is not None and len(test_loader) > 0:
            best_path = os.path.join(checkpoint_dir, f"{run_name}_best.pth")
            final_path = os.path.join(checkpoint_dir, f"{run_name}_final.pth")
            eval_path = best_path if os.path.isfile(best_path) else final_path

            if os.path.isfile(eval_path):
                checkpoint = torch.load(eval_path, map_location=device)
                model.load_state_dict(checkpoint.get("model_state_dict", checkpoint))
                print(f"Loaded {eval_path} for final test evaluation.")

            model.to(device)
            model.eval()
            criterion = BCEWithDiceLoss()
            test_metrics = validate_model(model, test_loader, criterion, device, threshold=0.5)
            print(f"Run {run_name} test_metrics: {test_metrics}")
        else:
            logger.info("No test set available for run %s; skipping final evaluation.", run_name)


if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(description="Train UNet++ segmentation model.")
    parser.add_argument("--config", type=str, default=None, help="Path to config.yaml")
    parser.add_argument("--run_index", type=int, default=None, help="Run only the given experiment index")
    args = parser.parse_args()

    main(config_path=args.config, run_index=args.run_index)
