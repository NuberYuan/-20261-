import sys
import json
import numpy as np
import cv2
from PIL import Image

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text provided externally
        ocr_text = "6517 0631"
        
        # Architecture keywords
        arch_keywords = [
            "CNN", "Transformer", "Attention", "Encoder", "Decoder",
            "Block", "Embedding", "Classifier", "Feature Extractor",
            "AlexNet", "ResNet", "VGG", "GoogLeNet"
        ]
        
        # Experiment result keywords
        exp_keywords = [
            "Loss", "Epoch", "Precision", "Recall", "F1",
            "Accuracy", "准确率", "实验", "曲线", "性能"
        ]
        
        # Dataset keywords
        dataset_keywords = ["样本", "类别", "示例", "输入"]
        
        # Check OCR text for keywords
        ocr_upper = ocr_text.upper()
        
        # Determine type based on priority
        img_type = "other"
        description = "无法确定图片类型"
        components = []
        code_info = ""
        
        # Check experiment result first (highest priority)
        exp_found = [kw for kw in exp_keywords if kw.upper() in ocr_upper]
        if exp_found:
            img_type = "experiment_result"
            description = "展示实验结果或性能对比"
            components = exp_found[:3]
            code_info = "需要实现实验评估和结果可视化功能"
        else:
            # Check architecture
            arch_found = [kw for kw in arch_keywords if kw.upper() in ocr_upper]
            if arch_found:
                img_type = "architecture"
                description = "展示模型网络结构和数据流关系"
                components = arch_found[:3]
                code_info = "需要实现模型架构和网络层定义"
            else:
                # Check dataset
                dataset_found = [kw for kw in dataset_keywords if kw in ocr_text]
                if dataset_found:
                    img_type = "dataset"
                    description = "展示数据集样本或输入示例"
                    components = []
                    code_info = "需要实现数据加载和预处理功能"
                else:
                    # Use basic visual info as fallback
                    height, width = img.shape[:2]
                    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
                    mean_brightness = np.mean(gray)
                    
                    # Simple heuristic: if image has text-like patterns
                    if mean_brightness > 200:
                        img_type = "other"
                        description = "图片内容不明确"
                        components = []
                        code_info = ""
        
        return {
            "type": img_type,
            "description": description,
            "important_components": components,
            "code_relevant_information": code_info
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
    else:
        image_path = r"D:\paper2code-master(1)\paper2code-master\backend\data\output\LEARNING_SIMPLE_ALGORITHMS_FROM_EXAMPLES\step1\images\image_1.png"
    
    result = analyze_image(image_path)
    print(json.dumps(result, ensure_ascii=False))
