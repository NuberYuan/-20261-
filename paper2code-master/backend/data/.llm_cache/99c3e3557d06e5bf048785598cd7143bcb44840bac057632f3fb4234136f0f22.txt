import os
import sys
import json
import numpy as np
import cv2

OCR_TEXT = """

0.20
(n=-1)
0.15
Daily Interval
(24 hours)

0.05
0.00

0.01
WUwulul
UM
"""

def read_image(image_path):
    data = np.fromfile(image_path, dtype=np.uint8)
    img = cv2.imdecode(data, cv2.IMREAD_COLOR)
    return img

def get_visual_basic_info(img):
    if img is None:
        return None
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    return {
        "height": int(img.shape[0]),
        "width": int(img.shape[1]),
        "gray_mean": float(gray.mean()),
        "gray_std": float(gray.std())
    }

def extract_components_from_text(text, mapping):
    components = []
    for key, val in mapping:
        if key in text.lower():
            components.append(val)
    seen = set()
    unique = []
    for c in components:
        if c not in seen:
            seen.add(c)
            unique.append(c)
    return unique

def classify_image(image_path, ocr_text):
    try:
        img = read_image(image_path)
        visual_info = get_visual_basic_info(img)
    except Exception:
        img = None
        visual_info = None

    text = ocr_text.lower()

    exp_keywords = [
        'accuracy', 'loss', 'epoch', 'precision', 'recall', 'f1',
        'mean', 'percentage', '%', 'curve', 'plot', 'performance',
        'position index', 'previous hour', 'daily interval', 'interval',
        't=', 't-', '0.20', '0.15', '0.10', '0.05', '0.00',
        '0.04', '0.03', '0.02', '0.01'
    ]
    if any(k in text for k in exp_keywords):
        model_mapping = [
            ('alexnet', 'AlexNet'),
            ('resnet', 'ResNet'),
            ('vgg', 'VGG'),
            ('googlenet', 'GoogLeNet'),
            ('transformer', 'Transformer'),
            ('attention', 'Attention'),
        ]
        metric_mapping = [
            ('accuracy', 'Accuracy'),
            ('precision', 'Precision'),
            ('recall', 'Recall'),
            ('f1', 'F1-score'),
            ('loss', 'Loss'),
            ('epoch', 'Epoch'),
            ('mean', 'Mean'),
            ('position index', 'Position Index'),
        ]
        components = extract_components_from_text(ocr_text, model_mapping)
        components.extend(extract_components_from_text(ocr_text, metric_mapping))
        seen = set()
        unique_components = []
        for c in components:
            if c not in seen:
                seen.add(c)
                unique_components.append(c)
        return {
            "type": "experiment_result",
            "description": "展示不同时间间隔下性能指标随位置索引变化的实验曲线",
            "important_components": unique_components,
            "code_relevant_information": "需要实现数据可视化，绘制多条曲线对比不同参数T下的实验结果"
        }

    arch_keywords = [
        'cnn', 'transformer', 'attention', 'encoder', 'decoder', 'block',
        'embedding', 'classifier', 'feature extractor', 'alexnet', 'resnet',
        'vgg', 'googlenet'
    ]
    if any(k in text for k in arch_keywords):
        arch_mapping = [
            ('cnn', 'CNN'),
            ('transformer', 'Transformer'),
            ('attention', 'Attention'),
            ('encoder', 'Encoder'),
            ('decoder', 'Decoder'),
            ('block', 'Block'),
            ('embedding', 'Embedding'),
            ('classifier', 'Classifier'),
            ('feature extractor', 'Feature Extractor'),
            ('alexnet', 'AlexNet'),
            ('resnet', 'ResNet'),
            ('vgg', 'VGG'),
            ('googlenet', 'GoogLeNet'),
        ]
        components = extract_components_from_text(ocr_text, arch_mapping)
        return {
            "type": "architecture",
            "description": "展示模型网络结构和数据流关系",
            "important_components": components,
            "code_relevant_information": "需要实现特征提取模块和分类头等网络结构"
        }

    dataset_keywords = [
        'sample', 'example', 'input image', 'category', 'class', 'dataset',
        'instance', 'training examples'
    ]
    if any(k in text for k in dataset_keywords):
        return {
            "type": "dataset",
            "description": "展示数据样本或类别图片",
            "important_components": [],
            "code_relevant_information": "需要准备或加载数据集样本"
        }

    return {
        "type": "other",
        "description": "无法确定图片类型",
        "important_components": [],
        "code_relevant_information": ""
    }

def main():
    try:
        if len(sys.argv) < 2:
            raise ValueError("Missing image path argument")
        image_path = sys.argv[1]
        result = classify_image(image_path, OCR_TEXT)
        print(json.dumps(result, ensure_ascii=False))
    except Exception:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))

if __name__ == "__main__":
    main()
