import sys
import json
import numpy as np
import cv2

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty for this image
        ocr_text = ""
        
        # Since OCR text is empty, use visual information
        # Get basic image properties
        height, width = img.shape[:2]
        
        # Check if image is mostly white/light (typical for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        mean_brightness = np.mean(gray)
        
        # Check color variance (architecture diagrams often have low color variance)
        color_variance = np.std(img)
        
        # Check if image has many distinct colors (dataset images often have)
        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        saturation = np.mean(hsv[:,:,1])
        
        # Heuristic classification based on visual features
        # Architecture diagrams: usually white background, low saturation, simple shapes
        # Dataset images: often colorful, high saturation
        # Experiment results: usually have lines/curves, moderate colors
        
        if mean_brightness > 200 and saturation < 50:
            # Likely architecture diagram (white background, low color)
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": [],
                "code_relevant_information": "需要实现模型网络结构"
            }
        elif saturation > 100:
            # Likely dataset images (colorful)
            return {
                "type": "dataset",
                "description": "展示数据样本或类别图片",
                "important_components": [],
                "code_relevant_information": "需要准备和处理数据集"
            }
        elif mean_brightness < 150:
            # Likely experiment results (darker plots)
            return {
                "type": "experiment_result",
                "description": "展示实验结果或性能对比",
                "important_components": [],
                "code_relevant_information": "需要实现实验评估和结果可视化"
            }
        else:
            # Default to other if cannot determine
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
            
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
