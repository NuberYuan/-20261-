import sys
import json
import numpy as np
import cv2

# OCR 文字由外部模块提供，此处作为已知信息内置，禁止再次调用 OCR
OCR_TEXT = """

def load_image_from_chinese_path(image_path):
    data = np.fromfile(image_path, dtype=np.uint8)
    img = cv2.imdecode(data, cv2.IMREAD_COLOR)
    if img is None:
        raise ValueError("无法读取图片")
    return img


def get_basic_visual_info(img):
    h, w, _ = img.shape
    aspect_ratio = float(w) / max(float(h), 1.0)
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    mean_brightness = float(np.mean(gray))
    std_brightness = float(np.std(gray))
    nonwhite_ratio = float(np.mean(gray < 240))

    b, g, r = cv2.split(img.astype(np.float32))
    color_diff = float(np.mean(np.abs(r - g)) + np.mean(np.abs(g - b)))

    return {
        "aspect_ratio": aspect_ratio,
        "mean_brightness": mean_brightness,
        "std_brightness": std_brightness,
        "nonwhite_ratio": nonwhite_ratio,
        "color_diff": color_diff,
    }


def extract_experiment_components(text):
    mapping = [
        ("mean", "Mean"),
        ("position index", "Position Index"),
        ("approx attention", "Approx attention"),
        ("attention", "Attention"),
        ("accuracy", "Accuracy"),
        ("loss", "Loss"),
        ("epoch", "Epoch"),
        ("precision", "Precision"),
        ("recall", "Recall"),
        ("f1", "F1-score"),
        ("alexnet", "AlexNet"),
        ("resnet", "ResNet"),
        ("vgg", "VGG"),
        ("googlenet", "GoogLeNet"),
        ("transformer", "Transformer"),
        ("cnn", "CNN"),
    ]

    comps = []
    for key, label in mapping:
        if key in text and label not in comps:
            comps.append(label)

    if "Approx attention" in comps and "Attention" in comps:
        comps.remove("Attention")

    return comps


def extract_architecture_components(text):
    mapping = [
        ("feature extractor", "Feature Extractor"),
        ("classifier", "Classifier"),
        ("encoder", "Encoder"),
        ("decoder", "Decoder"),
        ("attention", "Attention"),
        ("embedding", "Embedding"),
        ("block", "Block"),
        ("transformer", "Transformer"),
        ("alexnet", "AlexNet"),
        ("resnet", "ResNet"),
        ("vgg", "VGG"),
        ("googlenet", "GoogLeNet"),
        ("cnn", "CNN"),
    ]

    comps = []
    for key, label in mapping:
        if key in text and label not in comps:
            comps.append(label)

    return comps


def classify(ocr_text, visual):
    lower = ocr_text.lower()

    experiment_keywords = [
        "accuracy", "loss", "epoch", "precision", "recall", "f1",
        "mean", "position index", "t=", "performance", "comparison",
        "curve", "ablation", "baseline", "benchmark"
    ]

    architecture_keywords = [
        "cnn", "transformer", "attention", "encoder", "decoder",
        "block", "embedding", "classifier", "feature extractor",
        "alexnet", "resnet", "vgg", "googlenet"
    ]

    dataset_keywords = [
        "sample", "example", "input image", "class", "category",
        "dataset", "训练样本", "数据样本", "类别图片", "输入示例"
    ]

    # 冲突处理：experiment_result 优先级最高
    if any(keyword in lower for keyword in experiment_keywords):
        components = extract_experiment_components(lower)
        if "position index" in lower and ("attention" in lower or "mean" in lower):
            description = "展示不同时间步下注意力权重均值随位置索引的变化关系"
            code_info = "需要实现注意力权重统计与位置索引折线图绘制，分析不同T值的均值和波动。"
        else:
            description = "展示实验曲线、性能变化或指标对比"
            code_info = "需要实现结果可视化以及对应的指标计算。"

        return {
            "type": "experiment_result",
            "description": description,
            "important_components": components,
            "code_relevant_information": code_info,
        }

    if any(keyword in lower for keyword in architecture_keywords):
        components = extract_architecture_components(lower)
        return {
            "type": "architecture",
            "description": "展示模型网络结构和数据流关系",
            "important_components": components,
            "code_relevant_information": "需要实现模型组件和前向数据流。",
        }

    if any(keyword in lower for keyword in dataset_keywords):
        return {
            "type": "dataset",
            "description": "展示数据样本、类别图片或输入示例",
            "important_components": [],
            "code_relevant_information": "需要准备或加载对应数据集，并对输入样本进行预处理。",
        }

    # 简单视觉辅助：宽高比大且非白色像素少，可能为折线图/实验图
    if visual["aspect_ratio"] > 1.2 and visual["nonwhite_ratio"] < 0.3:
        return {
            "type": "experiment_result",
            "description": "根据视觉信息推断为实验曲线或趋势图",
            "important_components": [],
            "code_relevant_information": "需要实现趋势曲线绘制和统计分析。",
        }

    return {
        "type": "other",
        "description": "无法确定图片类型",
        "important_components": [],
        "code_relevant_information": "",
    }


def main():
    try:
        image_path = sys.argv[1]
        img = load_image_from_chinese_path(image_path)
        visual = get_basic_visual_info(img)
        result = classify(OCR_TEXT, visual)
        print(json.dumps(result, ensure_ascii=False, indent=2))
    except Exception:
        fallback = {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": "",
        }
        print(json.dumps(fallback, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
