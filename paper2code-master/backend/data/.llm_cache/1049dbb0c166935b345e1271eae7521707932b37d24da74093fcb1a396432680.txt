```python
import sys
import json
import numpy as np
import cv2
from PIL import Image
import os

def analyze_image(image_path):
    try:
        # OCR文字信息（外部提供）
        ocr_text = "output\nTape\nInput\nGr id"
        
        # 读取图片
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # 转换为灰度图进行简单分析
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        
        # 简单视觉信息：检查图片是否包含大量文本区域（可能是表格或数据展示）
        # 使用简单的阈值分析
        _, binary = cv2.threshold(gray, 200, 255, cv2.THRESH_BINARY_INV)
        
        # 计算非白色像素比例
        non_white_ratio = np.sum(binary > 0) / (binary.shape[0] * binary.shape[1])
        
        # 检查图片中是否有彩色元素（实验图表通常有彩色）
        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        saturation = hsv[:, :, 1]
        color_ratio = np.sum(saturation > 50) / (saturation.shape[0] * saturation.shape[1])
        
        # OCR关键词分析
        ocr_lower = ocr_text.lower()
        
        # 实验相关关键词
        experiment_keywords = [
            'accuracy', 'loss', 'epoch', 'precision', 'recall', 'f1',
            'performance', 'comparison', 'result', 'training', 'validation',
            'test', 'error', 'rate', 'score', 'metric', 'baseline',
            'experiment', 'evaluation', 'benchmark', 'convergence'
        ]
        
        # 架构相关关键词
        architecture_keywords = [
            'cnn', 'transformer', 'attention', 'encoder', 'decoder',
            'block', 'embedding', 'classifier', 'feature extractor',
            'alexnet', 'resnet', 'vgg', 'googlenet', 'layer',
            'network', 'architecture', 'module', 'input', 'output',
            'convolution', 'pooling', 'fully connected', 'softmax',
            'activation', 'normalization', 'dropout', 'batch norm'
        ]
        
        # 数据集相关关键词
        dataset_keywords = [
            'sample', 'example', 'dataset', 'class', 'category',
            'image', 'instance', 'label', 'annotation', 'ground truth',
            'training data', 'test data', 'input image', 'original'
        ]
        
        # 检查OCR中的关键词
        ocr_words = ocr_lower.split()
        
        experiment_score = 0
        architecture_score = 0
        dataset_score = 0
        
        for word in ocr_words:
            if word in experiment_keywords:
                experiment_score += 1
            if word in architecture_keywords:
                architecture_score += 1
            if word in dataset_keywords:
                dataset_score += 1
        
        # 检查OCR中的短语
        for keyword in experiment_keywords:
            if keyword in ocr_lower:
                experiment_score += 2
        
        for keyword in architecture_keywords:
            if keyword in ocr_lower:
                architecture_score += 2
        
        for keyword in dataset_keywords:
            if keyword in ocr_lower:
                dataset_score += 2
        
        # 根据OCR文本判断
        # OCR包含"output", "Tape", "Input", "Grid"
        # "Input"和"output"可能暗示架构图
        # "Grid"可能暗示数据展示
        
        # 检查是否有实验图表特征（彩色比例高）
        if color_ratio > 0.1:
            experiment_score += 1
        
        # 检查是否有大量文本（可能是表格）
        if non_white_ratio > 0.3:
            experiment_score += 1
        
        # 根据OCR具体内容调整
        if 'input' in ocr_lower and 'output' in ocr_lower:
            architecture_score += 3
        
        if 'grid' in ocr_lower:
            dataset_score += 2
        
        if 'tape' in ocr_lower:
            # Tape可能指磁带，可能与数据存储或序列数据相关
            dataset_score += 1
        
        # 确定类型
        # 优先级：experiment_result > architecture > dataset > other
        
        if experiment_score > architecture_score and experiment_score > dataset_score and experiment_score > 0:
            image_type = "experiment_result"
            description = "展示实验结果和性能指标"
            important_components = []
            code_relevant = "需要实现实验评估和结果可视化代码"
            
            # 提取可能的模型名称
            model_names = ['alexnet', 'resnet', 'vgg', 'googlenet', 'transformer', 'cnn']
            for model in model_names:
                if model in ocr_lower:
                    important_components.append(model.capitalize())
            
            # 提取指标名称
            metrics = ['accuracy', 'loss', 'precision', 'recall', 'f1']
            for metric in metrics:
                if metric in ocr_lower:
                    important_components.append(metric.capitalize())
            
            if not important_components:
                important_components = []
            
        elif architecture_score > dataset_score and architecture_score > 0:
            image_type = "architecture"
            description = "展示模型网络结构和数据流关系"
            important_components = []
            code_relevant = "需要实现网络结构模块和数据流处理"
            
            # 提取架构组件
            arch_components = ['cnn', 'transformer', 'attention', 'encoder', 'decoder',
                              'alexnet', 'resnet', 'vgg', 'googlenet', 'embedding',
                              'classifier', 'feature extractor']
            
            for comp in arch_components:
                if comp in ocr_lower:
                    if comp == 'cnn':
                        important_components.append('CNN')
                    elif comp == 'transformer':
                        important_components.append('Transformer')
                    elif comp == 'attention':
                        important_components.append('Attention Module')
                    elif comp == 'encoder':
                        important_components.append('Encoder')
                    elif comp == 'decoder':
                        important_components.append('Decoder')
                    elif comp == 'alexnet':
                        important_components.append('AlexNet')
                    elif comp == 'resnet':
                        important_components.append('ResNet')
                    elif comp == 'vgg':
                        important_components.append('VGG')
                    elif comp == 'googlenet':
                        important_components.append('GoogLeNet')
                    elif comp == 'embedding':
                        important_components.append('Embedding Layer')
                    elif comp == 'classifier':
                        important_components.append('Classifier')
                    elif comp == 'feature extractor':
                        important_components.append('Feature Extractor')
            
            if not important_components:
                important_components = []
            
        elif dataset_score > 0:
            image_type = "dataset"
            description = "展示数据样本和输入示例"
            important_components = []
            code_relevant = "需要实现数据加载和预处理代码"
            
        else:
            image_type = "other"
            description = "无法确定图片类型"
            important_components = []
            code_relevant = ""
        
        # 构建最终结果
        result = {
            "type": image_type,
            "description": description,
            "important_components": important_components,
            "code_relevant_information": code_relevant
        }
        
        return result
        
    except Exception as e:
        # 任何异常都返回other类型
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    try:
        if len(sys.argv) < 2:
            # 没有提供图片路径
            result = {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
        else:
            image_path = sys.argv[1]
            result = analyze_image(image_path)
        
        # 输出JSON
        print(json.dumps(result, ensure_ascii=False))
        
    except Exception as e:
        # 最终保底输出
        fallback = {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
        print(json.dumps(fallback, ensure_ascii=False))
```
