import sys
import json
import numpy as np
import cv2

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text provided externally
        ocr_text = "output Tape Input Gr id"
        ocr_lower = ocr_text.lower()
        
        # Architecture keywords
        arch_keywords = [
            "cnn", "transformer", "attention", "encoder", "decoder",
            "block", "embedding", "classifier", "feature extractor",
            "alexnet", "resnet", "vgg", "googlenet"
        ]
        
        # Experiment result keywords
        exp_keywords = [
            "accuracy", "loss", "epoch", "precision", "recall", "f1",
            "curve", "performance", "comparison"
        ]
        
        # Dataset keywords
        dataset_keywords = [
            "sample", "example", "input", "dataset", "class"
        ]
        
        # Check for experiment result first (highest priority)
        exp_matches = [kw for kw in exp_keywords if kw in ocr_lower]
        if exp_matches:
            components = []
            # Extract model names if present
            for kw in arch_keywords:
                if kw in ocr_lower:
                    components.append(kw.upper())
            # Add metric names
            for kw in exp_matches:
                components.append(kw.capitalize())
            
            return {
                "type": "experiment_result",
                "description": "展示实验结果和性能指标",
                "important_components": components[:5],
                "code_relevant_information": "需要实现实验评估和结果可视化功能"
            }
        
        # Check for architecture
        arch_matches = [kw for kw in arch_keywords if kw in ocr_lower]
        if arch_matches:
            components = []
            for kw in arch_matches:
                if kw in ["cnn", "transformer", "attention", "encoder", "decoder", "block", "embedding", "classifier"]:
                    components.append(kw.upper())
                else:
                    components.append(kw.capitalize())
            
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": components[:5],
                "code_relevant_information": "需要实现模型架构和特征提取模块"
            }
        
        # Check for dataset
        dataset_matches = [kw for kw in dataset_keywords if kw in ocr_lower]
        if dataset_matches:
            return {
                "type": "dataset",
                "description": "展示数据集样本或输入示例",
                "important_components": [],
                "code_relevant_information": "需要实现数据加载和预处理功能"
            }
        
        # Default case
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "图片分析失败",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
    else:
        image_path = r"D:\paper2code-master(1)\paper2code-master\backend\data\output\LEARNING_SIMPLE_ALGORITHMS_FROM_EXAMPLES\step1\images\image_5.png"
    
    result = analyze_image(image_path)
    print(json.dumps(result, ensure_ascii=False))
