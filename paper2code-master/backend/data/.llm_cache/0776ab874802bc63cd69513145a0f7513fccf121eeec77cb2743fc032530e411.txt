import sys
import os
import json
import numpy as np
import cv2


def extract_simple_visual_info(img):
    h, w = img.shape[:2]
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    saturation_mean = float(np.mean(hsv[:, :, 1]))
    brightness_mean = float(np.mean(hsv[:, :, 2]))

    white_mask = cv2.inRange(hsv, np.array([0, 0, 200]), np.array([180, 30, 255]))
    white_ratio = float(np.sum(white_mask > 0) / white_mask.size)

    color_mask = (hsv[:, :, 1] > 50) & (hsv[:, :, 2] > 30) & (hsv[:, :, 2] < 220)
    color_ratio = float(np.sum(color_mask) / color_mask.size)

    aspect_ratio = float(w / h) if h > 0 else 0.0

    return {
        "saturation_mean": saturation_mean,
        "brightness_mean": brightness_mean,
        "white_ratio": white_ratio,
        "color_ratio": color_ratio,
        "aspect_ratio": aspect_ratio,
        "height": h,
        "width": w
    }


def extract_components_from_text(text, candidates):
    if not text:
        return []
    found = []
    text_lower = text.lower()
    for item in candidates:
        if item.lower() in text_lower:
            found.append(item)
    return found


def classify(ocr_text, visual_info):
    architecture_keywords = [
        "CNN", "Transformer", "Attention", "Encoder", "Decoder", "Block",
        "Embedding", "Classifier", "Feature Extractor", "AlexNet", "ResNet",
        "VGG", "GoogLeNet", "Conv", "Convolution", "Layer", "Network",
        "Architecture", "Feature Map", "Module", "FC", "Softmax", "Relu",
        "Pooling", "Backbone", "Head", "Input", "Output"
    ]
    experiment_keywords = [
        "Accuracy", "Loss", "Epoch", "Precision", "Recall", "F1",
        "F1-score", "Score", "ROC", "AUC", "Validation", "Training",
        "Test", "Error", "Convergence", "Curve", "Comparison", "Benchmark",
        "Experiment", "mAP", "IoU", "AP", "AR", "Perplexity", "BLEU"
    ]
    dataset_keywords = [
        "Sample", "Example", "Dataset", "Class", "Label", "Category",
        "Instance", "CIFAR", "MNIST", "ImageNet", "Example images",
        "Input examples", "Ground truth", "Training samples", "Test samples"
    ]

    combined_text = ocr_text if ocr_text else ""

    if combined_text:
        if any(key.lower() in combined_text.lower() for key in experiment_keywords):
            comps = extract_components_from_text(
                combined_text,
                ["Accuracy", "Loss", "Epoch", "Precision", "Recall", "F1", "F1-score",
                 "AUC", "ROC", "mAP", "IoU", "Perplexity", "BLEU",
                 "AlexNet", "ResNet", "VGG", "GoogLeNet", "Transformer", "CNN",
                 "LSTM", "GRU", "RNN", "BERT", "GPT", "YOLO", "Faster R-CNN",
                 "Mask R-CNN", "U-Net"]
            )
            return {
                "type": "experiment_result",
                "description": "展示实验性能结果和指标对比",
                "important_components": comps,
                "code_relevant_information": "需要实现相关模型训练、评估和指标计算流程"
            }

        if any(key.lower() in combined_text.lower() for key in architecture_keywords):
            comps = extract_components_from_text(
                combined_text,
                ["AlexNet", "Transformer Encoder", "Attention Module", "ResNet", "VGG",
                 "GoogLeNet", "CNN", "Transformer", "Attention", "Encoder", "Decoder",
                 "Block", "Embedding", "Classifier", "Feature Extractor",
                 "Feature Map", "Module", "Backbone", "Head"]
            )
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": comps,
                "code_relevant_information": "需要实现模型前向传播结构和各模块"
            }

        if any(key.lower() in combined_text.lower() for key in dataset_keywords):
            ds_comps = extract_components_from_text(
                combined_text,
                ["CIFAR", "MNIST", "ImageNet", "COCO", "Pascal VOC", "SVHN"]
            )
            return {
                "type": "dataset",
                "description": "展示数据集样本或类别示例",
                "important_components": ds_comps,
                "code_relevant_information": "需要准备数据加载和预处理"
            }

    sat = visual_info.get("saturation_mean", 0)
    white_ratio = visual_info.get("white_ratio", 0)
    color_ratio = visual_info.get("color_ratio", 0)

    if white_ratio > 0.3 and color_ratio > 0.1 and sat > 30:
        return {
            "type": "experiment_result",
            "description": "展示实验曲线或对比结果",
            "important_components": [],
            "code_relevant_information": "需要实现模型评估和指标计算"
        }

    if white_ratio > 0.3 and color_ratio < 0.2 and sat < 50:
        return {
            "type": "architecture",
            "description": "展示模型网络结构和数据流关系",
            "important_components": [],
            "code_relevant_information": "需要实现模型前向传播结构"
        }

    if sat > 60 and color_ratio > 0.3:
        return {
            "type": "dataset",
            "description": "展示数据集样本或类别示例",
            "important_components": [],
            "code_relevant_information": "需要准备数据加载和预处理"
        }

    return {
        "type": "other",
        "description": "无法确定图片类型",
        "important_components": [],
        "code_relevant_information": ""
    }


def load_image(image_path):
    data = np.fromfile(image_path, dtype=np.uint8)
    img = cv2.imdecode(data, cv2.IMREAD_COLOR)
    if img is None:
        raise ValueError("无法读取图片")
    return img


def main():
    default_result = {
        "type": "other",
        "description": "无法确定图片类型",
        "important_components": [],
        "code_relevant_information": ""
    }

    use_ascii = False
    try:
        sys.stdout.reconfigure(encoding='utf-8')
    except Exception:
        use_ascii = True

    try:
        if len(sys.argv) < 2:
            raise ValueError("缺少图片路径参数")
        image_path = sys.argv[1]
        ocr_text = os.environ.get("OCR_TEXT", "")

        img = load_image(image_path)
        visual_info = extract_simple_visual_info(img)
        result = classify(ocr_text, visual_info)
        print(json.dumps(result, ensure_ascii=use_ascii))
    except Exception:
        print(json.dumps(default_result, ensure_ascii=use_ascii))


if __name__ == "__main__":
    main()
