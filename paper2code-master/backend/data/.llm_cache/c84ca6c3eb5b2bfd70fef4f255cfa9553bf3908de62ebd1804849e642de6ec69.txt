import sys
import json
import numpy as np
import cv2


def load_image(path):
    data = np.fromfile(path, dtype=np.uint8)
    img = cv2.imdecode(data, cv2.IMREAD_COLOR)
    if img is None:
        raise ValueError("无法读取图片")
    return cv2.cvtColor(img, cv2.COLOR_BGR2RGB)


def extract_visual_features(rgb):
    hsv = cv2.cvtColor(rgb, cv2.COLOR_RGB2HSV)

    white_mask = np.all(rgb > 215, axis=2)
    white_ratio = float(np.mean(white_mask))
    non_white_ratio = 1.0 - white_ratio

    sat = hsv[:, :, 1].astype(np.float32)
    val = hsv[:, :, 2].astype(np.float32)
    color_mask = (sat > 50) & (val > 60)
    color_ratio = float(np.mean(color_mask))

    hue = hsv[:, :, 0][color_mask]
    if len(hue) > 0:
        hue_quantized = (hue // 18).astype(np.int32)
        hue_count = int(len(np.unique(hue_quantized)))
    else:
        hue_count = 0

    q = (rgb // 32).astype(np.uint8)
    color_index = (
        q[:, :, 0].astype(np.int32) * 100 +
        q[:, :, 1].astype(np.int32) * 10 +
        q[:, :, 2].astype(np.int32)
    )
    unique_colors = int(len(np.unique(color_index)))

    h, w, _ = rgb.shape
    block_h = max(1, h // 6)
    block_w = max(1, w // 6)
    blocks = []
    for i in range(6):
        for j in range(6):
            block = rgb[i * block_h:(i + 1) * block_h, j * block_w:(j + 1) * block_w]
            if block.size == 0:
                continue
            mean_color = block.reshape(-1, 3).mean(axis=0)
            blocks.append(mean_color)

    if len(blocks) > 1:
        blocks = np.array(blocks)
        diff_sum = 0.0
        count = 0
        for i in range(len(blocks)):
            for j in range(i + 1, len(blocks)):
                diff_sum += float(np.linalg.norm(blocks[i] - blocks[j]))
                count += 1
        block_diff = diff_sum / count if count > 0 else 0.0
    else:
        block_diff = 0.0

    gray = cv2.cvtColor(rgb, cv2.COLOR_RGB2GRAY)
    gx = cv2.Sobel(gray, cv2.CV_64F, 1, 0, ksize=3)
    gy = cv2.Sobel(gray, cv2.CV_64F, 0, 1, ksize=3)
    mag = np.sqrt(gx * gx + gy * gy)
    gradient_density = float(np.mean(mag > 40))

    return {
        "white_ratio": white_ratio,
        "non_white_ratio": non_white_ratio,
        "color_ratio": color_ratio,
        "hue_count": hue_count,
        "unique_colors": unique_colors,
        "block_diff": block_diff,
        "gradient_density": gradient_density,
    }


def classify_visual(features):
    if (
        features["non_white_ratio"] > 0.30 and
        features["color_ratio"] > 0.08 and
        features["hue_count"] >= 5 and
        features["block_diff"] > 15.0
    ):
        return "dataset"

    if (
        features["white_ratio"] > 0.40 and
        features["color_ratio"] > 0.01 and
        features["hue_count"] >= 2 and
        features["non_white_ratio"] > 0.03
    ):
        return "experiment_result"

    if (
        features["white_ratio"] > 0.40 and
        features["color_ratio"] > 0.08 and
        features["hue_count"] >= 4
    ):
        return "experiment_result"

    if (
        features["white_ratio"] > 0.50 and
        features["non_white_ratio"] > 0.02 and
        features["gradient_density"] > 0.005
    ):
        if features["color_ratio"] > 0.01:
            return "experiment_result"
        return "architecture"

    if (
        features["white_ratio"] > 0.50 and
        features["non_white_ratio"] > 0.02 and
        features["color_ratio"] < 0.05
    ):
        return "architecture"

    if features["color_ratio"] > 0.15:
        return "dataset"

    return "other"


def classify_with_ocr(ocr_text):
    if not ocr_text:
        return None

    experiment_keywords = [
        "Accuracy", "Loss", "Epoch", "Precision", "Recall", "F1",
        "accuracy", "loss", "epoch", "precision", "recall",
        "F1-score", "comparison", "准确率", "损失", "曲线", "性能"
    ]
    architecture_keywords = [
        "CNN", "Transformer", "Attention", "Encoder", "Decoder",
        "Block", "Embedding", "Classifier", "Feature Extractor",
        "AlexNet", "ResNet", "VGG", "GoogLeNet"
    ]
    dataset_keywords = [
        "sample", "example", "input", "dataset", "样例", "样本", "输入示例", "类别"
    ]

    if any(k in ocr_text for k in experiment_keywords):
        return "experiment_result"
    if any(k in ocr_text for k in architecture_keywords):
        return "architecture"
    if any(k in ocr_text for k in dataset_keywords):
        return "dataset"
    return None


def main():
    try:
        image_path = sys.argv[1]
        ocr_text = ""

        rgb = load_image(image_path)
        features = extract_visual_features(rgb)

        ocr_type = classify_with_ocr(ocr_text)
        if ocr_type is not None:
            final_type = ocr_type
        else:
            final_type = classify_visual(features)

        if final_type == "experiment_result":
            description = "展示实验性能对比结果，包含准确率、损失或F1等指标"
            important_components = []
            code_relevant_information = "需要实现模型训练与评估逻辑，输出准确率、Loss等指标进行对比"
        elif final_type == "architecture":
            description = "展示模型网络结构和数据流关系"
            important_components = []
            code_relevant_information = "需要实现模型网络结构，包括特征提取模块和分类头"
        elif final_type == "dataset":
            description = "展示数据样本或输入示例"
            important_components = []
            code_relevant_information = "需要准备数据加载和预处理流程"
        else:
            description = "无法确定图片类型"
            important_components = []
            code_relevant_information = ""

        result = {
            "type": final_type,
            "description": description,
            "important_components": important_components,
            "code_relevant_information": code_relevant_information
        }
        print(json.dumps(result, ensure_ascii=False))
    except Exception:
        fallback = {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
        print(json.dumps(fallback, ensure_ascii=False))


if __name__ == "__main__":
    main()
