"""Evaluation metrics for SELF-RAG.

This module implements the evaluation metrics described in the paper
"SELF-RAG: Learning to Retrieve, Generate, and Critique through Self-Reflection".
Metrics include Accuracy, FactScore, strem (correctness), rouge (correctness),
MAUVE (fluency), citation precision, and citation recall.
"""

from __future__ import annotations

import logging
import math
import re
from collections import Counter
from typing import Any, Dict, List, Optional, Sequence, Tuple

import numpy as np

logger = logging.getLogger(__name__)

try:
    from rouge_score import rouge_scorer
except ImportError:
    logger.warning(
        "rouge_score not installed. ROUGE metric will be unavailable. "
        "Install with: pip install rouge-score"
    )
    rouge_scorer = None

try:
    from mauve import compute_mauve as _mauve_compute
except ImportError:
    logger.warning(
        "mauve not installed. MAUVE metric will be unavailable. "
        "Install with: pip install mauve-text"
    )
    _mauve_compute = None


def _safe_divide(numerator: float, denominator: float) -> float:
    """Safely divide two numbers, returning 0.0 if denominator is zero."""
    if denominator == 0:
        return 0.0
    return numerator / denominator


def _normalize_text(text: str) -> str:
    """Normalize text for metric computation."""
    if not text:
        return ""
    text = text.lower().strip()
    text = re.sub(r"\s+", " ", text)
    return text


def _tokenize(text: str) -> List[str]:
    """Simple whitespace tokenization."""
    return _normalize_text(text).split()


def compute_accuracy(
    predictions: Sequence[str],
    references: Sequence[str],
    threshold: float = 0.5,
) -> float:
    """Compute token-level accuracy between predictions and references.

    Args:
        predictions: List of generated text sequences.
        references: List of reference text sequences.
        threshold: Not used for accuracy but kept for API consistency.

    Returns:
        Accuracy score between 0 and 1.
    """
    if len(predictions) != len(references) or len(predictions) == 0:
        return 0.0

    total_tokens = 0
    correct_tokens = 0

    for pred, ref in zip(predictions, references):
        pred_tokens = _tokenize(pred)
        ref_tokens = _tokenize(ref)

        if len(ref_tokens) == 0:
            continue

        total_tokens += len(ref_tokens)
        # Compare token by token, padding predictions if needed
        for i, ref_token in enumerate(ref_tokens):
            if i < len(pred_tokens) and pred_tokens[i] == ref_token:
                correct_tokens += 1

    return _safe_divide(correct_tokens, total_tokens)


def compute_factscore(
    predictions: Sequence[str],
    references: Sequence[str],
    threshold: float = 0.5,
) -> float:
    """Compute FactScore - the proportion of atomic facts in the prediction
    that are supported by the reference.

    This is a simplified implementation that checks for key phrase overlap
    between prediction and reference as a proxy for factual consistency.

    Args:
        predictions: List of generated text sequences.
        references: List of reference text sequences.
        threshold: Threshold for considering a fact supported.

    Returns:
        FactScore between 0 and 1.
    """
    if len(predictions) != len(references) or len(predictions) == 0:
        return 0.0

    scores = []
    for pred, ref in zip(predictions, references):
        pred_norm = _normalize_text(pred)
        ref_norm = _normalize_text(ref)

        if not pred_norm or not ref_norm:
            scores.append(0.0)
            continue

        # Extract "atomic facts" as noun phrases or key n-grams
        pred_facts = _extract_facts(pred_norm)
        ref_facts = _extract_facts(ref_norm)

        if not pred_facts:
            scores.append(0.0)
            continue

        supported = 0
        for fact in pred_facts:
            # Check if the fact appears in the reference
            if fact in ref_norm:
                supported += 1

        scores.append(_safe_divide(supported, len(pred_facts)))

    return float(np.mean(scores)) if scores else 0.0


def _extract_facts(text: str) -> List[str]:
    """Extract atomic facts from text using noun phrase heuristics.

    This is a simplified fact extraction that splits on conjunctions and
    extracts key phrases.
    """
    facts = []
    # Split on common conjunctions and punctuation
    parts = re.split(r"\b(?:and|but|however|therefore|thus|while|although)\b|[,;]", text)

    for part in parts:
        part = part.strip()
        if len(part.split()) >= 3:  # Require at least 3 words for a fact
            facts.append(part)

    return facts


def compute_strem(
    predictions: Sequence[str],
    references: Sequence[str],
    threshold: float = 0.5,
) -> float:
    """Compute STREM (Semantic Textual Relatedness with Embedding Matching).

    This is a simplified implementation using token overlap as a proxy for
    semantic relatedness.

    Args:
        predictions: List of generated text sequences.
        references: List of reference text sequences.
        threshold: Threshold for considering a match.

    Returns:
        STREM score between 0 and 1.
    """
    if len(predictions) != len(references) or len(predictions) == 0:
        return 0.0

    scores = []
    for pred, ref in zip(predictions, references):
        pred_tokens = set(_tokenize(pred))
        ref_tokens = set(_tokenize(ref))

        if not pred_tokens or not ref_tokens:
            scores.append(0.0)
            continue

        # Jaccard similarity as a proxy for semantic relatedness
        intersection = pred_tokens.intersection(ref_tokens)
        union = pred_tokens.union(ref_tokens)
        jaccard = _safe_divide(len(intersection), len(union))

        scores.append(jaccard)

    return float(np.mean(scores)) if scores else 0.0


def compute_rouge(
    predictions: Sequence[str],
    references: Sequence[str],
    threshold: float = 0.5,
) -> Dict[str, float]:
    """Compute ROUGE scores (ROUGE-1, ROUGE-2, ROUGE-L).

    Args:
        predictions: List of generated text sequences.
        references: List of reference text sequences.
        threshold: Not used for ROUGE but kept for API consistency.

    Returns:
        Dictionary with ROUGE-1, ROUGE-2, ROUGE-L F1 scores.
    """
    if len(predictions) != len(references) or len(predictions) == 0:
        return {"rouge1": 0.0, "rouge2": 0.0, "rougeL": 0.0}

    if rouge_scorer is None:
        # Fallback to simple n-gram overlap
        return _compute_rouge_fallback(predictions, references)

    scorer = rouge_scorer.RougeScorer(["rouge1", "rouge2", "rougeL"], use_stemmer=True)

    rouge1_scores = []
    rouge2_scores = []
    rougeL_scores = []

    for pred, ref in zip(predictions, references):
        if not pred or not ref:
            rouge1_scores.append(0.0)
            rouge2_scores.append(0.0)
            rougeL_scores.append(0.0)
            continue

        scores = scorer.score(ref, pred)
        rouge1_scores.append(scores["rouge1"].fmeasure)
        rouge2_scores.append(scores["rouge2"].fmeasure)
        rougeL_scores.append(scores["rougeL"].fmeasure)

    return {
        "rouge1": float(np.mean(rouge1_scores)) if rouge1_scores else 0.0,
        "rouge2": float(np.mean(rouge2_scores)) if rouge2_scores else 0.0,
        "rougeL": float(np.mean(rougeL_scores)) if rougeL_scores else 0.0,
    }


def _compute_rouge_fallback(
    predictions: Sequence[str],
    references: Sequence[str],
) -> Dict[str, float]:
    """Fallback ROUGE computation using n-gram overlap."""
    rouge1_scores = []
    rouge2_scores = []
    rougeL_scores = []

    for pred, ref in zip(predictions, references):
        pred_tokens = _tokenize(pred)
        ref_tokens = _tokenize(ref)

        if not pred_tokens or not ref_tokens:
            rouge1_scores.append(0.0)
            rouge2_scores.append(0.0)
            rougeL_scores.append(0.0)
            continue

        # ROUGE-1: unigram overlap
        pred_counter = Counter(pred_tokens)
        ref_counter = Counter(ref_tokens)
        overlap_1 = sum((pred_counter & ref_counter).values())
        rouge1 = _safe_divide(2 * overlap_1, len(pred_tokens) + len(ref_tokens))
        rouge1_scores.append(rouge1)

        # ROUGE-2: bigram overlap
        pred_bigrams = _get_ngrams(pred_tokens, 2)
        ref_bigrams = _get_ngrams(ref_tokens, 2)
        if pred_bigrams and ref_bigrams:
            pred_bigram_counter = Counter(pred_bigrams)
            ref_bigram_counter = Counter(ref_bigrams)
            overlap_2 = sum((pred_bigram_counter & ref_bigram_counter).values())
            rouge2 = _safe_divide(
                2 * overlap_2, len(pred_bigrams) + len(ref_bigrams)
            )
        else:
            rouge2 = 0.0
        rouge2_scores.append(rouge2)

        # ROUGE-L: longest common subsequence
        rougeL = _lcs_ratio(pred_tokens, ref_tokens)
        rougeL_scores.append(rougeL)

    return {
        "rouge1": float(np.mean(rouge1_scores)) if rouge1_scores else 0.0,
        "rouge2": float(np.mean(rouge2_scores)) if rouge2_scores else 0.0,
        "rougeL": float(np.mean(rougeL_scores)) if rougeL_scores else 0.0,
    }


def _get_ngrams(tokens: List[str], n: int) -> List[Tuple[str, ...]]:
    """Get n-grams from a token list."""
    if len(tokens) < n:
        return []
    return [tuple(tokens[i : i + n]) for i in range(len(tokens) - n + 1)]


def _lcs_ratio(tokens1: List[str], tokens2: List[str]) -> float:
    """Compute the ratio of the longest common subsequence."""
    if not tokens1 or not tokens2:
        return 0.0

    # DP for LCS
    dp = [[0] * (len(tokens2) + 1) for _ in range(len(tokens1) + 1)]
    for i in range(1, len(tokens1) + 1):
        for j in range(1, len(tokens2) + 1):
            if tokens1[i - 1] == tokens2[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])

    lcs_length = dp[len(tokens1)][len(tokens2)]
    return _safe_divide(2 * lcs_length, len(tokens1) + len(tokens2))


def compute_mauve(
    predictions: Sequence[str],
    references: Sequence[str],
    threshold: float = 0.5,
) -> float:
    """Compute MAUVE score for fluency evaluation.

    Args:
        predictions: List of generated text sequences.
        references: List of reference text sequences.
        threshold: Not used for MAUVE but kept for API consistency.

    Returns:
        MAUVE score between 0 and 1.
    """
    if len(predictions) == 0 or len(references) == 0:
        return 0.0

    if _mauve_compute is None:
        logger.warning("MAUVE not available. Returning 0.0.")
        return 0.0

    try:
        # MAUVE requires non-empty texts
        valid_preds = [p for p in predictions if p and p.strip()]
        valid_refs = [r for r in references if r and r.strip()]

        if not valid_preds or not valid_refs:
            return 0.0

        result = _mauve_compute(
            p_text=valid_preds,
            q_text=valid_refs,
            device_id=-1,  # CPU
            max_text_length=1024,
            verbose=False,
        )
        return float(result.mauve)
    except Exception as e:
        logger.warning(f"MAUVE computation failed: {e}")
        return 0.0


def compute_citation_precision(
    predictions: Sequence[str],
    references: Sequence[str],
    citations: Optional[Sequence[Sequence[str]]] = None,
    threshold: float = 0.5,
) -> float:
    """Compute citation precision - proportion of citations that are correct.

    A citation is considered correct if the cited text appears in the reference.

    Args:
        predictions: List of generated text sequences.
        references: List of reference text sequences.
        citations: List of lists of citation strings for each prediction.
        threshold: Threshold for considering a citation correct.

    Returns:
        Citation precision between 0 and 1.
    """
    if len(predictions) != len(references) or len(predictions) == 0:
        return 0.0

    if citations is None:
        # Extract citations from predictions using bracket notation
        citations = [_extract_citations(pred) for pred in predictions]

    total_citations = 0
    correct_citations = 0

    for pred, ref, pred_citations in zip(predictions, references, citations):
        ref_norm = _normalize_text(ref)

        for citation in pred_citations:
            total_citations += 1
            citation_norm = _normalize_text(citation)
            if citation_norm and citation_norm in ref_norm:
                correct_citations += 1

    return _safe_divide(correct_citations, total_citations)


def compute_citation_recall(
    predictions: Sequence[str],
    references: Sequence[str],
    citations: Optional[Sequence[Sequence[str]]] = None,
    threshold: float = 0.5,
) -> float:
    """Compute citation recall - proportion of reference facts that are cited.

    Args:
        predictions: List of generated text sequences.
        references: List of reference text sequences.
        citations: List of lists of citation strings for each prediction.
        threshold: Threshold for considering a citation correct.

    Returns:
        Citation recall between 0 and 1.
    """
    if len(predictions) != len(references) or len(predictions) == 0:
        return 0.0

    if citations is None:
        citations = [_extract_citations(pred) for pred in predictions]

    total_facts = 0
    cited_facts = 0

    for pred, ref, pred_citations in zip(predictions, references, citations):
        ref_facts = _extract_facts(_normalize_text(ref))
        citation_texts = [_normalize_text(c) for c in pred_citations]

        for fact in ref_facts:
            total_facts += 1
            # Check if any citation covers this fact
            for citation in citation_texts:
                if citation and (fact in citation or citation in fact):
                    cited_facts += 1
                    break

    return _safe_divide(cited_facts, total_facts)


def _extract_citations(text: str) -> List[str]:
    """Extract citation strings from text.

    Citations are assumed to be in brackets like [citation] or [1].
    """
    citations = re.findall(r"\[([^\]]+)\]", text)
    # Filter out pure numbers (likely reference indices)
    return [c for c in citations if not c.strip().isdigit()]


def compute_metrics(
    predictions: Sequence[str],
    references: Sequence[str],
    citations: Optional[Sequence[Sequence[str]]] = None,
    config: Optional[Dict[str, Any]] = None,
    threshold: float = 0.5,
    batch_size: int = 32,
) -> Dict[str, float]:
    """Compute all evaluation metrics for SELF-RAG.

    Args:
        predictions: List of generated text sequences.
        references: List of reference text sequences.
        citations: Optional list of citation strings for each prediction.
        config: Configuration dictionary (optional).
        threshold: Threshold for binary metrics.
        batch_size: Batch size for processing (kept for API consistency).

    Returns:
        Dictionary containing all computed metrics.
    """
    if len(predictions) != len(references):
        logger.error(
            f"Predictions and references must have the same length. "
            f"Got {len(predictions)} and {len(references)}."
        )
        return {}

    if len(predictions) == 0:
        logger.warning("Empty predictions and references. Returning empty metrics.")
        return {}

    metrics: Dict[str, float] = {}

    # Accuracy
    metrics["accuracy"] = compute_accuracy(predictions, references, threshold)

    # FactScore
    metrics["factscore"] = compute_factscore(predictions, references, threshold)

    # STREM (correctness)
    metrics["strem"] = compute_strem(predictions, references, threshold)

    # ROUGE (correctness)
    rouge_scores = compute_rouge(predictions, references, threshold)
    metrics.update(rouge_scores)

    # MAUVE (fluency)
    metrics["mauve"] = compute_mauve(predictions, references, threshold)

    # Citation metrics
    metrics["citation_precision"] = compute_citation_precision(
        predictions, references, citations, threshold
    )
    metrics["citation_recall"] = compute_citation_recall(
        predictions, references, citations, threshold
    )

    # Log metrics
    logger.info("Computed metrics:")
    for name, value in metrics.items():
        logger.info(f"  {name}: {value:.4f}")

    return metrics
