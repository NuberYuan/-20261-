import sys
import json
import os
import numpy as np
import cv2

def analyze_image(image_path):
    # Default output
    result = {
        "type": "other",
        "description": "无法确定图片类型",
        "important_components": [],
        "code_relevant_information": ""
    }
    
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            result["description"] = "图片读取失败"
            return result
        
        # OCR text provided externally
        ocr_text = """output
Tape
8 41
Input
Gr id"""
        
        ocr_lower = ocr_text.lower()
        
        # Architecture keywords
        arch_keywords = [
            "cnn", "transformer", "attention", "encoder", "decoder",
            "block", "embedding", "classifier", "feature extractor",
            "alexnet", "resnet", "vgg", "googlenet"
        ]
        
        # Experiment result keywords
        exp_keywords = [
            "accuracy", "loss", "epoch", "precision", "recall", "f1",
            "curve", "performance", "comparison"
        ]
        
        # Dataset keywords
        dataset_keywords = [
            "sample", "example", "input image", "dataset", "class"
        ]
        
        # Check for experiment result first (highest priority)
        exp_found = [kw for kw in exp_keywords if kw in ocr_lower]
        if exp_found:
            result["type"] = "experiment_result"
            result["description"] = "展示实验结果或性能对比"
            result["important_components"] = []
            result["code_relevant_information"] = "需要实现实验评估和结果可视化功能"
            return result
        
        # Check for architecture
        arch_found = [kw for kw in arch_keywords if kw in ocr_lower]
        if arch_found:
            result["type"] = "architecture"
            result["description"] = "展示模型网络结构和数据流关系"
            result["important_components"] = arch_found
            result["code_relevant_information"] = "需要实现模型网络结构搭建"
            return result
        
        # Check for dataset
        dataset_found = [kw for kw in dataset_keywords if kw in ocr_lower]
        if dataset_found:
            result["type"] = "dataset"
            result["description"] = "展示数据集样本或输入示例"
            result["important_components"] = []
            result["code_relevant_information"] = "需要实现数据加载和预处理功能"
            return result
        
        # If OCR doesn't help, use basic visual info
        height, width = img.shape[:2]
        
        # Check if image is mostly uniform (likely a simple diagram)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        std_dev = np.std(gray)
        
        if std_dev < 30:
            # Very uniform image, likely a simple diagram or text
            result["type"] = "other"
            result["description"] = "图片内容过于简单，无法确定具体类型"
            result["important_components"] = []
            result["code_relevant_information"] = ""
        else:
            # Check color distribution
            hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
            saturation = np.mean(hsv[:,:,1])
            
            if saturation > 100:
                # High saturation might indicate data visualization
                result["type"] = "experiment_result"
                result["description"] = "可能包含实验数据可视化"
                result["important_components"] = []
                result["code_relevant_information"] = "需要实现数据可视化功能"
            else:
                result["type"] = "other"
                result["description"] = "无法确定图片类型"
                result["important_components"] = []
                result["code_relevant_information"] = ""
    
    except Exception as e:
        result["type"] = "other"
        result["description"] = f"图片分析出错: {str(e)}"
        result["important_components"] = []
        result["code_relevant_information"] = ""
    
    return result

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
    else:
        image_path = r"D:\paper2code-master(1)\paper2code-master\backend\data\output\LEARNING_SIMPLE_ALGORITHMS_FROM_EXAMPLES\step1\images\image_7.png"
    
    result = analyze_image(image_path)
    print(json.dumps(result, ensure_ascii=False))
