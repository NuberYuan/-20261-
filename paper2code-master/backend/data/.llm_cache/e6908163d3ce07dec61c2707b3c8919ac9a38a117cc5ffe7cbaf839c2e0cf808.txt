def load_config(config_path=None):
    """
    Load configuration from YAML file.

    Args:
        config_path (str, optional): Path to config file.
            If None, defaults to 'config/config.yaml' relative to project root.

    Returns:
        dict: Configuration dictionary.
    """
    if config_path is None:
        config_path = PROJECT_ROOT / 'config' / 'config.yaml'

    try:
        with open(config_path, 'r', encoding='utf-8') as f:
            config = yaml.safe_load(f)
        return config
    except FileNotFoundError:
        logging.warning(f"Config file not found at {config_path}. Using empty config.")
        return {}
    except Exception as e:
        logging.warning(f"Error loading config from {config_path}: {e}. Using empty config.")
        return {}


def _sigmoid(x):
    """Numerically stable sigmoid function."""
    return 1.0 / (1.0 + np.exp(-np.clip(x, -500, 500)))


def _compute_confusion_matrix(preds, targets, threshold=0.5):
    """
    Compute confusion matrix for binary classification.

    Args:
        preds (np.ndarray): Model predictions (probabilities or logits).
        targets (np.ndarray): Ground truth labels.
        threshold (float): Threshold for binarizing predictions.

    Returns:
        np.ndarray: 2x2 confusion matrix [[TN, FP], [FN, TP]].
    """
    # Binarize predictions
    preds_bin = (preds >= threshold).astype(np.int32)
    targets_bin = targets.astype(np.int32)

    # Flatten
    preds_flat = preds_bin.flatten()
    targets_flat = targets_bin.flatten()

    # Compute confusion matrix elements
    tn = np.sum((preds_flat == 0) & (targets_flat == 0))
    fp = np.sum((preds_flat == 1) & (targets_flat == 0))
    fn = np.sum((preds_flat == 0) & (targets_flat == 1))
    tp = np.sum((preds_flat == 1) & (targets_flat == 1))

    return np.array([[tn, fp], [fn, tp]])


def _compute_classification_metrics(confusion_matrix, eps=1e-7):
    """
    Compute classification metrics from confusion matrix.

    Args:
        confusion_matrix (np.ndarray): 2x2 confusion matrix.
        eps (float): Small epsilon to avoid division by zero.

    Returns:
        dict: Dictionary containing accuracy, precision, recall, f1, specificity.
    """
    tn, fp, fn, tp = confusion_matrix[0, 0], confusion_matrix[0, 1], confusion_matrix[1, 0], confusion_matrix[1, 1]

    accuracy = (tp + tn) / (tp + tn + fp + fn + eps)
    precision = tp / (tp + fp + eps)
    recall = tp / (tp + fn + eps)
    f1 = 2.0 * precision * recall / (precision + recall + eps)
    specificity = tn / (tn + fp + eps)

    return {
        'accuracy': float(accuracy),
        'precision': float(precision),
        'recall': float(recall),
        'f1': float(f1),
        'specificity': float(specificity)
    }


def evaluate_model(model, test_loader, device=None, threshold=0.5, config=None):
    """
    Evaluate a trained model on a test dataset.

    Args:
        model (nn.Module): Trained model to evaluate.
        test_loader (DataLoader): Test data loader.
        device (torch.device, optional): Device to run evaluation on.
            If None, uses CUDA if available, otherwise CPU.
        threshold (float): Threshold for binarizing predictions. Default: 0.5.
        config (dict, optional): Configuration dictionary.

    Returns:
        dict: Dictionary containing evaluation metrics including:
            - accuracy: Overall accuracy
            - precision: Precision (positive predictive value)
            - recall: Recall (sensitivity)
            - f1: F1 score
            - specificity: Specificity
            - iou: Intersection over Union
            - dice: Dice coefficient
            - confusion_matrix: 2x2 confusion matrix
            - inference_time: Total inference time in seconds
            - num_samples: Number of samples evaluated
    """
    if config is None:
        config = load_config()

    if device is None:
        device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

    model = model.to(device)
    model.eval()

    all_preds = []
    all_targets = []

    total_inference_time = 0.0
    num_samples = 0

    with torch.no_grad():
        for inputs, targets in test_loader:
            inputs = inputs.to(device)
            targets = targets.to(device)

            # Ensure targets are float
            if targets.dim() == 1:
                targets = targets.float().unsqueeze(1)
            else:
                targets = targets.float()

            # Measure inference time
            start_time = time.time()
            outputs = model(inputs)
            end_time = time.time()
            total_inference_time += (end_time - start_time)

            # Apply sigmoid to get probabilities
            probs = torch.sigmoid(outputs)

            # Move to CPU and convert to numpy
            probs_np = probs.cpu().numpy()
            targets_np = targets.cpu().numpy()

            all_preds.append(probs_np)
            all_targets.append(targets_np)

            num_samples += inputs.size(0)

    # Concatenate all predictions and targets
    all_preds = np.concatenate(all_preds, axis=0)
    all_targets = np.concatenate(all_targets, axis=0)

    # Compute confusion matrix
    confusion_matrix = _compute_confusion_matrix(all_preds, all_targets, threshold)

    # Compute classification metrics
    classification_metrics = _compute_classification_metrics(confusion_matrix)

    # Compute IoU and Dice using the metrics utility
    # For binary segmentation, we compute per-pixel metrics
    preds_bin = (all_preds >= threshold).astype(np.uint8)
    targets_bin = (all_targets >= threshold).astype(np.uint8)

    # Flatten for metric computation
    preds_flat = preds_bin.reshape(-1)
    targets_flat = targets_bin.reshape(-1)

    # Compute IoU and Dice
    intersection = np.sum(preds_flat * targets_flat)
    union = np.sum(preds_flat) + np.sum(targets_flat) - intersection
    iou = (intersection + 1e-7) / (union + 1e-7)
    dice = (2.0 * intersection + 1e-7) / (np.sum(preds_flat) + np.sum(targets_flat) + 1e-7)

    # Compile all metrics
    metrics = {
        'accuracy': classification_metrics['accuracy'],
        'precision': classification_metrics['precision'],
        'recall': classification_metrics['recall'],
        'f1': classification_metrics['f1'],
        'specificity': classification_metrics['specificity'],
        'iou': float(iou),
        'dice': float(dice),
        'confusion_matrix': confusion_matrix,
        'inference_time': total_inference_time,
        'num_samples': num_samples,
        'threshold': threshold
    }

    return metrics


def _generate_metrics_bar_chart(metrics, output_path):
    """
    Generate a bar chart of key metrics.

    Args:
        metrics (dict): Complete metrics dictionary containing
            'mean_f1', 'precision', 'recall', 'accuracy' keys.
        output_path (str): Path to save the chart.

    Returns:
        str or None: Path to the saved chart, or None if matplotlib is unavailable.
    """
    if plt is None:
        return None

    # Extract metrics - use the full metrics dict
    metric_names = ['mean_f1', 'precision', 'recall', 'accuracy']
    metric_values = []

    for name in metric_names:
        if name in metrics:
            metric_values.append(metrics[name])
        else:
            metric_values.append(0.0)

    # Create bar chart
    fig, ax = plt.subplots(figsize=(8, 6))
    bars = ax.bar(metric_names, metric_values, color=['#2E86AB', '#A23B72', '#F18F01', '#C73E1D'])
    ax.set_ylabel('Score')
    ax.set_title('Model Evaluation Metrics')
    ax.set_ylim([0, 1.0])

    # Add value labels on bars
    for bar, value in zip(bars, metric_values):
        height = bar.get_height()
        ax.text(bar.get_x() + bar.get_width() / 2., height + 0.01,
                f'{value:.4f}', ha='center', va='bottom', fontsize=10)

    # Add grid
    ax.yaxis.grid(True, alpha=0.3)
    ax.set_axisbelow(True)

    plt.tight_layout()
    plt.savefig(output_path, dpi=150, bbox_inches='tight')
    plt.close(fig)

    return output_path


def _generate_confusion_matrix_plot(confusion_matrix, output_path):
    """
    Generate a confusion matrix heatmap plot.

    Args:
        confusion_matrix (np.ndarray): 2x2 confusion matrix.
        output_path (str): Path to save the plot.

    Returns:
        str or None: Path to the saved plot, or None if matplotlib is unavailable.
    """
    if plt is None:
        return None

    # Ensure confusion matrix is a 2D array
    cm = np.asarray(confusion_matrix)
    if cm.ndim != 2 or cm.shape != (2, 2):
        logging.warning(f"Invalid confusion matrix shape: {cm.shape}. Expected (2, 2).")
        return None

    fig, ax = plt.subplots(figsize=(6, 5))
    im = ax.imshow(cm, interpolation='nearest', cmap=plt.cm.Blues)
    ax.figure.colorbar(im, ax=ax)

    # Set labels
    classes = ['Negative', 'Positive']
    ax.set(xticks=np.arange(cm.shape[1]),
           yticks=np.arange(cm.shape[0]),
           xticklabels=classes,
           yticklabels=classes,
           ylabel='True Label',
           xlabel='Predicted Label')

    # Add text annotations
    thresh = cm.max() / 2.0
    for i in range(cm.shape[0]):
        for j in range(cm.shape[1]):
            ax.text(j, i, format(cm[i, j], 'd'),
                    ha="center", va="center",
                    color="white" if cm[i, j] > thresh else "black")

    ax.set_title('Confusion Matrix')
    fig.tight_layout()
    plt.savefig(output_path, dpi=150, bbox_inches='tight')
    plt.close(fig)

    return output_path


def generate_report(metrics, run_name, output_dir, config=None):
    """
    Generate a comprehensive evaluation report.

    Args:
        metrics (dict): Dictionary of evaluation metrics from evaluate_model.
        run_name (str): Name of the run (e.g., 'unetpp_scratch').
        output_dir (str): Directory to save the report and plots.
        config (dict, optional): Configuration dictionary.

    Returns:
        str: Path to the generated report file.
    """
    if config is None:
        config = load_config()

    # Create output directory if it doesn't exist
    os.makedirs(output_dir, exist_ok=True)

    # Generate plots if matplotlib is available
    plots = {}

    # Bar chart of metrics
    bar_chart_path = os.path.join(output_dir, f'{run_name}_metrics_bar_chart.png')
    bar_chart_result = _generate_metrics_bar_chart(metrics, bar_chart_path)
    if bar_chart_result is not None:
        plots['metrics_bar_chart'] = bar_chart_result

    # Confusion matrix plot
    if 'confusion_matrix' in metrics:
        cm_plot_path = os.path.join(output_dir, f'{run_name}_confusion_matrix.png')
        cm_plot_result = _generate_confusion_matrix_plot(metrics['confusion_matrix'], cm_plot_path)
        if cm_plot_result is not None:
            plots['confusion_matrix_plot'] = cm_plot_result

    # Generate text report
    report_lines = []
    report_lines.append("=" * 70)
    report_lines.append(f"EVALUATION REPORT - {run_name}")
    report_lines.append("=" * 70)
    report_lines.append("")

    # Project info
    project_config = config.get('project', {})
    report_lines.append(f"Project: {project_config.get('name', 'N/A')}")
    report_lines.append(f"Paper: {project_config.get('paper', 'N/A')}")
    report_lines.append(f"Framework: {project_config.get('framework', 'N/A')}")
    report_lines.append("")

    # Model info
    model_config = config.get('model', {})
    architectures = model_config.get('architectures', ['UNetPlusPlus'])
    report_lines.append(f"Model Architecture: {', '.join(architectures)}")
    report_lines.append(f"Number of Classes: {model_config.get('num_classes', 1)}")
    report_lines.append("")

    # Training info
    training_config = config.get('training', {})
    report_lines.append("--- Training Configuration ---")
    report_lines.append(f"Epochs: {training_config.get('epochs', 'N/A')}")
    optimizer_config = training_config.get('optimizer', {})
    report_lines.append(f"Optimizer: {optimizer_config.get('name', 'N/A')}")
    report_lines.append(f"Learning Rate: {optimizer_config.get('lr', 'N/A')}")
    report_lines.append(f"Loss Function: {training_config.get('loss_function', 'N/A')}")
    report_lines.append("")

    # Evaluation metrics
    report_lines.append("--- Evaluation Metrics ---")
    report_lines.append(f"Number of Samples: {metrics.get('num_samples', 'N/A')}")
    report_lines.append(f"Threshold: {metrics.get('threshold', 0.5)}")
    report_lines.append("")

    # Classification metrics
    report_lines.append("Classification Metrics:")
    report_lines.append(f"  Accuracy:    {metrics.get('accuracy', 0.0):.4f}")
    report_lines.append(f"  Precision:   {metrics.get('precision', 0.0):.4f}")
    report_lines.append(f"  Recall:      {metrics.get('recall', 0.0):.4f}")
    report_lines.append(f"  F1 Score:    {metrics.get('f1', 0.0):.4f}")
    report_lines.append(f"  Specificity: {metrics.get('specificity', 0.0):.4f}")
    report_lines.append("")

    # Segmentation metrics
    report_lines.append("Segmentation Metrics:")
    report_lines.append(f"  IoU:  {metrics.get('iou', 0.0):.4f}")
    report_lines.append(f"  Dice: {metrics.get('dice', 0.0):.4f}")
    report_lines.append("")

    # Confusion matrix
    if 'confusion_matrix' in metrics:
        cm = metrics['confusion_matrix']
        report_lines.append("Confusion Matrix:")
        report_lines.append(f"  True Negatives:  {cm[0, 0]}")
        report_lines.append(f"  False Positives: {cm[0, 1]}")
        report_lines.append(f"  False Negatives: {cm[1, 0]}")
        report_lines.append(f"  True Positives:  {cm[1, 1]}")
        report_lines.append("")

    # Inference time
    if 'inference_time' in metrics:
        inference_time = metrics['inference_time']
        num_samples = metrics.get('num_samples', 1)
        avg_time_per_sample = inference_time / max(num_samples, 1)
        report_lines.append("--- Inference Performance ---")
        report_lines.append(f"Total Inference Time: {inference_time:.4f} seconds")
        report_lines.append(f"Average Time per Sample: {avg_time_per_sample:.6f} seconds")
        report_lines.append("")

    # Plots
    if plots:
        report_lines.append("--- Generated Plots ---")
        for plot_name, plot_path in plots.items():
            report_lines.append(f"  {plot_name}: {plot_path}")
        report_lines.append("")

    # Notes
    report_lines.append("--- Notes ---")
    report_lines.append("Metrics are computed on the test set.")
    report_lines.append("For strict reproduction of the paper, note that the paper does not")
    report_lines.append("specify a separate validation set. The best_metric is set to")
    report_lines.append("val_accuracy as an engineering deviation.")
    report_lines.append("")

    report_lines.append("=" * 70)

    # Write report to file
    report_path = os.path.join(output_dir, f'{run_name}_report.txt')
    with open(report_path, 'w', encoding='utf-8') as f:
        f.write('\n'.join(report_lines))

    # Also save metrics as JSON for programmatic access
    metrics_json_path = os.path.join(output_dir, f'{run_name}_metrics.json')
    metrics_serializable = {}
    for key, value in metrics.items():
        if isinstance(value, np.ndarray):
            metrics_serializable[key] = value.tolist()
        elif isinstance(value, (np.integer, np.floating)):
            metrics_serializable[key] = float(value)
        else:
            metrics_serializable[key] = value

    with open(metrics_json_path, 'w', encoding='utf-8') as f:
        json.dump(metrics_serializable, f, indent=2)

    logging.info(f"Report generated: {report_path}")
    logging.info(f"Metrics saved: {metrics_json_path}")

    return report_path
