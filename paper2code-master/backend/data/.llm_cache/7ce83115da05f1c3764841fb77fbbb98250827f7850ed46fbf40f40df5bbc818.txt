class Edge:
    """Represents an edge in the graph with mark and orientation tracking."""

    __slots__ = ['src', 'tgt', 'mark', 'next_out', 'next_in', 'prev_out', 'prev_in']

    def __init__(self, src: int, tgt: int, mark: int = 0):
        self.src = src
        self.tgt = tgt
        self.mark = mark
        # Linked list pointers for incident edge lists
        self.next_out = None
        self.next_in = None
        self.prev_out = None
        self.prev_in = None

    def __repr__(self):
        return f"Edge({self.src}->{self.tgt}, mark={self.mark})"


class Node:
    """Represents a graph node with mark and incident edge lists."""

    __slots__ = ['id', 'mark', 'edges_out', 'edges_in', 'dist', 'visited', 'in_queue']

    def __init__(self, node_id: int, mark: int = 0):
        self.id = node_id
        self.mark = mark
        # Two-dimensional array: edges_out[mark][orientation], edges_in[mark][orientation]
        # orientation: 0 = outgoing, 1 = incoming
        self.edges_out: Dict[int, List[Edge]] = {}
        self.edges_in: Dict[int, List[Edge]] = {}
        # For Bellman-Ford
        self.dist = float('inf')
        self.visited = False
        self.in_queue = False

    def add_edge_out(self, edge: Edge):
        """Add an outgoing edge to the node's incident list."""
        if edge.mark not in self.edges_out:
            self.edges_out[edge.mark] = []
        self.edges_out[edge.mark].append(edge)

    def add_edge_in(self, edge: Edge):
        """Add an incoming edge to the node's incident list."""
        if edge.mark not in self.edges_in:
            self.edges_in[edge.mark] = []
        self.edges_in[edge.mark].append(edge)

    def get_edges_out(self, mark: Optional[int] = None) -> List[Edge]:
        """Get outgoing edges, optionally filtered by mark."""
        if mark is None:
            result = []
            for edges in self.edges_out.values():
                result.extend(edges)
            return result
        return self.edges_out.get(mark, [])

    def get_edges_in(self, mark: Optional[int] = None) -> List[Edge]:
        """Get incoming edges, optionally filtered by mark."""
        if mark is None:
            result = []
            for edges in self.edges_in.values():
                result.extend(edges)
            return result
        return self.edges_in.get(mark, [])

    def __repr__(self):
        return f"Node({self.id}, mark={self.mark})"


class Graph:
    """
    Graph data structure with mark-based linked lists for nodes and edges.
    
    Implements the enhanced GP 2 graph data structure where:
    - Nodes are stored in separate linked lists based on their marks
    - Each node has a two-dimensional array of linked lists storing incident
      edges based on mark and orientation
    """

    def __init__(self):
        self.nodes: Dict[int, Node] = {}
        self.edges: List[Edge] = []
        # Nodes grouped by mark
        self.nodes_by_mark: Dict[int, List[Node]] = {}
        # Edges grouped by mark
        self.edges_by_mark: Dict[int, List[Edge]] = {}
        self.next_node_id = 0

    def add_node(self, mark: int = 0) -> Node:
        """Add a new node to the graph."""
        node = Node(self.next_node_id, mark)
        self.nodes[self.next_node_id] = node
        if mark not in self.nodes_by_mark:
            self.nodes_by_mark[mark] = []
        self.nodes_by_mark[mark].append(node)
        self.next_node_id += 1
        return node

    def add_edge(self, src: int, tgt: int, mark: int = 0) -> Edge:
        """Add a new edge to the graph."""
        if src not in self.nodes or tgt not in self.nodes:
            raise ValueError(f"Node {src} or {tgt} does not exist")
        edge = Edge(src, tgt, mark)
        self.edges.append(edge)
        self.nodes[src].add_edge_out(edge)
        self.nodes[tgt].add_edge_in(edge)
        if mark not in self.edges_by_mark:
            self.edges_by_mark[mark] = []
        self.edges_by_mark[mark].append(edge)
        return edge

    def get_nodes_by_mark(self, mark: int) -> List[Node]:
        """Get all nodes with a specific mark."""
        return self.nodes_by_mark.get(mark, [])

    def get_edges_by_mark(self, mark: int) -> List[Edge]:
        """Get all edges with a specific mark."""
        return self.edges_by_mark.get(mark, [])

    def get_node(self, node_id: int) -> Optional[Node]:
        """Get a node by its ID."""
        return self.nodes.get(node_id)

    def get_neighbors(self, node_id: int) -> List[int]:
        """Get all neighbors of a node."""
        if node_id not in self.nodes:
            return []
        node = self.nodes[node_id]
        neighbors = set()
        for edge in node.get_edges_out():
            neighbors.add(edge.tgt)
        for edge in node.get_edges_in():
            neighbors.add(edge.src)
        return list(neighbors)

    def get_adjacency_list(self) -> Dict[int, List[int]]:
        """Get the adjacency list representation of the graph."""
        adj = {}
        for node_id in self.nodes:
            adj[node_id] = self.get_neighbors(node_id)
        return adj

    def get_edge_count(self) -> int:
        """Get the total number of edges."""
        return len(self.edges)

    def get_node_count(self) -> int:
        """Get the total number of nodes."""
        return len(self.nodes)

    def clear_marks(self):
        """Reset all node and edge marks to 0."""
        for node in self.nodes.values():
            node.mark = 0
        for edge in self.edges:
            edge.mark = 0

    def __repr__(self):
        return f"Graph(nodes={len(self.nodes)}, edges={len(self.edges)})"

class GraphProgramRule:
    """Base class for graph program rules."""

    def apply(self, graph: Graph, host: Node) -> bool:
        """Apply the rule to the graph at the host node. Returns True if applied."""
        raise NotImplementedError


class ConnectednessRule(GraphProgramRule):
    """
    Rule for connectedness checking.
    
    Marks all nodes reachable from the start node.
    """

    def apply(self, graph: Graph, host: Node) -> bool:
        """Mark the host node as visited and propagate."""
        if host.mark == 1:
            return False
        host.mark = 1
        return True


class AcyclicityRule(GraphProgramRule):
    """
    Rule for acyclicity checking.
    
    Detects cycles by checking if a node is visited twice.
    """

    def apply(self, graph: Graph, host: Node) -> bool:
        """Check if the host node creates a cycle."""
        if host.mark == 1:
            return True  # Cycle detected
        host.mark = 1
        return False


class BellmanFordRule(GraphProgramRule):
    """
    Rule for Bellman-Ford shortest path.
    
    Relaxes edges from the host node.
    """

    def __init__(self, weight: int = 1):
        self.weight = weight

    def apply(self, graph: Graph, host: Node) -> bool:
        """Relax edges from the host node."""
        changed = False
        for edge in host.get_edges_out():
            target = graph.get_node(edge.tgt)
            if target and host.dist + self.weight < target.dist:
                target.dist = host.dist + self.weight
                changed = True
        return changed

class GraphProgram:
    """Base class for graph programs."""

    def __init__(self, graph: Graph):
        self.graph = graph

    def run(self) -> Any:
        """Execute the graph program."""
        raise NotImplementedError


class ConnectednessProgram(GraphProgram):
    """
    Linear-time connectedness check.
    
    Uses a BFS-like approach with mark-based node storage.
    Time complexity: O(n + m) where n = nodes, m = edges.
    """

    def run(self) -> bool:
        """Check if the graph is connected."""
        if self.graph.get_node_count() == 0:
            return True

        # Reset marks
        self.graph.clear_marks()

        # Start BFS from the first node
        start_node = next(iter(self.graph.nodes.values()))
        start_node.mark = 1

        queue = deque([start_node])
        visited_count = 1

        while queue:
            current = queue.popleft()
            # Get all neighbors using the mark-based structure
            for edge in current.get_edges_out():
                neighbor = self.graph.get_node(edge.tgt)
                if neighbor and neighbor.mark == 0:
                    neighbor.mark = 1
                    visited_count += 1
                    queue.append(neighbor)
            for edge in current.get_edges_in():
                neighbor = self.graph.get_node(edge.src)
                if neighbor and neighbor.mark == 0:
                    neighbor.mark = 1
                    visited_count += 1
                    queue.append(neighbor)

        return visited_count == self.graph.get_node_count()


class AcyclicityProgram(GraphProgram):
    """
    Linear-time acyclicity check.
    
    Uses DFS-based cycle detection with mark-based node storage.
    Time complexity: O(n + m) where n = nodes, m = edges.
    """

    def run(self) -> bool:
        """Check if the graph is acyclic."""
        if self.graph.get_node_count() == 0:
            return True

        # Reset marks: 0 = unvisited, 1 = in progress, 2 = done
        self.graph.clear_marks()

        def dfs(node: Node) -> bool:
            """DFS traversal. Returns True if cycle detected."""
            node.mark = 1  # In progress

            for edge in node.get_edges_out():
                neighbor = self.graph.get_node(edge.tgt)
                if neighbor:
                    if neighbor.mark == 1:
                        return True  # Back edge -> cycle
                    if neighbor.mark == 0:
                        if dfs(neighbor):
                            return True

            node.mark = 2  # Done
            return False

        for node in self.graph.nodes.values():
            if node.mark == 0:
                if dfs(node):
                    return False  # Has cycle

        return True  # Acyclic


class BellmanFordProgram(GraphProgram):
    """
    Bellman-Ford shortest path implementation.
    
    Time complexity: O(n * m) where n = nodes, m = edges.
    Matches imperative implementations.
    """

    def __init__(self, graph: Graph, source: int = 0, weight: int = 1):
        super().__init__(graph)
        self.source = source
        self.weight = weight

    def run(self) -> Dict[int, float]:
        """Compute shortest paths from source to all nodes."""
        n = self.graph.get_node_count()
        if n == 0:
            return {}

        # Initialize distances
        for node in self.graph.nodes.values():
            node.dist = float('inf')
        source_node = self.graph.get_node(self.source)
        if source_node is None:
            raise ValueError(f"Source node {self.source} does not exist")
        source_node.dist = 0

        # Relax edges n-1 times
        for _ in range(n - 1):
            updated = False
            for edge in self.graph.edges:
                src_node = self.graph.get_node(edge.src)
                tgt_node = self.graph.get_node(edge.tgt)
                if src_node and tgt_node:
                    if src_node.dist != float('inf') and \
                       src_node.dist + self.weight < tgt_node.dist:
                        tgt_node.dist = src_node.dist + self.weight
                        updated = True
            if not updated:
                break

        # Check for negative cycles (not applicable with positive weights)
        for edge in self.graph.edges:
            src_node = self.graph.get_node(edge.src)
            tgt_node = self.graph.get_node(edge.tgt)
            if src_node and tgt_node:
                if src_node.dist != float('inf') and \
                   src_node.dist + self.weight < tgt_node.dist:
                    raise ValueError("Graph contains a negative cycle")

        return {node_id: node.dist for node_id, node in self.graph.nodes.items()}

def generate_random_graph(num_nodes: int, edge_probability: float = 0.3,
                          seed: Optional[int] = None) -> Graph:
    """Generate a random graph with the given number of nodes."""
    if seed is not None:
        random.seed(seed)

    graph = Graph()
    for i in range(num_nodes):
        graph.add_node(mark=0)

    for i in range(num_nodes):
        for j in range(num_nodes):
            if i != j and random.random() < edge_probability:
                graph.add_edge(i, j, mark=0)

    return graph


def generate_disconnected_graph(num_nodes: int, num_components: int = 2,
                                seed: Optional[int] = None) -> Graph:
    """Generate a disconnected graph with multiple components."""
    if seed is not None:
        random.seed(seed)

    graph = Graph()
    for i in range(num_nodes):
        graph.add_node(mark=0)

    # Connect nodes within components
    nodes_per_component = num_nodes // num_components
    for comp in range(num_components):
        start = comp * nodes_per_component
        end = start + nodes_per_component
        for i in range(start, end):
            for j in range(i + 1, end):
                if random.random() < 0.5:
                    graph.add_edge(i, j, mark=0)

    return graph


def generate_acyclic_graph(num_nodes: int, seed: Optional[int] = None) -> Graph:
    """Generate a directed acyclic graph."""
    if seed is not None:
        random.seed(seed)

    graph = Graph()
    for i in range(num_nodes):
        graph.add_node(mark=0)

    for i in range(num_nodes):
        for j in range(i + 1, num_nodes):
            if random.random() < 0.3:
                graph.add_edge(i, j, mark=0)

    return graph


def generate_cyclic_graph(num_nodes: int, seed: Optional[int] = None) -> Graph:
    """Generate a graph with cycles."""
    if seed is not None:
        random.seed(seed)

    graph = Graph()
    for i in range(num_nodes):
        graph.add_node(mark=0)

    # Create a cycle
    for i in range(num_nodes):
        graph.add_edge(i, (i + 1) % num_nodes, mark=0)

    # Add random extra edges
    for i in range(num_nodes):
        for j in range(num_nodes):
            if i != j and random.random() < 0.1:
                graph.add_edge(i, j, mark=0)

    return graph

class EnhancedGP2Compiler:
    """
    Enhanced GP 2 compiler with new graph data structure.
    
    Implements the paper's core contributions:
    1. Nodes stored in separate linked lists based on marks
    2. Two-dimensional array of linked lists for incident edges
    3. Linear-time connectedness and acyclicity checks
    4. Bellman-Ford with O(nm) time complexity
    """

    def __init__(self, config: Optional[Dict[str, Any]] = None):
        self.config = config or {}
        self.logger = self._setup_logger()
        self.results = {}

    def _setup_logger(self) -> logging.Logger:
        """Set up logging based on config."""
        logger = logging.getLogger(__name__)
        log_level = self.config.get('logging', {}).get('level', 'INFO')
        log_format = self.config.get('logging', {}).get(
            'format', '%(asctime)s - %(name)s - %(levelname)s - %(message)s')

        handler = logging.StreamHandler()
        handler.setFormatter(logging.Formatter(log_format))
        logger.addHandler(handler)
        logger.setLevel(getattr(logging, log_level, logging.INFO))
        return logger

    def run(self, config: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """
        Run the enhanced GP 2 compiler with the new graph data structure.
        
        Args:
            config: Configuration dictionary (optional, defaults to self.config)
            
        Returns:
            Dictionary containing results of the graph program executions
        """
        if config is not None:
            self.config = config

        self.logger.info("Starting Enhanced GP 2 Compiler with new graph data structure")
        self.logger.info("Paper: RULE-BASED GRAPH PROGRAMS MATCHING THE TIME COMPLEXITY OF IMPERATIVE ALGORITHMS")

        # Get experiment configuration
        experiment_config = self.config.get('experiment', {})
        runs = experiment_config.get('runs', [{'id': 'run_1'}])

        results = {}
        for run_config in runs:
            run_id = run_config.get('id', 'run_1')
            self.logger.info(f"Executing run: {run_id}")
            results[run_id] = self._execute_run(run_id)

        self.results = results
        self.logger.info("Enhanced GP 2 Compiler execution completed")
        return results

    def _execute_run(self, run_id: str) -> Dict[str, Any]:
        """Execute a single run of the compiler."""
        seed = self.config.get('system', {}).get('seed', 42)
        random.seed(seed)

        run_results = {}

        # Case Study 1: Connectedness Check
        self.logger.info("Case Study 1: Connectedness Check")
        run_results['connectedness'] = self._test_connectedness()

        # Case Study 2: Acyclicity Check
        self.logger.info("Case Study 2: Acyclicity Check")
        run_results['acyclicity'] = self._test_acyclicity()

        # Case Study 3: Bellman-Ford Shortest Path
        self.logger.info("Case Study 3: Bellman-Ford Shortest Path")
        run_results['bellman_ford'] = self._test_bellman_ford()

        # Time complexity analysis
        self.logger.info("Time Complexity Analysis")
        run_results['time_complexity'] = self._analyze_time_complexity()

        return run_results

    def _verify_linear_time(self, results: Dict[str, Any], program_type: str) -> bool:
        """
        Verify linear time complexity by checking that time scales roughly
        linearly with graph size.
        """
        # Extract times for different graph sizes
        times = []
        sizes = []
        for key, result in results.items():
            if 'time_seconds' in result and 'nodes' in result:
                times.append(result['time_seconds'])
                sizes.append(result['nodes'])

        if len(times) < 2:
            return True  # Not enough data to verify

        # Check if time per node is roughly constant (linear scaling)
        time_per_node = [t / max(s, 1) for t, s in zip(times, sizes)]
        if len(time_per_node) >= 2:
            ratio = max(time_per_node) / max(min(time_per_node), 1e-10)
            return ratio < 10  # Allow some variance

        return True

    def _verify_onm_time(self, results: Dict[str, Any]) -> bool:
        """
        Verify O(n*m) time complexity for Bellman-Ford.
        """
        times = []
        products = []
        for key, result in results.items():
            if 'time_seconds' in result and 'nodes' in result and 'edges' in result:
                times.append(result['time_seconds'])
                products.append(result['nodes'] * result['edges'])

        if len(times) < 2:
            return True

        # Check if time per (n*m) is roughly constant
        time_per_product = [t / max(p, 1) for t, p in zip(times, products)]
        if len(time_per_product) >= 2:
            ratio = max(time_per_product) / max(min(time_per_product), 1e-10)
            return ratio < 10

        return True

    def _load_config(self, config_path: str) -> Dict[str, Any]:
        """Load configuration from YAML file."""
        if yaml is None:
            self.logger.warning("PyYAML not available, using default config")
            return {}

        try:
            with open(config_path, 'r', encoding='utf-8') as f:
                config = yaml.safe_load(f)
            return config or {}
        except FileNotFoundError:
            self.logger.warning(f"Config file {config_path} not found")
            return {}
        except Exception as e:
            self.logger.error(f"Error loading config: {e}")
            return {}
