```python
import os
import sys
import json
import numpy as np
import cv2


def load_image(image_path):
    """读取图片，支持中文路径。"""
    data = np.fromfile(image_path, dtype=np.uint8)
    img = cv2.imdecode(data, cv2.IMREAD_COLOR)
    return img


def get_ocr_text():
    """获取OCR文字：优先从命令行第二个参数读取，其次从环境变量OCR_TEXT读取。"""
    if len(sys.argv) >= 3:
        return sys.argv[2]
    env_text = os.environ.get('OCR_TEXT', '')
    return env_text


def compute_visual_features(img):
    """计算简单的视觉特征，用于辅助判断。"""
    h, w = img.shape[:2]
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    saturation = hsv[:, :, 1]
    color_ratio = float(np.mean(saturation > 50))

    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    white_ratio = float(np.mean(gray > 240))

    return {
        'height': h,
        'width': w,
        'color_ratio': color_ratio,
        'white_ratio': white_ratio
    }


def extract_components(ocr_lower, candidate_dict):
    """从OCR小写文本中提取出现的组件/指标名称。"""
    found = []
    for key, value in candidate_dict.items():
        if key in ocr_lower:
            found.append(value)
    seen = set()
    result = []
    for item in found:
        if item not in seen:
            seen.add(item)
            result.append(item)
    return result


def main():
    try:
        image_path = sys.argv[1] if len(sys.argv) > 1 else ''
        if not image_path:
            result = {
                'type': 'other',
                'description': '无法确定图片类型',
                'important_components': [],
                'code_relevant_information': ''
            }
            print(json.dumps(result, ensure_ascii=False))
            return

        img = load_image(image_path)
        if img is None:
            result = {
                'type': 'other',
                'description': '无法确定图片类型',
                'important_components': [],
                'code_relevant_information': ''
            }
            print(json.dumps(result, ensure_ascii=False))
            return

        ocr_text = get_ocr_text()
        ocr_lower = ocr_text.lower()

        # 关键词列表
        experiment_keywords = [
            'epoch', 'loss', 'accuracy', 'precision', 'recall',
            'f1', 'auc', 'curve', 'performance', 'comparison',
            'bar chart', 'histogram', 'training'
        ]
        architecture_keywords = [
            'cnn', 'transformer', 'attention', 'encoder', 'decoder',
            'block', 'embedding', 'classifier', 'feature extractor',
            'alexnet', 'resnet', 'vgg', 'googlenet',
            'deep learning architecture', 'convolutional'
        ]
        dataset_keywords = [
            'sample', 'data sample', 'input example', 'category',
            'class', 'example', 'dataset', 'annotation', 'image sample'
        ]

        # 按优先级判断类型
        is_experiment = any(k in ocr_lower for k in experiment_keywords)
        is_architecture = any(k in ocr_lower for k in architecture_keywords)
        is_dataset = any(k in ocr_lower for k in dataset_keywords)

        visual = compute_visual_features(img)

        img_type = 'other'
        if is_experiment:
            img_type = 'experiment_result'
        elif is_architecture:
            img_type = 'architecture'
        elif is_dataset:
            # 有数据集关键词且视觉上颜色丰富，认为是数据集样本
            img_type = 'dataset'
        elif not ocr_lower.strip():
            # 无OCR文字时无法可靠判断，保持other
            img_type = 'other'
        else:
            # 有OCR但未命中任何关键词，简单视觉辅助
            if visual['color_ratio'] > 0.15 and visual['white_ratio'] < 0.75:
                img_type = 'dataset'
            else:
                img_type = 'other'

        components = []
        if img_type == 'experiment_result':
            model_dict = {
                'alexnet': 'AlexNet',
                'resnet': 'ResNet',
                'vgg': 'VGG',
                'googlenet': 'GoogLeNet',
                'inception': 'Inception',
                'densenet': 'DenseNet',
                'mobilenet': 'MobileNet',
                'efficientnet': 'EfficientNet',
                'transformer': 'Transformer',
                'bert': 'BERT',
                'cnn': 'CNN',
                'rnn': 'RNN',
                'lstm': 'LSTM',
                'gru': 'GRU'
            }
            metric_dict = {
                'accuracy': 'Accuracy',
                'loss': 'Loss',
                'precision': 'Precision',
                'recall': 'Recall',
                'f1': 'F1-score',
                'auc': 'AUC',
                'epoch': 'Epoch',
                'epochs': 'Epochs'
            }
            components = (extract_components(ocr_lower, model_dict) +
                          extract_components(ocr_lower, metric_dict))
        elif img_type == 'architecture':
            comp_dict = {
                'alexnet': 'AlexNet',
                'resnet': 'ResNet',
                'vgg': 'VGG',
                'googlenet': 'GoogLeNet',
                'transformer': 'Transformer',
                'attention': 'Attention Module',
                'encoder': 'Encoder',
                'decoder': 'Decoder',
                'block': 'Block',
                'embedding': 'Embedding Layer',
                'classifier': 'Classifier',
                'feature extractor': 'Feature Extractor',
                'cnn': 'CNN'
            }
            components = extract_components(ocr_lower, comp_dict)
        elif img_type == 'dataset':
            components = []
        else:
            components = []

        if img_type == 'experiment_result':
            description = '展示模型性能对比或训练过程的实验曲线/柱状图'
            code_relevant = '需要实现相关模型并进行训练，记录并绘制性能指标变化曲线'
        elif img_type == 'architecture':
            description = '展示模型网络结构和数据流关系'
            code_relevant = '需要实现相关网络模块及其连接关系'
        elif img_type == 'dataset':
            description = '展示数据样本或类别示例'
            code_relevant = '需要准备或加载相应数据集'
        else:
            description = '无法确定图片类型'
            code_relevant = ''

        result = {
            'type': img_type,
            'description': description,
            'important_components': components,
            'code_relevant_information': code_relevant
        }
        print(json.dumps(result, ensure_ascii=False))

    except Exception:
        result = {
            'type': 'other',
            'description': '无法确定图片类型',
            'important_components': [],
            'code_relevant_information': ''
        }
        print(json.dumps(result, ensure_ascii=False))


if __name__ == '__main__':
    main()
```
