import sys
import json
import numpy as np
import cv2

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty in this case
        ocr_text = ""
        
        # Since OCR is empty, use visual features and keywords
        # Get basic image properties
        height, width = img.shape[:2]
        
        # Check if image is mostly grayscale (common for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        color_std = np.std(img, axis=2).mean()
        
        # Check color distribution
        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        saturation = hsv[:,:,1].mean()
        
        # Architecture diagrams often have:
        # - Multiple distinct regions/blocks
        # - Lines connecting blocks
        # - Text labels
        # - Often white/light background
        
        # Experiment results often have:
        # - Colored lines/curves
        # - Grid backgrounds
        # - Legends
        # - Axis labels
        
        # Dataset images often have:
        # - Multiple similar images in grid
        # - Natural colors
        # - High color variance
        
        # Simple heuristic based on color statistics
        if saturation < 30 and color_std < 50:
            # Likely architecture diagram (mostly black/white/gray)
            result_type = "architecture"
            description = "展示模型网络结构和数据流关系"
            components = []
            code_info = "需要实现模型网络结构"
        elif saturation > 60 and color_std > 60:
            # Likely dataset images (natural colors)
            result_type = "dataset"
            description = "展示数据集样本或类别图片"
            components = []
            code_info = "需要实现数据加载和预处理模块"
        else:
            # Could be experiment results or other
            # Check for typical experiment plot characteristics
            # Look for axis-like structures (lines near edges)
            edges = cv2.Canny(gray, 50, 150)
            
            # Check if image has grid-like structure (common in plots)
            # Simple check: look for many horizontal/vertical lines
            horizontal_lines = 0
            vertical_lines = 0
            
            # Sample some rows and columns
            for i in range(0, height, max(1, height//20)):
                row = edges[i, :]
                if np.sum(row) > width * 0.3:
                    horizontal_lines += 1
            
            for j in range(0, width, max(1, width//20)):
                col = edges[:, j]
                if np.sum(col) > height * 0.3:
                    vertical_lines += 1
            
            if horizontal_lines > 3 and vertical_lines > 3:
                # Likely a plot/graph
                result_type = "experiment_result"
                description = "展示实验结果曲线或性能对比"
                components = []
                code_info = "需要实现实验评估和结果可视化模块"
            else:
                result_type = "other"
                description = "无法确定图片类型"
                components = []
                code_info = ""
        
        return {
            "type": result_type,
            "description": description,
            "important_components": components,
            "code_relevant_information": code_info
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
