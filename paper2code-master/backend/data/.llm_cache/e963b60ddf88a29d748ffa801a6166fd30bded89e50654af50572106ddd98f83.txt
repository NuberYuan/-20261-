import sys
import json
import numpy as np
import cv2
import os

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片文件",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty in this case
        ocr_text = ""
        
        # Since OCR text is empty, use visual features
        # Get basic image properties
        height, width = img.shape[:2]
        
        # Check if image is mostly grayscale (common for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        color_std = np.std(img, axis=2).mean()
        
        # Check color distribution
        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        saturation_mean = np.mean(hsv[:,:,1])
        
        # Check if image has many distinct colors (common for datasets)
        small_img = cv2.resize(img, (64, 64))
        unique_colors = len(np.unique(small_img.reshape(-1, 3), axis=0))
        
        # Check aspect ratio
        aspect_ratio = width / height if height > 0 else 0
        
        # Since no OCR text available, use visual heuristics
        # Check for typical chart/plot characteristics
        # Look for grid-like patterns or axis lines (without using Hough)
        
        # Convert to grayscale and threshold
        _, binary = cv2.threshold(gray, 200, 255, cv2.THRESH_BINARY)
        
        # Check white pixel ratio (charts often have white background)
        white_ratio = np.sum(binary > 0) / (height * width)
        
        # Check for colored elements (typical in plots)
        colored_pixels = np.sum(np.max(img, axis=2) - np.min(img, axis=2) > 30)
        colored_ratio = colored_pixels / (height * width)
        
        # Decision logic based on visual features
        # Since no OCR text, rely on visual characteristics
        
        # Check if it looks like a plot/chart (has axes, grid, etc.)
        # Look for horizontal and vertical lines (without Hough)
        edges_x = np.sum(np.abs(np.diff(gray, axis=1)), axis=0)
        edges_y = np.sum(np.abs(np.diff(gray, axis=0)), axis=1)
        
        # Check for strong horizontal/vertical line patterns
        strong_h_lines = np.sum(edges_y > (np.mean(edges_y) + 2*np.std(edges_y)))
        strong_v_lines = np.sum(edges_x > (np.mean(edges_x) + 2*np.std(edges_x)))
        
        # Check for text-like regions (high contrast areas)
        text_regions = np.sum(binary < 100) / (height * width)
        
        # Default to other
        result_type = "other"
        description = "无法确定图片类型"
        components = []
        code_info = ""
        
        # Heuristic: If image has strong line patterns and moderate color, likely experiment result
        if strong_h_lines > 5 and strong_v_lines > 5 and colored_ratio > 0.05:
            result_type = "experiment_result"
            description = "展示实验结果图表，包含数据曲线或性能对比"
            components = ["实验指标"]
            code_info = "需要实现实验数据可视化模块，包括曲线绘制和性能指标展示"
        
        # Heuristic: If image has many unique colors and is square-ish, likely dataset
        elif unique_colors > 100 and 0.8 < aspect_ratio < 1.2:
            result_type = "dataset"
            description = "展示数据集样本或类别示例"
            components = ["数据样本"]
            code_info = "需要实现数据加载和预处理模块，支持数据增强和批次处理"
        
        # Heuristic: If image is mostly grayscale with block-like structures, likely architecture
        elif color_std < 30 and white_ratio > 0.3:
            result_type = "architecture"
            description = "展示模型网络结构和数据流关系"
            components = []
            code_info = "需要实现模型网络结构定义和特征提取模块"
        
        # Additional check: Look for typical plot characteristics
        # Check for axis-like structures (lines near edges)
        border_thickness = 5
        border_region = gray[border_thickness:height-border_thickness, border_thickness:width-border_thickness]
        border_std = np.std(border_region)
        
        if result_type == "other" and border_std > 50:
            result_type = "experiment_result"
            description = "展示实验结果数据图表"
            components = ["实验数据"]
            code_info = "需要实现数据可视化功能，包括图表绘制和结果展示"
        
        return {
            "type": result_type,
            "description": description,
            "important_components": components,
            "code_relevant_information": code_info
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "未提供图片路径",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
