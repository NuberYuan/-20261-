import sys
import json
import numpy as np
import cv2
from PIL import Image
import os

def analyze_image(image_path, ocr_text):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # Get basic image info
        height, width = img.shape[:2]
        
        # Convert OCR text to lowercase for matching
        ocr_lower = ocr_text.lower()
        
        # Define keywords for each category
        architecture_keywords = [
            "cnn", "transformer", "attention", "encoder", "decoder", 
            "block", "embedding", "classifier", "feature extractor",
            "alexnet", "resnet", "vgg", "googlenet"
        ]
        
        experiment_keywords = [
            "accuracy", "loss", "epoch", "precision", "recall", "f1",
            "curve", "performance", "comparison", "training", "validation"
        ]
        
        dataset_keywords = [
            "dataset", "sample", "example", "input", "image", "class",
            "category", "label"
        ]
        
        # Check for architecture
        architecture_components = []
        for kw in architecture_keywords:
            if kw in ocr_lower:
                architecture_components.append(kw)
        
        # Check for experiment results
        experiment_components = []
        for kw in experiment_keywords:
            if kw in ocr_lower:
                experiment_components.append(kw)
        
        # Check for dataset
        dataset_components = []
        for kw in dataset_keywords:
            if kw in ocr_lower:
                dataset_components.append(kw)
        
        # Determine type based on priority
        if experiment_components:
            # Check if it's really experiment result (has metrics or curves)
            if any(kw in ocr_lower for kw in ["accuracy", "loss", "epoch", "precision", "recall", "f1", "curve"]):
                return {
                    "type": "experiment_result",
                    "description": "展示实验结果和性能指标对比",
                    "important_components": experiment_components[:5],
                    "code_relevant_information": "需要实现模型评估和指标计算功能"
                }
        
        if architecture_components:
            # Check if it's really architecture (has model structure)
            if any(kw in ocr_lower for kw in ["cnn", "transformer", "attention", "encoder", "decoder", "block", "embedding", "classifier"]):
                return {
                    "type": "architecture",
                    "description": "展示模型网络结构和数据流关系",
                    "important_components": architecture_components[:5],
                    "code_relevant_information": "需要实现模型特征提取模块和分类头"
                }
        
        if dataset_components:
            return {
                "type": "dataset",
                "description": "展示数据集样本和类别信息",
                "important_components": dataset_components[:5],
                "code_relevant_information": "需要实现数据加载和预处理功能"
            }
        
        # If no clear match, return other
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "图片分析失败",
            "important_components": [],
            "code_relevant_information": ""
        }

def main():
    try:
        # Get image path from command line
        image_path = sys.argv[1]
        
        # OCR text provided externally
        ocr_text = "Feaean"
        
        # Analyze image
        result = analyze_image(image_path, ocr_text)
        
        # Output JSON
        print(json.dumps(result, ensure_ascii=False))
        
    except Exception as e:
        # Fallback output
        fallback = {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
        print(json.dumps(fallback, ensure_ascii=False))

if __name__ == "__main__":
    main()
