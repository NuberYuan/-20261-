```python
import os
import json
import random
import logging
from typing import Dict, List, Optional, Tuple, Any, Union

import yaml
import torch
from torch.utils.data import Dataset, random_split

logger = logging.getLogger(__name__)


class SCICOQADataset(Dataset):
    """
    SCICOQA dataset for scientific paper-code alignment quality assurance.

    The dataset contains 635 paper-code discrepancies (92 real, 543 synthetic).
    Real data is extracted from GitHub issues and reproducibility papers, filtered
    and validated using LLMs, and reformulated into standardized descriptions.
    Synthetic data is sampled from GitHub repositories, with code discrepancies
    generated by GPT-5 and verified to exactly match the original code.
    """

    def __init__(
        self,
        version: str = "default",
        config: Optional[Dict[str, Any]] = None,
        data_path: Optional[str] = None,
        transform: Optional[Any] = None,
        seed: int = 42,
    ):
        """
        Initialize the SCICOQA dataset.

        Args:
            version: Dataset version (default: 'default')
            config: Configuration dictionary loaded from config.yaml
            data_path: Path to the dataset file (overrides config)
            transform: Optional transform to apply (kept for API compatibility)
            seed: Random seed for reproducibility
        """
        self.version = version
        self.config = config if config is not None else {}
        self.seed = seed
        self.transform = transform

        # Resolve data path from argument or config
        self.data_path = data_path
        if self.data_path is None:
            self.data_path = self._resolve_data_path()

        # Dataset storage
        self.samples: List[Dict[str, Any]] = []
        self.labels: List[int] = []
        self.features: List[torch.Tensor] = []

        # Load and preprocess data
        self._load_data()
        self.preprocess_data()

    def _resolve_data_path(self) -> Optional[str]:
        """Resolve the data path from config, checking each level safely."""
        data_cfg = self.config.get("data", {})
        if not isinstance(data_cfg, dict):
            return None

        # Check version-specific path first
        version_cfg = data_cfg.get(self.version)
        if isinstance(version_cfg, dict):
            path = version_cfg.get("path")
            if path:
                return path

        # Fall back to root path
        root = data_cfg.get("root")
        if root:
            return root

        return None

    def _load_data(self) -> None:
        """
        Load raw data from the data source.

        The dataset contains 635 paper-code discrepancies:
        - 92 real discrepancies from GitHub issues and reproducibility papers
        - 543 synthetic discrepancies from GitHub repositories

        If no data file is available, generate a synthetic dataset matching
        the expected distribution for testing purposes.
        """
        if self.data_path and os.path.exists(self.data_path):
            self._load_from_file(self.data_path)
        else:
            logger.warning(
                "Data path not found or not provided. "
                "Generating synthetic dataset with expected distribution "
                "(92 real + 543 synthetic = 635 samples)."
            )
            self._generate_synthetic_data()

    def _load_from_file(self, path: str) -> None:
        """Load data from a JSON or JSONL file."""
        ext = os.path.splitext(path)[1].lower()
        try:
            if ext == ".json":
                with open(path, "r", encoding="utf-8") as f:
                    data = json.load(f)
                if isinstance(data, list):
                    self.samples = data
                elif isinstance(data, dict):
                    self.samples = data.get("samples", data.get("data", []))
                else:
                    raise ValueError(f"Unexpected JSON format in {path}")
            elif ext in (".jsonl", ".ndjson"):
                self.samples = []
                with open(path, "r", encoding="utf-8") as f:
                    for line in f:
                        line = line.strip()
                        if line:
                            self.samples.append(json.loads(line))
            else:
                raise ValueError(f"Unsupported file format: {ext}")
        except Exception as e:
            logger.error(f"Failed to load data from {path}: {e}")
            raise

    def _generate_synthetic_data(self) -> None:
        """
        Generate synthetic data matching the expected dataset distribution.

        Creates 635 samples: 92 real + 543 synthetic discrepancies.
        Each sample contains paper text, code snippet, and discrepancy description.
        """
        rng = random.Random(self.seed)
        self.samples = []

        # Generate 92 real samples
        for i in range(92):
            sample = {
                "id": f"real_{i:04d}",
                "source": "github_issue" if i % 2 == 0 else "reproducibility_paper",
                "paper_text": f"Paper description for real discrepancy {i}",
                "code_snippet": f"def function_{i}(x):\n    return x * {i}",
                "discrepancy": f"Real discrepancy {i}: code does not match paper description",
                "is_synthetic": False,
            }
            self.samples.append(sample)

        # Generate 543 synthetic samples
        for i in range(543):
            sample = {
                "id": f"synthetic_{i:04d}",
                "source": "github_repository",
                "paper_text": f"Paper description for synthetic discrepancy {i}",
                "code_snippet": f"def function_{i}(x):\n    return x + {i}",
                "discrepancy": f"Synthetic discrepancy {i}: generated by GPT-5",
                "is_synthetic": True,
            }
            self.samples.append(sample)

        # Shuffle to mix real and synthetic samples
        rng.shuffle(self.samples)

    def preprocess_data(self) -> None:
        """
        Preprocess the dataset according to the paper's methodology.

        For real data: extracted from GitHub issues and reproducibility papers,
        filtered and validated using LLMs, reformulated into standardized descriptions.

        For synthetic data: sampled from GitHub repositories, code discrepancies
        generated by GPT-5, verified to exactly match the original code.

        This method converts raw samples into feature tensors and labels.
        """
        if not self.samples:
            logger.warning("No samples to preprocess")
            return

        self.features = []
        self.labels = []

        for sample in self.samples:
            # Extract text components
            paper_text = sample.get("paper_text", "")
            code_snippet = sample.get("code_snippet", "")
            discrepancy = sample.get("discrepancy", "")
            is_synthetic = sample.get("is_synthetic", False)

            # Combine text components for feature representation
            combined_text = f"{paper_text} [SEP] {code_snippet} [SEP] {discrepancy}"

            # Convert to feature tensor (simple hash-based bag-of-words for demo)
            # In practice, this would use a proper tokenizer and embedding
            feature = self._text_to_feature(combined_text)

            # Label: 1 for real discrepancy, 0 for synthetic
            label = 0 if is_synthetic else 1

            self.features.append(feature)
            self.labels.append(label)

        # Convert to tensors
        if self.features:
            self.features = torch.stack(self.features)
            self.labels = torch.tensor(self.labels, dtype=torch.long)

        logger.info(
            f"Preprocessed {len(self.samples)} samples: "
            f"{sum(1 for s in self.samples if not s.get('is_synthetic', False))} real, "
            f"{sum(1 for s in self.samples if s.get('is_synthetic', False))} synthetic"
        )

    def _text_to_feature(self, text: str, feature_dim: int = 128) -> torch.Tensor:
        """
        Convert text to a fixed-size feature tensor.

        Uses a simple hash-based bag-of-words representation.
        In production, this would use a proper tokenizer and embedding.

        Args:
            text: Input text
            feature_dim: Dimension of the output feature vector

        Returns:
            Feature tensor of shape (feature_dim,)
        """
        feature = torch.zeros(feature_dim, dtype=torch.float32)
        tokens = text.lower().split()

        for token in tokens:
            # Simple hash-based indexing
            idx = hash(token) % feature_dim
            feature[idx] += 1.0

        # L2 normalization
        norm = torch.norm(feature)
        if norm > 0:
            feature = feature / norm

        return feature

    def split_dataset(
        self,
        ratios: Optional[List[float]] = None,
        seed: Optional[int] = None,
    ) -> Tuple["SCICOQADataset", "SCICOQADataset", Optional["SCICOQADataset"]]:
        """
        Split the dataset into train/val/test subsets.

        Args:
            ratios: List of ratios for train/val/test splits.
                    If None, uses 0.8/0.1/0.1 as default.
            seed: Random seed for reproducibility

        Returns:
            Tuple of (train_dataset, val_dataset, test_dataset)
            test_dataset is None if only two ratios are provided
        """
        if ratios is None:
            ratios = [0.8, 0.1, 0.1]

        if len(ratios) not in (2, 3):
            raise ValueError("ratios must have 2 or 3 elements")

        if abs(sum(ratios) - 1.0) > 1e-6:
            raise ValueError(f"Ratios must sum to 1.0, got {sum(ratios)}")

        seed = seed if seed is not None else self.seed
        generator = torch.Generator().manual_seed(seed)

        total_size = len(self)
        if total_size == 0:
            raise ValueError("Cannot split an empty dataset")

        # Calculate split sizes
        train_size = int(total_size * ratios[0])
        val_size = int(total_size * ratios[1])

        if len(ratios) == 3:
            test_size = total_size - train_size - val_size
            sizes = [train_size, val_size, test_size]
        else:
            test_size = total_size - train_size - val_size
            sizes = [train_size, val_size + test_size]

        # Perform the split
        splits = random_split(self, sizes, generator=generator)

        if len(ratios) == 3:
            train_ds, val_ds, test_ds = splits
        else:
            train_ds, val_ds = splits
            test_ds = None

        logger.info(
            f"Split dataset: train={len(train_ds)}, val={len(val_ds)}, "
            f"test={len(test_ds) if test_ds else 0}"
        )

        return train_ds, val_ds, test_ds

    def __len__(self) -> int:
        """Return the number of samples in the dataset."""
        return len(self.samples)

    def __getitem__(self, idx: int) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Get a sample from the dataset.

        Args:
            idx: Index of the sample

        Returns:
            Tuple of (feature, label)
        """
        if idx < 0 or idx >= len(self):
            raise IndexError(f"Index {idx} out of range [0, {len(self)})")

        feature = self.features[idx]
        label = self.labels[idx]

        if self.transform is not None:
            feature = self.transform(feature)

        return feature, label

    def get_sample_info(self, idx: int) -> Dict[str, Any]:
        """Get raw sample information for a given index."""
        if idx < 0 or idx >= len(self):
            raise IndexError(f"Index {idx} out of range [0, {len(self)})")
        return self.samples[idx]

    def get_statistics(self) -> Dict[str, Any]:
        """Get dataset statistics."""
        if not self.samples:
            return {"total": 0, "real": 0, "synthetic": 0}

        real_count = sum(1 for s in self.samples if not s.get("is_synthetic", False))
        synthetic_count = sum(1 for s in self.samples if s.get("is_synthetic", False))

        return {
            "total": len(self.samples),
            "real": real_count,
            "synthetic": synthetic_count,
            "version": self.version,
        }


def load_config(config_path: str = "config/config.yaml") -> Dict[str, Any]:
    """
    Load configuration from YAML file.

    Args:
        config_path: Path to the config file

    Returns:
        Configuration dictionary
    """
    if not os.path.exists(config_path):
        logger.warning(f"Config file {config_path} not found, using empty config")
        return {}

    with open(config_path, "r", encoding="utf-8") as f:
        config = yaml.safe_load(f)

    return config if config is not None else {}


def get_dataloaders(
    config: Optional[Dict[str, Any]] = None,
    version: str = "default",
    batch_size: int = 32,
    shuffle: bool = True,
    num_workers: int = 0,
    seed: Optional[int] = None,
) -> Dict[str, Any]:
    """
    Create dataloaders for train/val/test splits.

    Args:
        config: Configuration dictionary
        version: Dataset version (default: 'default')
        batch_size: Batch size for dataloaders
        shuffle: Whether to shuffle training data
        num_workers: Number of worker processes
        seed: Random seed

    Returns:
        Dictionary containing train/val/test dataloaders and dataset statistics
    """
    if config is None:
        config = load_config()

    # Extract seed from config if not provided
    if seed is None:
        system_cfg = config.get("system", {})
        if isinstance(system_cfg, dict):
            seed = system_cfg.get("seed", 42)
        else:
            seed = 42

    # Extract num_workers from config if not provided
    if num_workers == 0:
        system_cfg = config.get("system", {})
        if isinstance(system_cfg, dict):
            num_workers = system_cfg.get("num_workers", 0)

    # Create dataset
    dataset = SCICOQADataset(version=version, config=config, seed=seed)

    # Get split ratios from config
    split_ratios = None
    experiment_cfg = config.get("experiment", {})
    if isinstance(experiment_cfg, dict):
        runs = experiment_cfg.get("runs", [])
        if isinstance(runs, list) and runs:
            run = runs[0] if isinstance(runs[0], dict) else {}
            split_strategy = run.get("split_strategy", {})
            if isinstance(split_strategy, dict):
                ratios = split_strategy.get("ratios")
                if ratios is not None:
                    split_ratios = ratios

    # Split dataset
    train_ds, val_ds, test_ds = dataset.split_dataset(ratios=split_ratios, seed=seed)

    # Create dataloaders
    train_loader = torch.utils.data.DataLoader(
        train_ds,
        batch_size=batch_size,
        shuffle=shuffle,
        num_workers=num_workers,
        drop_last=False,
    )

    val_loader = torch.utils.data.DataLoader(
        val_ds,
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        drop_last=False,
    )

    test_loader = None
    if test_ds is not None:
        test_loader = torch.utils.data.DataLoader(
            test_ds,
            batch_size=batch_size,
            shuffle=False,
            num_workers=num_workers,
            drop_last=False,
        )

    # Get statistics
    stats = dataset.get_statistics()

    return {
        "train_loader": train_loader,
        "val_loader": val_loader,
        "test_loader": test_loader,
        "train_dataset": train_ds,
        "val_dataset": val_ds,
        "test_dataset": test_ds,
        "statistics": stats,
        "version": version,
    }


if __name__ == "__main__":
    # Basic test when run directly
    logging.basicConfig(level=logging.INFO)

    # Load config
    cfg = load_config()

    # Create dataset
    ds = SCICOQADataset(version="default", config=cfg)

    # Print statistics
    stats = ds.get_statistics()
    print(f"Dataset statistics: {stats}")

    # Test getitem
    if len(ds) > 0:
        feature, label = ds[0]
        print(f"Sample 0: feature shape={feature.shape}, label={label}")

    # Test split
    train_ds, val_ds, test_ds = ds.split_dataset()
    print(f"Split sizes: train={len(train_ds)}, val={len(val_ds)}, test={len(test_ds) if test_ds else 0}")

    # Test dataloaders
    loaders = get_dataloaders(config=cfg, version="default", batch_size=16)
    print(f"Dataloaders created: {loaders['statistics']}")
```
