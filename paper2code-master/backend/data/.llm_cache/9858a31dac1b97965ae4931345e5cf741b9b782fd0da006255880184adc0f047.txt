import sys
import json
import numpy as np
import cv2

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty in this case
        ocr_text = ""
        
        # Since OCR is empty, use visual information
        # Get basic image properties
        height, width = img.shape[:2]
        
        # Check if image is mostly grayscale (common for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        color_std = np.std(img, axis=2).mean()
        
        # Check color distribution
        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        saturation = hsv[:,:,1].mean()
        
        # For architecture diagrams: often have distinct blocks with different colors
        # For experiment results: often have lines/curves with specific colors
        # For datasets: often contain natural images or multiple samples
        
        # Simple heuristic based on color statistics
        if saturation < 30 and color_std < 30:
            # Very low saturation and low color variance - likely architecture diagram
            result_type = "architecture"
            description = "展示模型网络结构和数据流关系"
            components = []
            code_info = "需要实现神经网络模型结构"
        elif saturation > 50 and color_std > 50:
            # High saturation and variance - could be dataset or experiment result
            # Check aspect ratio - experiment results often wider
            aspect_ratio = width / height
            if aspect_ratio > 1.5:
                result_type = "experiment_result"
                description = "展示实验结果对比或性能变化"
                components = []
                code_info = "需要实现实验评估和结果可视化"
            else:
                result_type = "dataset"
                description = "展示数据集样本或输入示例"
                components = []
                code_info = "需要实现数据加载和预处理"
        else:
            # Medium saturation - ambiguous
            result_type = "other"
            description = "无法确定图片类型"
            components = []
            code_info = ""
        
        return {
            "type": result_type,
            "description": description,
            "important_components": components,
            "code_relevant_information": code_info
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
