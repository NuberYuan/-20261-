```python
import math
from typing import Optional, Tuple

import torch
import torch.nn as nn
import torch.nn.functional as F
import yaml
import os


class Llama213B(nn.Module):
    """
    Llama2-13B model structure for SELF-RAG.

    This class defines the model architecture with random initialization.
    No pretrained weights are loaded here; weight initialization is handled
    in training/train.py's initialize_model function.

    The model follows the standard Llama2 architecture with:
    - Token embeddings
    - Multiple decoder layers (self-attention + feed-forward)
    - Final RMSNorm
    - Output projection (tied with token embeddings)

    Note: This is a structural definition only. The actual parameter count
    may differ slightly from the reference implementation due to
    normalization layers and padding strategies (±5% tolerance).
    """

    def __init__(self, config: Optional[dict] = None):
        """
        Initialize the Llama2-13B model.

        Args:
            config: Configuration dictionary loaded from config/config.yaml.
                   If None, defaults are used for standalone testing.
        """
        super(Llama213B, self).__init__()

        if config is None:
            config = {}

        # Model hyperparameters based on Llama2-13B architecture
        # These are standard values for the 13B model
        self.vocab_size = 32000  # Llama2 vocabulary size
        self.hidden_size = 5120  # Hidden dimension for 13B model
        self.intermediate_size = 13824  # FFN intermediate size for 13B
        self.num_hidden_layers = 40  # Number of transformer layers for 13B
        self.num_attention_heads = 40  # Number of attention heads for 13B
        self.num_key_value_heads = 40  # Number of KV heads (grouped query attention)
        self.max_position_embeddings = 4096  # Maximum sequence length
        self.rms_norm_eps = 1e-5  # RMSNorm epsilon
        self.dropout_ratio = 0.0  # Default dropout ratio

        # Try to read from config if available
        model_config = config.get('model', {})
        if model_config:
            # These are not explicitly in the config but we check for safety
            self.dropout_ratio = model_config.get('dropout_ratio', 0.0)
            if self.dropout_ratio is None:
                self.dropout_ratio = 0.0

        # Training config for dropout
        training_config = config.get('training', {})
        if training_config:
            train_dropout = training_config.get('dropout_ratio', None)
            if train_dropout is not None:
                self.dropout_ratio = train_dropout

        # Ensure dropout_ratio is a valid float
        if self.dropout_ratio is None:
            self.dropout_ratio = 0.0
        self.dropout_ratio = float(self.dropout_ratio)

        # Token embeddings
        self.embed_tokens = nn.Embedding(
            num_embeddings=self.vocab_size,
            embedding_dim=self.hidden_size,
            padding_idx=0  # Index 0 is typically padding in Llama2
        )

        # Dropout layer
        self.embed_dropout = nn.Dropout(self.dropout_ratio)

        # Transformer layers
        self.layers = nn.ModuleList([
            self._create_decoder_layer(layer_idx)
            for layer_idx in range(self.num_hidden_layers)
        ])

        # Final layer norm
        self.norm = RMSNorm(self.hidden_size, eps=self.rms_norm_eps)

        # Output projection (tied with embeddings)
        self.lm_head = nn.Linear(
            self.hidden_size,
            self.vocab_size,
            bias=False
        )

        # Tie weights between embedding and output projection
        self.lm_head.weight = self.embed_tokens.weight

        # Initialize weights
        self._init_weights()

    def _create_decoder_layer(self, layer_idx: int) -> nn.Module:
        """
        Create a single decoder layer with self-attention and feed-forward.

        Args:
            layer_idx: Index of the layer (for potential layer-specific config)

        Returns:
            A decoder layer module
        """
        return DecoderLayer(
            hidden_size=self.hidden_size,
            intermediate_size=self.intermediate_size,
            num_attention_heads=self.num_attention_heads,
            num_key_value_heads=self.num_key_value_heads,
            rms_norm_eps=self.rms_norm_eps,
            dropout_ratio=self.dropout_ratio,
            max_position_embeddings=self.max_position_embeddings
        )

    def _init_weights(self):
        """
        Initialize model weights with appropriate distributions.
        This is a basic initialization; the main initialization is done
        in training/train.py's initialize_model function.
        """
        # Initialize embeddings
        nn.init.normal_(self.embed_tokens.weight, mean=0.0, std=0.02)

        # Initialize output projection (tied with embeddings, so already initialized)

        # Initialize decoder layers
        for layer in self.layers:
            layer._init_weights()

        # Initialize final norm (RMSNorm has no bias, just weight initialized to 1)
        nn.init.ones_(self.norm.weight)

    def forward(
        self,
        input_ids: torch.Tensor,
        attention_mask: Optional[torch.Tensor] = None,
        position_ids: Optional[torch.Tensor] = None,
        labels: Optional[torch.Tensor] = None,
        use_cache: bool = False,
        past_key_values: Optional[Tuple[Tuple[torch.Tensor]]] = None,
    ) -> dict:
        """
        Forward pass of the Llama2-13B model.

        Args:
            input_ids: Token indices of shape (batch_size, seq_len)
            attention_mask: Attention mask of shape (batch_size, seq_len)
                           with 1 for real tokens and 0 for padding
            position_ids: Position indices of shape (batch_size, seq_len)
            labels: Target token indices for loss computation
            use_cache: Whether to return cached key-value states
            past_key_values: Cached key-value states from previous forward passes

        Returns:
            dict containing:
                - logits: Output logits of shape (batch_size, seq_len, vocab_size)
                - loss: Cross-entropy loss if labels are provided
                - past_key_values: Cached key-value states if use_cache is True
        """
        batch_size, seq_len = input_ids.shape

        # Get token embeddings
        hidden_states = self.embed_tokens(input_ids)
        hidden_states = self.embed_dropout(hidden_states)

        # Generate position_ids if not provided
        if position_ids is None:
            position_ids = torch.arange(
                seq_len, dtype=torch.long, device=input_ids.device
            )
            position_ids = position_ids.unsqueeze(0).expand(batch_size, -1)

        # Create causal attention mask if not provided
        if attention_mask is None:
            # Create a causal mask (upper triangular)
            causal_mask = torch.triu(
                torch.ones((seq_len, seq_len), dtype=torch.bool, device=input_ids.device),
                diagonal=1
            )
            attention_mask = ~causal_mask.unsqueeze(0).unsqueeze(0)
            attention_mask = attention_mask.expand(batch_size, 1, seq_len, seq_len)
        else:
            # Convert attention_mask to 4D if it's 2D
            if attention_mask.dim() == 2:
                # Expand to 4D: (batch_size, 1, seq_len, seq_len)
                attention_mask = attention_mask.unsqueeze(1).unsqueeze(2)
                attention_mask = attention_mask.expand(batch_size, 1, seq_len, seq_len)

        # Process through decoder layers
        new_past_key_values = [] if use_cache else None

        for idx, layer in enumerate(self.layers):
            layer_outputs = layer(
                hidden_states=hidden_states,
                attention_mask=attention_mask,
                position_ids=position_ids,
                past_key_value=past_key_values[idx] if past_key_values is not None else None,
                use_cache=use_cache
            )
            hidden_states = layer_outputs[0]
            if use_cache:
                new_past_key_values.append(layer_outputs[1])

        # Apply final normalization
        hidden_states = self.norm(hidden_states)

        # Compute output logits
        logits = self.lm_head(hidden_states)

        # Compute loss if labels are provided
        loss = None
        if labels is not None:
            # Shift logits and labels for next-token prediction
            shift_logits = logits[:, :-1, :].contiguous()
            shift_labels = labels[:, 1:].contiguous()

            # Flatten the tokens
            loss = F.cross_entropy(
                shift_logits.view(-1, self.vocab_size),
                shift_labels.view(-1),
                ignore_index=-100  # Ignore padding tokens
            )

        output = {
            'logits': logits,
            'loss': loss,
        }

        if use_cache:
            output['past_key_values'] = tuple(new_past_key_values)

        return output


class RMSNorm(nn.Module):
    """
    Root Mean Square Layer Normalization.
    """

    def __init__(self, hidden_size: int, eps: float = 1e-5):
        """
        Initialize RMSNorm.

        Args:
            hidden_size: Size of the hidden dimension
            eps: Epsilon for numerical stability
        """
        super(RMSNorm, self).__init__()
        self.weight = nn.Parameter(torch.ones(hidden_size))
        self.eps = eps

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Apply RMS normalization.

        Args:
            x: Input tensor of shape (..., hidden_size)

        Returns:
            Normalized tensor of same shape
        """
        # Compute RMS
        rms = torch.sqrt(torch.mean(x.float() ** 2, dim=-1, keepdim=True) + self.eps)
        # Normalize
        x_normed = x / rms
        # Scale by weight
        return self.weight * x_normed.to(x.dtype)


class DecoderLayer(nn.Module):
    """
    Single decoder layer with self-attention and feed-forward network.
    """

    def __init__(
        self,
        hidden_size: int,
        intermediate_size: int,
        num_attention_heads: int,
        num_key_value_heads: int,
        rms_norm_eps: float,
        dropout_ratio: float,
        max_position_embeddings: int
    ):
        """
        Initialize decoder layer.

        Args:
            hidden_size: Hidden dimension size
            intermediate_size: FFN intermediate size
            num_attention_heads: Number of attention heads
            num_key_value_heads: Number of KV heads
            rms_norm_eps: RMSNorm epsilon
            dropout_ratio: Dropout probability
            max_position_embeddings: Maximum sequence length
        """
        super(DecoderLayer, self).__init__()

        self.hidden_size = hidden_size
        self.num_attention_heads = num_attention_heads
        self.num_key_value_heads = num_key_value_heads
        self.head_dim = hidden_size // num_attention_heads
        self.dropout_ratio = dropout_ratio

        # Self-attention
        self.self_attn = SelfAttention(
            hidden_size=hidden_size,
            num_attention_heads=num_attention_heads,
            num_key_value_heads=num_key_value_heads,
            head_dim=self.head_dim,
            dropout_ratio=dropout_ratio,
            max_position_embeddings=max_position_embeddings
        )

        # Feed-forward network
        self.mlp = FeedForward(
            hidden_size=hidden_size,
            intermediate_size=intermediate_size,
            dropout_ratio=dropout_ratio
        )

        # Layer norms
        self.input_layernorm = RMSNorm(hidden_size, eps=rms_norm_eps)
        self.post_attention_layernorm = RMSNorm(hidden_size, eps=rms_norm_eps)

    def _init_weights(self):
        """Initialize weights for this layer."""
        self.self_attn._init_weights()
        self.mlp._init_weights()
        nn.init.ones_(self.input_layernorm.weight)
        nn.init.ones_(self.post_attention_layernorm.weight)

    def forward(
        self,
        hidden_states: torch.Tensor,
        attention_mask: Optional[torch.Tensor] = None,
        position_ids: Optional[torch.Tensor] = None,
        past_key_value: Optional[Tuple[torch.Tensor, torch.Tensor]] = None,
        use_cache: bool = False
    ) -> Tuple[torch.Tensor, Optional[Tuple[torch.Tensor, torch.Tensor]]]:
        """
        Forward pass of decoder layer.

        Args:
            hidden_states: Input hidden states of shape (batch_size, seq_len, hidden_size)
            attention_mask: Attention mask
            position_ids: Position indices
            past_key_value: Cached key-value states
            use_cache: Whether to return cached states

        Returns:
            Tuple of (output_hidden_states, past_key_value)
        """
        # Self-attention with residual connection
        residual = hidden_states
        hidden_states = self.input_layernorm(hidden_states)

        attn_outputs = self.self_attn(
            hidden_states=hidden_states,
            attention_mask=attention_mask,
            position_ids=position_ids,
            past_key_value=past_key_value,
            use_cache=use_cache
        )
        hidden_states = attn_outputs[0]
        hidden_states = residual + hidden_states

        # Feed-forward with residual connection
        residual = hidden_states
        hidden_states = self.post_attention_layernorm(hidden_states)
        hidden_states = self.mlp(hidden_states)
        hidden_states = residual + hidden_states

        outputs = (hidden_states,)
        if use_cache:
            outputs = outputs + (attn_outputs[1],)

        return outputs


class SelfAttention(nn.Module):
    """
    Multi-head self-attention with grouped query attention.
    """

    def __init__(
        self,
        hidden_size: int,
        num_attention_heads: int,
        num_key_value_heads: int,
        head_dim: int,
        dropout_ratio: float,
        max_position_embeddings: int
    ):
        """
        Initialize self-attention.

        Args:
            hidden_size: Hidden dimension size
            num_attention_heads: Number of attention heads
            num_key_value_heads: Number of KV heads
            head_dim: Dimension per head
            dropout_ratio: Dropout probability
            max_position_embeddings: Maximum sequence length
        """
        super(SelfAttention, self).__init__()

        self.hidden_size = hidden_size
        self.num_attention_heads = num_attention_heads
        self.num_key_value_heads = num_key_value_heads
        self.head_dim = head_dim
        self.dropout_ratio = dropout_ratio
        self.max_position_embeddings = max_position_embeddings

        # Number of KV groups
        self.num_key_value_groups = num_attention_heads // num_key_value_heads

        # Projection layers
        self.q_proj = nn.Linear(hidden_size, num_attention_heads * head_dim, bias=False)
        self.k_proj = nn.Linear(hidden_size, num_key_value_heads * head_dim, bias=False)
        self.v_proj = nn.Linear(hidden_size, num_key_value_heads * head_dim, bias=False)
        self.o_proj = nn.Linear(num_attention_heads * head_dim, hidden_size, bias=False)

        # Dropout
        self.attn_dropout = nn.Dropout(dropout_ratio)

        # Rotary position embeddings
        self.rotary_emb = RotaryEmbedding(
            dim=head_dim,
            max_position_embeddings=max_position_embeddings
        )

    def _init_weights(self):
        """Initialize weights for attention layers."""
        nn.init.normal_(self.q_proj.weight, mean=0.0, std=0.02)
        nn.init.normal_(self.k_proj.weight, mean=0.0, std=0.02)
        nn.init.normal_(self.v_proj.weight, mean=0.0, std=0.02)
        nn.init.normal_(self.o_proj.weight, mean=0.0, std=0.02)

    def forward(
        self,
        hidden_states: torch.Tensor,
        attention_mask: Optional[torch.Tensor] = None,
        position_ids: Optional[torch.Tensor] = None,
        past_key_value: Optional[Tuple[torch.Tensor, torch.Tensor]] = None,
        use_cache: bool = False
    ) -> Tuple[torch.Tensor, Optional[Tuple[torch.Tensor, torch.Tensor]]]:
        """
        Forward pass of self-attention.

        Args:
            hidden_states: Input of shape (batch_size, seq_len, hidden_size)
            attention_mask: Attention mask
            position_ids: Position indices
            past_key_value: Cached key-value states
            use_cache: Whether to return cached states

        Returns:
            Tuple of (output, past_key_value)
        """
        batch_size, seq_len, _ = hidden_states.shape

        # Project to Q, K, V
        query_states = self.q_proj(hidden_states)
        key_states = self.k_proj(hidden_states)
        value_states = self.v_proj(hidden_states)

        # Reshape to (batch_size, num_heads, seq_len, head_dim)
        query_states = query_states.view(
            batch_size, seq_len, self.num_attention_heads, self.head_dim
        ).transpose(1, 2)
        key_states = key_states.view(
            batch_size, seq_len, self.num_key_value_heads, self.head_dim
        ).transpose(1, 2)
        value_states = value_states.view(
            batch_size, seq_len, self.num_key_value_heads, self.head_dim
        ).transpose(1, 2)

        # Apply rotary position embeddings
        if position_ids is None:
            position_ids = torch.arange(
                seq_len, dtype=torch.long, device=hidden_states.device
            )
            position_ids = position_ids.unsqueeze(0).expand(batch_size, -1)

        cos, sin = self.rotary_emb(value_states, position_ids)
        query_states, key_states = apply_rotary_pos_emb(query_states, key_states, cos, sin)

        # Handle past key-value states
        if past_key_value is not None:
            past_key, past_value = past_key_value
            key_states = torch.cat([past_key, key_states], dim=2)
            value_states = torch.cat([past_value, value_states], dim=2)

        # Store current key-value states
        current_key_value = (key_states, value_states) if use_cache else None

        # Repeat KV heads to match query heads
        if self.num_key_value_groups > 1:
            key_states = repeat_kv(key_states, self.num_key_value_groups)
            value_states = repeat_kv(value_states, self.num_key_value_groups)

        # Compute attention scores
        attn_weights = torch.matmul(query_states, key_states.transpose(2, 3)) / math.sqrt(self.head_dim)

        # Apply attention mask
        if attention_mask is not None:
            # Convert mask to float and add to attention weights
            # mask: 1 for attend, 0 for mask
            attn_weights = attn_weights + (1.0 - attention_mask.float()) * -10000.0

        # Apply softmax
        attn_weights = F.softmax(attn_weights, dim=-1, dtype=torch.float32).to(query_states.dtype)
        attn_weights = self.attn_dropout(attn_weights)

        # Apply attention to values
        attn_output = torch.matmul(attn_weights, value_states)

        # Reshape and project
        attn_output = attn_output.transpose(1, 2).contiguous()
        attn_output = attn_output.view(batch_size, seq_len, -1)
        attn_output = self.o_proj(attn_output)

        outputs = (attn_output,)
        if use_cache:
            outputs = outputs + (current_key_value,)

        return outputs


class FeedForward(nn.Module):
    """
    Feed-forward network with SwiGLU activation.
    """

    def __init__(self, hidden_size: int, intermediate_size: int, dropout_ratio: float):
        """
        Initialize feed-forward network.

        Args:
            hidden_size: Hidden dimension size
            intermediate_size: FFN intermediate size
            dropout_ratio: Dropout probability
        """
        super(FeedForward, self).__init__()

        self.gate_proj = nn.Linear(hidden_size, intermediate_size, bias=False)
        self.up_proj = nn.Linear(hidden_size, intermediate_size, bias=False)
        self.down_proj = nn.Linear(intermediate_size, hidden_size, bias=False)
        self.dropout = nn.Dropout(dropout_ratio)

    def _init_weights(self):
        """Initialize weights."""
        nn.init.normal_(self.gate_proj.weight, mean=0.0, std=0.02)
        nn.init.normal_(self.up_proj.weight, mean=0.0, std=0.02)
        nn.init.normal_(self.down_proj.weight, mean=0.0, std=0.02)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Forward pass.

        Args:
            x: Input tensor of shape (batch_size, seq_len, hidden_size)

        Returns:
            Output tensor of shape (batch_size, seq_len, hidden_size)
        """
        # SwiGLU activation
        gate = F.silu(self.gate_proj(x))
        up = self.up_proj(x)
        hidden = gate * up
        output = self.down_proj(hidden)
        output = self.dropout(output)
        return output


class RotaryEmbedding(nn.Module):
    """
    Rotary position embeddings.
    """

    def __init__(self, dim: int, max_position_embeddings: int = 4096, base: float = 10000.0):
        """
        Initialize rotary embeddings.

        Args:
            dim: Dimension of the embeddings
            max_position_embeddings: Maximum sequence length
            base: Base for the frequency calculation
        """
        super(RotaryEmbedding, self).__init__()

        self.dim = dim
        self.max_position_embeddings = max_position_embeddings
        self.base = base

        # Precompute inverse frequency
        inv_freq = 1.0 / (base ** (torch.arange(0, dim, 2).float() / dim))
        self.register_buffer("inv_freq", inv_freq, persistent=False)

        # Precompute cache for positions
        self._build_cache(max_position_embeddings)

    def _build_cache(self, max_position_embeddings: int):
        """Build the cache for cos and sin values."""
        position_ids = torch.arange(max_position_embeddings, dtype=torch.float32)
        inv_freq = self.inv_freq

        # Compute angles
        angles = torch.outer(position_ids, inv_freq)
        # Compute cos and sin
        cos_cache = torch.cos(angles).unsqueeze(0).unsqueeze(0)  # (1, 1, max_pos, dim/2)
        sin_cache = torch.sin(angles).unsqueeze(0).unsqueeze(0)

        self.register_buffer("cos_cache", cos_cache, persistent=False)
        self.register_buffer("sin_cache", sin_cache, persistent=False)

    def forward(
        self,
        x: torch.Tensor,
        position_ids: torch.Tensor
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Get cos and sin values for given positions.

        Args:
            x: Input tensor (used for device/dtype)
            position_ids: Position indices of shape (batch_size, seq_len)

        Returns:
            Tuple of (cos, sin) tensors
        """
        # Clamp position_ids to max_position_embeddings - 1
        position_ids = torch.clamp(position_ids, max=self.max_position_embeddings - 1)

        # Gather cos and sin values
        cos = self.cos_cache[:, :, position_ids, :]  # (batch_size, 1, seq_len, dim/2)
        sin = self.sin_cache[:, :, position_ids, :]

        # Expand to match head_dim
        cos = cos.expand(-1, x.shape[1], -1, -1)  # (batch_size, num_heads, seq_len, dim/2)
        sin = sin.expand(-1, x.shape[1], -1, -1)

        return cos.to(x.dtype), sin.to(x.dtype)


def apply_rotary_pos_emb(
    q: torch.Tensor,
    k: torch.Tensor,
    cos: torch.Tensor,
    sin: torch.Tensor
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Apply rotary position embeddings to query and key.

    Args:
        q: Query tensor of shape (batch_size, num_heads, seq_len, head_dim)
        k: Key tensor of shape (batch_size, num_heads, seq_len, head_dim)
        cos: Cosine values of shape (batch_size, num_heads, seq_len, head_dim/2)
        sin: Sine values of shape (batch_size, num_heads, seq_len, head_dim/2)

    Returns:
        Tuple of (rotated_q, rotated_k)
    """
    # Split into first half and second half
    q_first, q_second = q[..., : q.shape[-1] // 2], q[..., q.shape[-1] // 2:]
    k_first, k_second = k[..., : k.shape[-1] // 2], k[..., k.shape[-1] // 2:]

    # Apply rotation
    q_rotated = torch.cat(
        [
            q_first * cos - q_second * sin,
            q_first * sin + q_second * cos,
        ],
        dim=-1
    )
    k_rotated = torch.cat(
        [
            k_first * cos - k_second * sin,
            k_first * sin + k_second * cos,
        ],
        dim=-1
    )

    return q_rotated, k_rotated


def repeat_kv(
    x: torch.Tensor,
    n_rep: int
) -> torch.Tensor:
    """
    Repeat key-value heads to match the number of query heads.

    Args:
        x: Tensor of shape (batch_size, num_kv_heads, seq_len, head_dim)
        n_rep: Number of repetitions

    Returns:
        Tensor of shape (batch_size, num_kv_heads * n_rep, seq_len, head_dim)
    """
    batch_size, num_kv_heads, seq_len, head_dim = x.shape
    if n_rep == 1:
        return x
    x = x[:, :, None, :, :].expand(batch_size, num_kv_heads, n_rep, seq_len, head_dim)
    return x.reshape(batch_size, num_kv_heads * n_rep, seq_len, head_dim)


def load_config(config_path: str = "config/config.yaml") -> dict:
    """
    Load configuration from YAML file.

    Args:
        config_path: Path to the config file

    Returns:
        Configuration dictionary
    """
    if not os.path.exists(config_path):
        return {}

    with open(config_path, 'r', encoding='utf-8') as f:
        config = yaml.safe_load(f)

    return config if config is not None else {}


def create_llama213b(config: Optional[dict] = None) -> Llama213B:
    """
    Factory function to create a Llama213B model.

    Args:
        config: Configuration dictionary

    Returns:
        Llama213B model instance
    """
    if config is None:
        config = load_config()

    return Llama213B(config=config)
```
