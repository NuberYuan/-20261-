import sys
import json
import numpy as np
import cv2

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        if img is None:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty in this case
        ocr_text = ""
        
        # Since OCR is empty, use visual information
        # Get basic image properties
        height, width = img.shape[:2]
        
        # Check if image is mostly white/light background (typical for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        mean_brightness = np.mean(gray)
        
        # Check color variance (architecture diagrams often have low color variance)
        color_variance = np.std(img)
        
        # Check aspect ratio (architecture diagrams often wider than tall)
        aspect_ratio = width / height if height > 0 else 0
        
        # Since OCR is empty and we can't use complex CV,
        # we'll make a conservative judgment based on basic visual properties
        # Architecture diagrams typically have:
        # - Light background
        # - Multiple distinct regions (blocks)
        # - Often wider than tall
        
        # Check if image has multiple distinct color regions
        # Use simple color quantization
        small = cv2.resize(img, (64, 64))
        hsv = cv2.cvtColor(small, cv2.COLOR_BGR2HSV)
        hue_hist = cv2.calcHist([hsv], [0], None, [180], [0, 180])
        distinct_hues = np.sum(hue_hist > 0.01 * np.max(hue_hist))
        
        # Check for text-like patterns (high contrast regions)
        _, binary = cv2.threshold(gray, 200, 255, cv2.THRESH_BINARY)
        dark_ratio = np.sum(binary < 128) / (height * width)
        
        # Architecture diagrams often have:
        # - Light background (high mean brightness)
        # - Multiple distinct regions
        # - Some text labels (dark pixels)
        
        if mean_brightness > 150 and distinct_hues > 5 and dark_ratio > 0.01:
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": [],
                "code_relevant_information": "需要实现模型网络结构，包含多个处理模块"
            }
        elif mean_brightness < 100 and distinct_hues < 10:
            # Dark image with few colors - could be experiment result
            return {
                "type": "experiment_result",
                "description": "展示实验结果数据",
                "important_components": [],
                "code_relevant_information": "需要实现实验评估和结果可视化"
            }
        else:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
            
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
