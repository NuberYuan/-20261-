import sys
import json
import numpy as np
import cv2
from PIL import Image

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty for this image
        ocr_text = ""
        
        # Architecture keywords
        arch_keywords = [
            "CNN", "Transformer", "Attention", "Encoder", "Decoder",
            "Block", "Embedding", "Classifier", "Feature Extractor",
            "AlexNet", "ResNet", "VGG", "GoogLeNet"
        ]
        
        # Experiment result keywords
        exp_keywords = [
            "Loss", "Epoch", "Precision", "Recall", "F1",
            "Accuracy", "准确率", "损失"
        ]
        
        # Check OCR text for architecture
        arch_components = []
        for kw in arch_keywords:
            if kw.lower() in ocr_text.lower():
                arch_components.append(kw)
        
        # Check OCR text for experiment results
        exp_components = []
        for kw in exp_keywords:
            if kw.lower() in ocr_text.lower():
                exp_components.append(kw)
        
        # Since OCR is empty, use visual features
        # Get basic image properties
        height, width = img.shape[:2]
        
        # Check if image looks like a chart/plot (experiment result)
        # Convert to grayscale and check for typical chart characteristics
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        
        # Check color distribution - charts often have limited colors
        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        
        # Check for grid-like patterns (common in charts)
        # Look for horizontal and vertical lines
        edges_x = cv2.Sobel(gray, cv2.CV_64F, 1, 0, ksize=3)
        edges_y = cv2.Sobel(gray, cv2.CV_64F, 0, 1, ksize=3)
        
        # Calculate edge density
        edge_density = (np.abs(edges_x) > 50).sum() + (np.abs(edges_y) > 50).sum()
        total_pixels = height * width
        edge_ratio = edge_density / total_pixels
        
        # Check for text-like regions (high contrast areas)
        _, binary = cv2.threshold(gray, 200, 255, cv2.THRESH_BINARY)
        white_ratio = (binary == 255).sum() / total_pixels
        
        # Check if image has multiple distinct color regions (typical for diagrams)
        # Quantize colors
        quantized = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        quantized = quantized // 64 * 64
        unique_colors = len(np.unique(quantized.reshape(-1, 3), axis=0))
        
        # Decision logic
        # Since OCR is empty, rely on visual characteristics
        
        # Check for chart-like appearance (experiment result)
        # Charts typically have: white background, grid lines, limited colors
        if white_ratio > 0.3 and edge_ratio > 0.01 and unique_colors < 50:
            # Likely a chart/plot
            return {
                "type": "experiment_result",
                "description": "展示实验结果的图表，包含性能指标变化",
                "important_components": ["实验指标"],
                "code_relevant_information": "需要实现数据可视化模块，展示实验结果对比"
            }
        
        # Check for architecture diagram
        # Architecture diagrams typically have: multiple colored blocks, arrows, moderate complexity
        if unique_colors > 20 and unique_colors < 200 and edge_ratio > 0.005:
            # Check for block-like structures (high contrast regions)
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": [],
                "code_relevant_information": "需要实现模型网络结构定义和参数配置"
            }
        
        # Check for dataset images
        # Dataset images typically have: natural colors, high color variety
        if unique_colors > 200:
            return {
                "type": "dataset",
                "description": "展示数据集样本或输入示例",
                "important_components": [],
                "code_relevant_information": "需要实现数据加载和预处理模块"
            }
        
        # Default to other if cannot determine
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
