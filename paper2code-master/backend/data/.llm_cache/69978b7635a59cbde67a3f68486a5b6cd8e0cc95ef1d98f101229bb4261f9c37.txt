import sys
import json
import numpy as np
import cv2
from PIL import Image

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text provided externally
        ocr_text = """Electricity
Traffic
Retail
Volatility
Average
Max
Ave
Min
21.6%
30.0%
23.4%
20.0%
10.2%
10.0%
{7.6%
5.6%
4.9%
1.5%
2.0%
3.2%
3.6%
0.0%
-10.0%
No
Gating
No Static
No Variable
No Self-
No Local
Encoder"""
        
        # Convert OCR text to lowercase for matching
        ocr_lower = ocr_text.lower()
        
        # Check for experiment_result indicators
        experiment_indicators = [
            'accuracy', 'loss', 'epoch', 'precision', 'recall', 'f1',
            'volatility', 'average', 'max', 'min', '%'
        ]
        
        # Check for architecture indicators
        architecture_indicators = [
            'cnn', 'transformer', 'attention', 'encoder', 'decoder',
            'block', 'embedding', 'classifier', 'feature extractor',
            'alexnet', 'resnet', 'vgg', 'googlenet'
        ]
        
        # Check for dataset indicators
        dataset_indicators = [
            'sample', 'example', 'input image', 'dataset', 'class'
        ]
        
        # Determine type based on priority
        is_experiment = any(ind in ocr_lower for ind in experiment_indicators)
        is_architecture = any(ind in ocr_lower for ind in architecture_indicators)
        is_dataset = any(ind in ocr_lower for ind in dataset_indicators)
        
        # Priority: experiment_result > architecture > dataset > other
        if is_experiment:
            img_type = "experiment_result"
            description = "展示实验结果对比，包含性能指标和统计信息"
            
            # Extract important components
            components = []
            if 'volatility' in ocr_lower:
                components.append('Volatility')
            if 'average' in ocr_lower:
                components.append('Average')
            if 'max' in ocr_lower:
                components.append('Max')
            if 'min' in ocr_lower:
                components.append('Min')
            
            # Add model names if present
            for arch in ['alexnet', 'resnet', 'vgg', 'googlenet', 'transformer', 'cnn']:
                if arch in ocr_lower:
                    components.append(arch.upper())
            
            code_info = "需要实现实验对比分析模块，处理性能指标数据并生成可视化结果"
            
        elif is_architecture:
            img_type = "architecture"
            description = "展示模型网络结构和数据流关系"
            
            # Extract architecture components
            components = []
            if 'encoder' in ocr_lower:
                components.append('Encoder')
            if 'attention' in ocr_lower:
                components.append('Attention Module')
            if 'block' in ocr_lower:
                components.append('Block')
            if 'embedding' in ocr_lower:
                components.append('Embedding')
            if 'classifier' in ocr_lower:
                components.append('Classifier')
            if 'feature extractor' in ocr_lower:
                components.append('Feature Extractor')
            
            code_info = "需要实现模型架构定义，包含特征提取和分类模块"
            
        elif is_dataset:
            img_type = "dataset"
            description = "展示数据集样本或输入示例"
            components = []
            code_info = "需要实现数据加载和预处理模块"
            
        else:
            img_type = "other"
            description = "无法确定图片类型"
            components = []
            code_info = ""
        
        # Clean up components - remove duplicates and empty strings
        components = list(dict.fromkeys([c for c in components if c]))
        
        return {
            "type": img_type,
            "description": description,
            "important_components": components,
            "code_relevant_information": code_info
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
