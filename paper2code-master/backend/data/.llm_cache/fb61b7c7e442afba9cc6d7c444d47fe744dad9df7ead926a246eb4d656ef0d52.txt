```python
import os

from typing import Dict, List, Optional, Tuple, Any, Union

import yaml

import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader

# Add project root to path for imports
PROJECT_ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
if PROJECT_ROOT not in sys.path:
    sys.path.insert(0, PROJECT_ROOT)

from data.dataset import (

logger = logging.getLogger(__name__)


def load_config(config_path: Optional[str] = None) -> Dict[str, Any]:
    """
    Load configuration from YAML file.

    Args:
        config_path: Path to the config file. If None, uses default path.

    Returns:
        Configuration dictionary.
    """
    if config_path is None:
        config_path = os.path.join(PROJECT_ROOT, "config", "config.yaml")

    with open(config_path, "r", encoding="utf-8") as f:
        config = yaml.safe_load(f)

    return config if config is not None else {}


def initialize_model(
    model_name: str = "self-rag-7b",
    config: Optional[Dict[str, Any]] = None,
    training_mode: str = "scratch",
) -> nn.Module:
    """
    Initialize a model based on the specified architecture and training mode.

    Args:
        model_name: Name of the model architecture ('self-rag-7b', 'self-rag-13b',
                   'gpt4-data-collection', or 'contriever-msmarco').
        config: Configuration dictionary loaded from config.yaml.
        training_mode: Training mode ('scratch' or 'transfer').

    Returns:
        Initialized PyTorch model.

    Raises:
        ValueError: If model_name is not recognized.
    """
    if config is None:
        config = load_config()

    # Determine model class based on model_name
    if model_name == "self-rag-7b":
        model = Llama27B(config=config)
    elif model_name == "self-rag-13b":
        model = Llama213B(config=config)
    elif model_name == "gpt4-data-collection":
        model = GPT4DataCollection(config=config)
    elif model_name == "contriever-msmarco":
        model = ContrieverMSMARCORetriever(config=config)
    else:
        raise ValueError(f"Unknown model name: {model_name}")

    # For scratch training, model is already randomly initialized
    # No pretrained weights are loaded
    if training_mode == "scratch":
        logger.info(f"Model {model_name} initialized with random weights (scratch mode)")
        return model

    # For transfer mode, we would load pretrained weights
    # But per requirements, we do NOT load any pretrained weights
    # This is a structural definition only
    logger.info(
        f"Model {model_name} initialized in {training_mode} mode "
        f"(no pretrained weights loaded per requirements)"
    )
    return model


def train_model(
    model: nn.Module,
    train_loader: DataLoader,
    val_loader: Optional[DataLoader] = None,
    config: Optional[Dict[str, Any]] = None,
    device: Optional[torch.device] = None,
    run_name: str = "self-rag",
) -> Dict[str, Any]:
    """
    Train the model for the specified number of epochs.

    Args:
        model: The PyTorch model to train.
        train_loader: DataLoader for training data.
        val_loader: Optional DataLoader for validation data. If None, validation is skipped.
        config: Configuration dictionary loaded from config.yaml.
        device: Device to train on. If None, auto-detected.
        run_name: Name of the training run (for logging).

    Returns:
        Dictionary containing training history and best model state.
    """
    if config is None:
        config = load_config()

    # Determine device
    if device is None:
        device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

    # Move model to device
    model = model.to(device)

    # Get training hyperparameters from config
    training_config = config.get("training", {})
    epochs = training_config.get("epochs", 30)
    if epochs is None:
        epochs = 30

    # Get optimizer configuration
    optimizer_config = training_config.get("optimizer", {})
    optimizer_name = optimizer_config.get("name", "adamw")
    if optimizer_name is None:
        optimizer_name = "adamw"

    lr = optimizer_config.get("lr", 1e-4)
    if lr is None:
        lr = 1e-4

    weight_decay = optimizer_config.get("weight_decay", 0.01)
    if weight_decay is None:
        weight_decay = 0.01

    momentum = optimizer_config.get("momentum", 0.9)
    if momentum is None:
        momentum = 0.9

    # Create optimizer
    if optimizer_name.lower() == "adam":
        optimizer = optim.Adam(
            model.parameters(),
            lr=float(lr),
            weight_decay=float(weight_decay),
        )
    elif optimizer_name.lower() == "adamw":
        optimizer = optim.AdamW(
            model.parameters(),
            lr=float(lr),
            weight_decay=float(weight_decay),
        )
    elif optimizer_name.lower() == "sgd":
        optimizer = optim.SGD(
            model.parameters(),
            lr=float(lr),
            momentum=float(momentum),
            weight_decay=float(weight_decay),
        )
    else:
        # Default to AdamW
        logger.warning(f"Unknown optimizer '{optimizer_name}', using AdamW")
        optimizer = optim.AdamW(
            model.parameters(),
            lr=float(lr),
            weight_decay=float(weight_decay),
        )

    # Get scheduler configuration
    scheduler_config = training_config.get("scheduler", {})
    scheduler_name = scheduler_config.get("name", "step_lr")
    if scheduler_name is None:
        scheduler_name = "step_lr"

    step_size = scheduler_config.get("step_size", 10)
    if step_size is None:
        step_size = 10

    gamma = scheduler_config.get("gamma", 0.1)
    if gamma is None:
        gamma = 0.1

    # Create scheduler
    if scheduler_name.lower() in ["steplr", "step_lr", "step"]:
        scheduler = optim.lr_scheduler.StepLR(
            optimizer,
            step_size=int(step_size),
            gamma=float(gamma),
        )
    elif scheduler_name.lower() in ["cosine", "cosineannealing"]:
        scheduler = optim.lr_scheduler.CosineAnnealingLR(
            optimizer,
            T_max=epochs,
        )
    else:
        # Default to StepLR
        logger.warning(f"Unknown scheduler '{scheduler_name}', using StepLR")
        scheduler = optim.lr_scheduler.StepLR(
            optimizer,
            step_size=int(step_size),
            gamma=float(gamma),
        )

    # Loss function: Standard next token objective (cross-entropy)
    criterion = nn.CrossEntropyLoss(ignore_index=-100)

    # Get checkpoint configuration
    checkpoint_config = config.get("checkpoint", {})
    save_best = checkpoint_config.get("save_best", True)
    best_metric = checkpoint_config.get("best_metric", "val_accuracy")
    best_mode = checkpoint_config.get("best_mode", "max")
    save_final = checkpoint_config.get("save_final", True)
    save_optimizer_state = checkpoint_config.get("save_optimizer_state", True)

    # Get logging configuration
    logging_config = config.get("logging", {})
    log_dir = logging_config.get("log_dir", "./logs")
    os.makedirs(log_dir, exist_ok=True)

    # Get system configuration
    system_config = config.get("system", {})
    seed = system_config.get("seed", 42)

    # Set random seed for reproducibility
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)

    # Training history
    history = {
        "train_loss": [],
        "val_loss": [],
        "val_accuracy": [],
        "epochs": [],
        "learning_rates": [],
    }

    # Best model tracking
    best_val_accuracy = 0.0
    best_val_loss = float("inf")
    best_model_state = None
    best_epoch = -1

    # Determine if we should validate
    should_validate = val_loader is not None

    # Training loop
    logger.info(f"Starting training for {epochs} epochs on {device}")
    logger.info(f"Train batches: {len(train_loader)}")
    if should_validate:
        logger.info(f"Validation batches: {len(val_loader)}")

    for epoch in range(1, epochs + 1):
        epoch_start_time = time.time()

        # Training phase
        model.train()
        train_loss = 0.0
        train_batches = 0

        for batch_idx, batch in enumerate(train_loader):
            # Prepare inputs
            # The dataset returns instruction/output pairs
            # For next-token prediction, we need to tokenize and create input/target pairs
            # Since we don't have a tokenizer in this setup, we use a simple approach:
            # Convert text to token indices using a simple character-level encoding
            # This is a placeholder for the actual tokenization process

            # For the training loop to work, we need to handle the batch data
            # The batch contains: instruction, output, dataset, id, context
            # We'll create a simple tokenization for demonstration purposes

            # Get batch data
            instructions = batch["instruction"]
            outputs = batch["output"]

            # Create input_ids and labels for next-token prediction
            # This is a simplified approach - in practice, you'd use a proper tokenizer
            input_ids_list = []
            labels_list = []

            for instr, out in zip(instructions, outputs):
                # Combine instruction and output for training
                text = f"{instr} {out}"

                # Simple character-level tokenization (placeholder)
                # Convert characters to token IDs (0-255 for ASCII)
                tokens = [ord(c) % 32000 for c in text[:512]]  # Limit sequence length

                if len(tokens) < 2:
                    tokens = [0, 0]  # Ensure at least 2 tokens

                # Pad to fixed length (e.g., 512)
                seq_len = 512
                if len(tokens) < seq_len:
                    tokens = tokens + [0] * (seq_len - len(tokens))
                else:
                    tokens = tokens[:seq_len]

                # Create input_ids (all tokens) and labels (shifted by 1)
                input_ids = torch.tensor(tokens, dtype=torch.long)
                labels = torch.tensor(tokens, dtype=torch.long)

                input_ids_list.append(input_ids)
                labels_list.append(labels)

            # Stack into batch tensors
            input_ids = torch.stack(input_ids_list).to(device)
            labels = torch.stack(labels_list).to(device)

            # Create attention mask (1 for non-padding, 0 for padding)
            attention_mask = (input_ids != 0).long().to(device)

            # Forward pass
            optimizer.zero_grad()

            try:
                # Try model forward with labels
                outputs = model(
                    input_ids=input_ids,
                    attention_mask=attention_mask,
                    labels=labels,
                )

                if isinstance(outputs, dict):
                    logits = outputs.get("logits")
                    loss = outputs.get("loss")
                    if loss is None:
                        # Compute loss from logits
                        shift_logits = logits[:, :-1, :].contiguous()
                        shift_labels = labels[:, 1:].contiguous()
                        loss = criterion(
                            shift_logits.view(-1, logits.size(-1)),
                            shift_labels.view(-1),
                        )
                else:
                    # Model returns logits directly
                    logits = outputs
                    shift_logits = logits[:, :-1, :].contiguous()
                    shift_labels = labels[:, 1:].contiguous()
                    loss = criterion(
                        shift_logits.view(-1, logits.size(-1)),
                        shift_labels.view(-1),
                    )
            except TypeError:
                # Model doesn't accept labels parameter
                logits = model(input_ids=input_ids, attention_mask=attention_mask)
                if isinstance(logits, dict):
                    logits = logits.get("logits")

                shift_logits = logits[:, :-1, :].contiguous()
                shift_labels = labels[:, 1:].contiguous()
                loss = criterion(
                    shift_logits.view(-1, logits.size(-1)),
                    shift_labels.view(-1),
                )

            # Backward pass
            loss.backward()

            # Gradient clipping
            torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)

            # Optimizer step
            optimizer.step()

            # Accumulate loss
            train_loss += loss.item()
            train_batches += 1

            # Log progress
            if (batch_idx + 1) % 10 == 0:
                logger.info(
                    f"Epoch {epoch}/{epochs} | Batch {batch_idx + 1}/{len(train_loader)} | "
                    f"Loss: {loss.item():.4f}"
                )

        # Compute average training loss
        avg_train_loss = train_loss / max(train_batches, 1)
        history["train_loss"].append(avg_train_loss)
        history["epochs"].append(epoch)

        # Get current learning rate
        current_lr = optimizer.param_groups[0]["lr"]
        history["learning_rates"].append(current_lr)

        # Validation phase (only if val_loader is provided)
        val_loss = None
        val_accuracy = None

        if should_validate:
            model.eval()
            val_loss = 0.0
            val_correct = 0
            val_total = 0
            val_batches = 0

            with torch.no_grad():
                for batch in val_loader:
                    # Prepare validation data
                    instructions = batch["instruction"]
                    outputs = batch["output"]

                    input_ids_list = []
                    labels_list = []

                    for instr, out in zip(instructions, outputs):
                        text = f"{instr} {out}"
                        tokens = [ord(c) % 32000 for c in text[:512]]

                        if len(tokens) < 2:
                            tokens = [0, 0]

                        seq_len = 512
                        if len(tokens) < seq_len:
                            tokens = tokens + [0] * (seq_len - len(tokens))
                        else:
                            tokens = tokens[:seq_len]

                        input_ids = torch.tensor(tokens, dtype=torch.long)
                        labels = torch.tensor(tokens, dtype=torch.long)

                        input_ids_list.append(input_ids)
                        labels_list.append(labels)

                    input_ids = torch.stack(input_ids_list).to(device)
                    labels = torch.stack(labels_list).to(device)
                    attention_mask = (input_ids != 0).long().to(device)

                    try:
                        outputs = model(
                            input_ids=input_ids,
                            attention_mask=attention_mask,
                            labels=labels,
                        )

                        if isinstance(outputs, dict):
                            logits = outputs.get("logits")
                            loss = outputs.get("loss")
                            if loss is None:
                                shift_logits = logits[:, :-1, :].contiguous()
                                shift_labels = labels[:, 1:].contiguous()
                                loss = criterion(
                                    shift_logits.view(-1, logits.size(-1)),
                                    shift_labels.view(-1),
                                )
                        else:
                            logits = outputs
                            shift_logits = logits[:, :-1, :].contiguous()
                            shift_labels = labels[:, 1:].contiguous()
                            loss = criterion(
                                shift_logits.view(-1, logits.size(-1)),
                                shift_labels.view(-1),
                            )
                    except TypeError:
                        logits = model(input_ids=input_ids, attention_mask=attention_mask)
                        if isinstance(logits, dict):
                            logits = logits.get("logits")

                        shift_logits = logits[:, :-1, :].contiguous()
                        shift_labels = labels[:, 1:].contiguous()
                        loss = criterion(
                            shift_logits.view(-1, logits.size(-1)),
                            shift_labels.view(-1),
                        )

                    val_loss += loss.item()
                    val_batches += 1

                    # Calculate accuracy (next token prediction accuracy)
                    if isinstance(logits, dict):
                        logits = logits.get("logits")

                    predictions = logits.argmax(dim=-1)
                    # Compare predictions with labels (shifted)
                    pred_shift = predictions[:, :-1]
                    label_shift = labels[:, 1:]
                    mask = label_shift != -100
                    correct = (pred_shift == label_shift) & mask
                    val_correct += correct.sum().item()
                    val_total += mask.sum().item()

            avg_val_loss = val_loss / max(val_batches, 1)
            val_accuracy = val_correct / max(val_total, 1)

            history["val_loss"].append(avg_val_loss)
            history["val_accuracy"].append(val_accuracy)

            logger.info(
                f"Epoch {epoch}/{epochs} | Train Loss: {avg_train_loss:.4f} | "
                f"Val Loss: {avg_val_loss:.4f} | Val Acc: {val_accuracy:.4f}"
            )

            # Save best model based on validation metric
            if save_best:
                if best_mode == "max" and val_accuracy > best_val_accuracy:
                    best_val_accuracy = val_accuracy
                    best_epoch = epoch
                    best_model_state = copy.deepcopy(model.state_dict())
                    logger.info(f"New best model! Val Acc: {val_accuracy:.4f} (epoch {epoch})")
                elif best_mode == "min" and avg_val_loss < best_val_loss:
                    best_val_loss = avg_val_loss
                    best_epoch = epoch
                    best_model_state = copy.deepcopy(model.state_dict())
                    logger.info(f"New best model! Val Loss: {avg_val_loss:.4f} (epoch {epoch})")
        else:
            logger.info(f"Epoch {epoch}/{epochs} | Train Loss: {avg_train_loss:.4f}")

        # Step scheduler
        scheduler.step()

        epoch_time = time.time() - epoch_start_time
        logger.info(f"Epoch {epoch} completed in {epoch_time:.2f}s")

    # Save final model
    final_model_path = None
    if save_final:
        final_model_path = os.path.join(log_dir, f"{run_name}_final.pt")
        save_dict = {
            "model_state_dict": model.state_dict(),
            "epoch": epochs,
            "config": config,
        }
        if save_optimizer_state:
            save_dict["optimizer_state_dict"] = optimizer.state_dict()

        torch.save(save_dict, final_model_path)
        logger.info(f"Final model saved to {final_model_path}")

    # Save best model
    best_model_path = None
    if save_best and best_model_state is not None:
        best_model_path = os.path.join(log_dir, f"{run_name}_best.pt")
        save_dict = {
            "model_state_dict": best_model_state,
            "epoch": best_epoch,
            "best_val_accuracy": best_val_accuracy,
            "best_val_loss": best_val_loss,
            "config": config,
        }
        if save_optimizer_state:
            save_dict["optimizer_state_dict"] = optimizer.state_dict()

        torch.save(save_dict, best_model_path)
        logger.info(f"Best model saved to {best_model_path} (epoch {best_epoch})")

    # Compile results
    results = {
        "history": history,
        "best_val_accuracy": best_val_accuracy,
        "best_val_loss": best_val_loss,
        "best_epoch": best_epoch,
        "final_model_path": final_model_path,
        "best_model_path": best_model_path,
    }

    return results


def save_model(
    model: nn.Module,
    path: str,
    config: Optional[Dict[str, Any]] = None,
    optimizer: Optional[torch.optim.Optimizer] = None,
    epoch: Optional[int] = None,
    metrics: Optional[Dict[str, float]] = None,
) -> str:
    """
    Save model weights and optionally optimizer state to disk.

    Args:
        model: The PyTorch model to save.
        path: Path where to save the model.
        config: Configuration dictionary loaded from config.yaml.
        optimizer: Optional optimizer whose state should be saved.
        epoch: Optional epoch number to save.
        metrics: Optional metrics dictionary to save.

    Returns:
        The path where the model was saved.
    """
    if config is None:
        config = load_config()

    # Ensure directory exists
    os.makedirs(os.path.dirname(path) if os.path.dirname(path) else ".", exist_ok=True)

    # Prepare save dictionary
    save_dict = {
        "model_state_dict": model.state_dict(),
        "config": config,
    }

    if optimizer is not None:
        save_dict["optimizer_state_dict"] = optimizer.state_dict()

    if epoch is not None:
        save_dict["epoch"] = epoch

    if metrics is not None:
        save_dict["metrics"] = metrics

    # Save to disk
    torch.save(save_dict, path)
    logger.info(f"Model saved to {path}")

    return path


def main() -> None:
    """
    Main entry point for training.
    """
    # Set up logging
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s - %(name)s - %(levelname)s - %(message)s",
    )

    # Load config
    config = load_config()

    # Get reproduction mode
    reproduction_mode = config.get("reproduction_mode", "engineering")

    # Get experiment runs
    experiment_config = config.get("experiment", {})
    runs = experiment_config.get("runs", [])

    if not runs:
        logger.warning("No experiment runs found in config. Using default run.")
        runs = [
            {
                "name": "self-rag-7b-scratch",
                "model": "self-rag-7b",
                "training_mode": "scratch",
            }
        ]

    # Get system configuration
    system_config = config.get("system", {})
    device_str = system_config.get("device", None)
    if device_str is None:
        device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    else:
        device = torch.device(device_str)

    num_workers = system_config.get("num_workers", 4)
    seed = system_config.get("seed", 42)

    # Get training configuration
    training_config = config.get("training", {})
    batch_size = training_config.get("batch_size", 32)
    if batch_size is None:
        batch_size = 32

    # Get split strategy
    split_strategy = config.get("split_strategy", {})
    val_split_ratio = split_strategy.get("val_split_ratio", 0.2)

    # Get checkpoint configuration
    checkpoint_config = config.get("checkpoint", {})
    resume_from = checkpoint_config.get("resume_from", None)

    # Process each run
    for run in runs:
        run_name = run.get("name", "self-rag")
        model_name = run.get("model", "self-rag-7b")
        training_mode = run.get("training_mode", "scratch")
        dataset_version = run.get("dataset_version", "default")

        logger.info(f"=== Starting run: {run_name} ===")
        logger.info(f"Model: {model_name}, Mode: {training_mode}, Version: {dataset_version}")

        # Initialize model
        model = initialize_model(
            model_name=model_name,
            config=config,
            training_mode=training_mode,
        )

        # Count parameters
        param_count = sum(p.numel() for p in model.parameters())
        logger.info(f"Model parameter count: {param_count:,}")

        # Create dataloaders
        if reproduction_mode == "engineering":
            # Engineering mode: train -> val -> best -> test
            # Split training set into train/val (80/20)
            train_loader, val_loader, test_loader = get_dataloaders(
                config=config,
                version=dataset_version,
                batch_size=batch_size,
                num_workers=num_workers,
                seed=seed,
            )
            logger.info(
                f"Engineering mode: Train={len(train_loader.dataset)}, "
                f"Val={len(val_loader.dataset)}, Test={len(test_loader.dataset)}"
            )
        else:
            # Strict mode: train -> final -> test (no validation)
            # Create only train and test loaders
            # For strict mode, we still use get_dataloaders but ignore val_loader
            train_loader, _, test_loader = get_dataloaders(
                config=config,
                version=dataset_version,
                batch_size=batch_size,
                num_workers=num_workers,
                seed=seed,
            )
            val_loader = None
            logger.info(
                f"Strict mode: Train={len(train_loader.dataset)}, "
                f"Test={len(test_loader.dataset)} (no validation)"
            )

        # Train model
        results = train_model(
            model=model,
            train_loader=train_loader,
            val_loader=val_loader,
            config=config,
            device=device,
            run_name=run_name,
        )

        # Log results
        logger.info(f"Run {run_name} completed:")
        logger.info(f"  Best val accuracy: {results['best_val_accuracy']:.4f}")
        logger.info(f"  Best val loss: {results['best_val_loss']:.4f}")
        logger.info(f"  Best epoch: {results['best_epoch']}")

        # Save training history
        log_dir = config.get("logging", {}).get("log_dir", "./logs")
        os.makedirs(log_dir, exist_ok=True)
        history_path = os.path.join(log_dir, f"{run_name}_history.json")
        with open(history_path, "w", encoding="utf-8") as f:
            json.dump(results["history"], f, indent=2)
        logger.info(f"Training history saved to {history_path}")

    logger.info("All runs completed successfully!")


if __name__ == "__main__":
    main()
```
