import sys
import json
import numpy as np
import cv2
import os

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty in this case
        ocr_text = ""
        
        # Since OCR is empty, use visual features and filename context
        # Get image basic properties
        height, width = img.shape[:2]
        
        # Check if image is mostly grayscale (common for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        color_std = np.std(img, axis=2).mean()
        
        # Check color distribution
        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        saturation = hsv[:,:,1].mean()
        
        # Simple heuristics based on visual characteristics
        # Architecture diagrams often have high contrast, block-like structures
        # Experiment results often have lines/curves
        # Dataset images often have natural colors
        
        # Check if image has many distinct colors (dataset images)
        unique_colors = len(np.unique(img.reshape(-1, 3), axis=0))
        
        # Check aspect ratio
        aspect_ratio = width / height if height > 0 else 0
        
        # Default to other
        result_type = "other"
        description = "无法确定图片类型"
        components = []
        code_info = ""
        
        # Since OCR is empty, rely on visual characteristics
        # Check for typical architecture diagram features
        # Architecture diagrams often have:
        # - High contrast
        # - Multiple distinct regions
        # - Often grayscale or limited color palette
        
        # Check for experiment result features
        # Often have:
        # - Lines/curves
        # - Specific color schemes (blue, red, green lines)
        # - Axis-like structures
        
        # Check for dataset features
        # Often have:
        # - Natural images
        # - Multiple samples
        # - Rich colors
        
        # Simple heuristic: check color variance and saturation
        if saturation < 50 and color_std < 50:
            # Likely architecture diagram (grayscale, high contrast)
            result_type = "architecture"
            description = "展示模型网络结构和数据流关系"
            components = []
            code_info = "需要实现模型网络结构"
        elif unique_colors > 10000 and saturation > 100:
            # Likely dataset images (natural, colorful)
            result_type = "dataset"
            description = "展示数据集样本或输入示例"
            components = []
            code_info = "需要实现数据加载和预处理模块"
        elif aspect_ratio > 1.5 or aspect_ratio < 0.67:
            # Wide or tall images often are experiment results
            result_type = "experiment_result"
            description = "展示实验结果或性能对比"
            components = []
            code_info = "需要实现实验评估和结果可视化"
        else:
            # Default based on image complexity
            if unique_colors < 5000:
                result_type = "architecture"
                description = "展示模型网络结构和数据流关系"
                components = []
                code_info = "需要实现模型网络结构"
            else:
                result_type = "other"
                description = "无法确定图片类型"
                components = []
                code_info = ""
        
        return {
            "type": result_type,
            "description": description,
            "important_components": components,
            "code_relevant_information": code_info
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
