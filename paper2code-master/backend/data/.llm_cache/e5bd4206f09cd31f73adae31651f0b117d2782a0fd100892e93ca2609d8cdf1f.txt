import sys
import json
import numpy as np
import cv2
from PIL import Image
import os

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text provided externally
        ocr_text = """1.0
0.9
0.6
0.5
Train-Test Set Division
0.4
20-80
40-60
50-50
60-40
80-20
Epochs"""
        
        ocr_lower = ocr_text.lower()
        
        # Check for experiment_result indicators
        experiment_indicators = [
            "accuracy", "loss", "epoch", "precision", "recall", "f1",
            "train-test", "train test", "training", "validation",
            "performance", "curve", "comparison"
        ]
        
        # Check for architecture indicators
        architecture_indicators = [
            "cnn", "transformer", "attention", "encoder", "decoder",
            "block", "embedding", "classifier", "feature extractor",
            "alexnet", "resnet", "vgg", "googlenet"
        ]
        
        # Check for dataset indicators
        dataset_indicators = [
            "sample", "example", "input", "dataset", "data sample",
            "class", "category"
        ]
        
        # Determine type based on priority
        is_experiment = any(ind in ocr_lower for ind in experiment_indicators)
        is_architecture = any(ind in ocr_lower for ind in architecture_indicators)
        is_dataset = any(ind in ocr_lower for ind in dataset_indicators)
        
        # Priority: experiment_result > architecture > dataset > other
        if is_experiment:
            result_type = "experiment_result"
            description = "展示模型训练过程中的性能指标变化曲线"
            
            # Extract important components
            components = []
            if "epoch" in ocr_lower:
                components.append("Epoch")
            if "accuracy" in ocr_lower:
                components.append("Accuracy")
            if "loss" in ocr_lower:
                components.append("Loss")
            if "precision" in ocr_lower:
                components.append("Precision")
            if "recall" in ocr_lower:
                components.append("Recall")
            if "f1" in ocr_lower:
                components.append("F1-score")
            
            # Add model names if present
            for model in ["alexnet", "resnet", "vgg", "googlenet", "cnn", "transformer"]:
                if model in ocr_lower:
                    components.append(model.upper())
            
            code_info = "需要实现模型训练和评估流程，记录并可视化训练指标"
            
        elif is_architecture:
            result_type = "architecture"
            description = "展示模型网络结构和数据流关系"
            
            # Extract architecture components
            components = []
            arch_terms = {
                "cnn": "CNN",
                "transformer": "Transformer",
                "attention": "Attention Module",
                "encoder": "Encoder",
                "decoder": "Decoder",
                "block": "Block",
                "embedding": "Embedding",
                "classifier": "Classifier",
                "feature extractor": "Feature Extractor",
                "alexnet": "AlexNet",
                "resnet": "ResNet",
                "vgg": "VGG",
                "googlenet": "GoogLeNet"
            }
            
            for term, component in arch_terms.items():
                if term in ocr_lower:
                    components.append(component)
            
            code_info = "需要实现模型网络结构定义和特征提取模块"
            
        elif is_dataset:
            result_type = "dataset"
            description = "展示数据集样本或输入示例"
            components = []
            code_info = "需要实现数据加载和预处理流程"
            
        else:
            result_type = "other"
            description = "无法确定图片类型"
            components = []
            code_info = ""
        
        # Clean up components (remove duplicates, limit to 5)
        components = list(dict.fromkeys(components))[:5]
        
        return {
            "type": result_type,
            "description": description,
            "important_components": components,
            "code_relevant_information": code_info
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
    else:
        image_path = r"D:\paper2code-master(1)\paper2code-master\backend\data\output\论文4\step1\images\image_10.png"
    
    result = analyze_image(image_path)
    print(json.dumps(result, ensure_ascii=False))
