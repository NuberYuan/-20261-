```python
import os
import random
import yaml
import torch
from torch.utils.data import Dataset, Subset, random_split
from PIL import Image
from torchvision import transforms


class FourMedicalImagingDatasetsCellNucleiColonPolypLiverLungNoduleDataset(Dataset):
    """
    Dataset class for four medical imaging datasets:
    cell nuclei, colon polyp, liver, lung nodule.

    Supports three preprocessing versions: 'color', 'grayscale', 'segmented'.
    The version is selected via the `version` parameter (default: 'color').
    """

    def __init__(self, root_dir, version='color', config=None, transform=None):
        """
        Args:
            root_dir (str): Root directory containing the dataset folders.
            version (str): Preprocessing version. One of 'color', 'grayscale', 'segmented'.
            config (dict, optional): Loaded config dictionary. If None, will try to load from default path.
            transform (callable, optional): Optional transform to be applied on a sample.
        """
        self.root_dir = root_dir
        self.version = version
        self.config = config
        self.transform = transform

        if self.config is None:
            # Try to load config from default location
            config_path = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), 'config', 'config.yaml')
            if os.path.exists(config_path):
                with open(config_path, 'r', encoding='utf-8') as f:
                    self.config = yaml.safe_load(f)
            else:
                self.config = {}

        # Check if the requested version is unresolved
        version_status = self.config.get('data', {}).get(self.version, {}).get('status', 'resolved')
        if version_status == 'unresolved':
            reason = self.config.get('data', {}).get(self.version, {}).get('reason', 'Unknown reason')
            raise NotImplementedError(
                f"Dataset version '{self.version}' is unresolved: {reason}"
            )

        # Build image paths and labels
        self.image_paths = []
        self.labels = []

        # Define dataset folders and their corresponding labels
        # Expected folder structure: root_dir/<dataset_name>/images/*.png (or .jpg, .tif)
        # For binary segmentation tasks, we treat this as image classification:
        #   - cell_nuclei: 670 images (binary: nucleus vs background)
        #   - colon_polyp: 7379 images (binary: polyp vs non-polyp)
        #   - liver: 331 images (binary: liver vs non-liver)
        #   - lung_nodule: 1012 images (binary: nodule vs non-nodule)
        #
        # Since the original task is segmentation, we derive classification labels
        # from the presence of the target object in the mask. For simplicity,
        # we assume each dataset folder contains 'images' and 'masks' subfolders.
        # If masks are not available, we assign labels based on dataset name
        # (e.g., all samples from a dataset get label 1 if the dataset represents
        # the positive class, 0 otherwise). This is a simplification for the
        # classification task.

        dataset_configs = [
            {'name': 'cell_nuclei', 'expected_count': 670, 'label': 1},
            {'name': 'colon_polyp', 'expected_count': 7379, 'label': 1},
            {'name': 'liver', 'expected_count': 331, 'label': 1},
            {'name': 'lung_nodule', 'expected_count': 1012, 'label': 1},
        ]

        for ds_config in dataset_configs:
            ds_name = ds_config['name']
            ds_path = os.path.join(self.root_dir, ds_name)
            if not os.path.exists(ds_path):
                print(f"Warning: Dataset folder '{ds_name}' not found at {ds_path}. Skipping.")
                continue

            # Look for images in common subdirectories
            images_dir = None
            for sub in ['images', 'image', 'img', 'train', 'test', '']:
                candidate = os.path.join(ds_path, sub) if sub else ds_path
                if os.path.isdir(candidate):
                    # Check if it contains image files
                    for fname in os.listdir(candidate):
                        if fname.lower().endswith(('.png', '.jpg', '.jpeg', '.tif', '.tiff', '.bmp')):
                            images_dir = candidate
                            break
                if images_dir:
                    break

            if images_dir is None:
                print(f"Warning: No image directory found for dataset '{ds_name}'. Skipping.")
                continue

            # Collect all image paths
            for fname in sorted(os.listdir(images_dir)):
                if fname.lower().endswith(('.png', '.jpg', '.jpeg', '.tif', '.tiff', '.bmp')):
                    img_path = os.path.join(images_dir, fname)
                    self.image_paths.append(img_path)
                    self.labels.append(ds_config['label'])

        if len(self.image_paths) == 0:
            # Create a synthetic dataset for smoke testing when no real data is available
            print(f"Warning: No images found under {self.root_dir}. Creating synthetic dataset for smoke testing.")
            self._create_synthetic_dataset()

        print(f"Loaded {len(self.image_paths)} images for version '{self.version}'.")

    def _create_synthetic_dataset(self):
        """Create a small synthetic dataset to allow smoke testing."""
        import numpy as np
        os.makedirs(self.root_dir, exist_ok=True)
        for ds_name in ['cell_nuclei', 'colon_polyp', 'liver', 'lung_nodule']:
            ds_path = os.path.join(self.root_dir, ds_name, 'images')
            os.makedirs(ds_path, exist_ok=True)
            for i in range(5):
                img = np.random.randint(0, 255, (64, 64, 3), dtype=np.uint8)
                img_path = os.path.join(ds_path, f'synthetic_{i}.png')
                Image.fromarray(img).save(img_path)
                self.image_paths.append(img_path)
                self.labels.append(1)

    def __len__(self):
        return len(self.image_paths)

    def __getitem__(self, idx):
        """
        Returns:
            (PIL.Image, int): Original PIL image and label.
            No transform is applied here; use TransformSubset for augmentation.
        """
        img_path = self.image_paths[idx]
        label = self.labels[idx]

        # Load image with PIL
        image = Image.open(img_path).convert('RGB')

        return image, label

    def preprocess_image(self, image, version=None):
        """
        Apply preprocessing pipeline based on the selected version.

        Args:
            image (PIL.Image): Input image.
            version (str, optional): Override version. Defaults to self.version.

        Returns:
            torch.Tensor: Preprocessed image tensor.
        """
        if version is None:
            version = self.version

        if version == 'color':
            transform = transforms.Compose([
                transforms.ToTensor(),
                transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
            ])
        elif version == 'grayscale':
            transform = transforms.Compose([
                transforms.Grayscale(num_output_channels=1),
                transforms.ToTensor(),
                transforms.Lambda(lambda x: x.repeat(3, 1, 1)),
                transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
            ])
        elif version == 'segmented':
            # Segmentation preprocessing is NOT_FOUND in the paper.
            # Fallback to color preprocessing with a warning.
            print("Warning: 'segmented' version is not fully implemented. Falling back to 'color' preprocessing.")
            transform = transforms.Compose([
                transforms.ToTensor(),
                transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
            ])
        else:
            raise ValueError(f"Unknown version: {version}. Must be one of 'color', 'grayscale', 'segmented'.")

        return transform(image)

    def split_dataset(self, train_ratio=0.8, val_ratio=0.1, seed=None):
        """
        Split the dataset into train, validation, and test subsets.

        Args:
            train_ratio (float): Proportion of training data.
            val_ratio (float): Proportion of validation data.
            seed (int, optional): Random seed for reproducibility.

        Returns:
            tuple: (train_subset, val_subset, test_subset) as Subset objects.
        """
        if seed is not None:
            random.seed(seed)
            torch.manual_seed(seed)

        dataset_size = len(self)
        train_size = int(train_ratio * dataset_size)
        val_size = int(val_ratio * dataset_size)
        test_size = dataset_size - train_size - val_size

        if test_size < 0:
            raise ValueError("train_ratio + val_ratio must be <= 1.0")

        # Use random_split for reproducibility
        train_subset, val_subset, test_subset = random_split(
            self,
            [train_size, val_size, test_size],
            generator=torch.Generator().manual_seed(seed) if seed is not None else None
        )

        return train_subset, val_subset, test_subset


class TransformSubset(Dataset):
    """
    A lightweight wrapper around a Subset that applies a transform to each sample.
    This avoids mutating the shared underlying Dataset object.
    """

    def __init__(self, subset, transform=None):
        """
        Args:
            subset (Subset): The subset to wrap.
            transform (callable, optional): Transform to apply to each sample.
        """
        self.subset = subset
        self.transform = transform

    def __len__(self):
        return len(self.subset)

    def __getitem__(self, idx):
        x, y = self.subset[idx]
        if self.transform is not None:
            x = self.transform(x)
        return x, y


def get_transform(train=True, version='color'):
    """
    Returns the appropriate transform for training or evaluation.

    Args:
        train (bool): If True, include augmentation (RandomHorizontalFlip).
        version (str): Preprocessing version.

    Returns:
        callable: Transform function.
    """
    if version == 'color':
        base_transforms = [
            transforms.ToTensor(),
            transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
        ]
    elif version == 'grayscale':
        base_transforms = [
            transforms.Grayscale(num_output_channels=1),
            transforms.ToTensor(),
            transforms.Lambda(lambda x: x.repeat(3, 1, 1)),
            transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
        ]
    elif version == 'segmented':
        # Fallback to color for now
        base_transforms = [
            transforms.ToTensor(),
            transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
        ]
    else:
        raise ValueError(f"Unknown version: {version}")

    if train:
        # Add augmentation for training only
        return transforms.Compose([
            transforms.RandomHorizontalFlip(p=0.5),
            *base_transforms,
        ])
    else:
        # No augmentation for validation/test
        return transforms.Compose(base_transforms)


def get_dataloaders(config=None, version='color', batch_size=32, num_workers=4, seed=42):
    """
    Create DataLoaders for train, validation, and test sets.

    Args:
        config (dict, optional): Loaded config dictionary.
        version (str): Preprocessing version. Must be one of 'color', 'grayscale', 'segmented'.
        batch_size (int): Batch size for DataLoaders.
        num_workers (int): Number of worker processes.
        seed (int): Random seed.

    Returns:
        dict: Dictionary with 'train', 'val', 'test' DataLoaders.
    """
    if config is None:
        # Try to load config from default location
        config_path = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), 'config', 'config.yaml')
        if os.path.exists(config_path):
            with open(config_path, 'r', encoding='utf-8') as f:
                config = yaml.safe_load(f)
        else:
            config = {}

    # Get data root from config
    data_root = config.get('data', {}).get('root', './data')

    # Create dataset
    dataset = FourMedicalImagingDatasetsCellNucleiColonPolypLiverLungNoduleDataset(
        root_dir=data_root,
        version=version,
        config=config
    )

    # Split dataset
    train_ratio = config.get('split_strategy', {}).get('default_train_ratio', 0.8)
    val_ratio = config.get('split_strategy', {}).get('val_split_ratio', 0.1)
    split_seed = config.get('split_strategy', {}).get('seed', seed)

    train_subset, val_subset, test_subset = dataset.split_dataset(
        train_ratio=train_ratio,
        val_ratio=val_ratio,
        seed=split_seed
    )

    # Wrap subsets with transforms
    train_transform = get_transform(train=True, version=version)
    eval_transform = get_transform(train=False, version=version)

    train_dataset = TransformSubset(train_subset, transform=train_transform)
    val_dataset = TransformSubset(val_subset, transform=eval_transform)
    test_dataset = TransformSubset(test_subset, transform=eval_transform)

    # Create DataLoaders
    train_loader = torch.utils.data.DataLoader(
        train_dataset,
        batch_size=batch_size,
        shuffle=True,
        num_workers=num_workers,
        pin_memory=True
    )
    val_loader = torch.utils.data.DataLoader(
        val_dataset,
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        pin_memory=True
    )
    test_loader = torch.utils.data.DataLoader(
        test_dataset,
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        pin_memory=True
    )

    return {
        'train': train_loader,
        'val': val_loader,
        'test': test_loader,
    }
```
