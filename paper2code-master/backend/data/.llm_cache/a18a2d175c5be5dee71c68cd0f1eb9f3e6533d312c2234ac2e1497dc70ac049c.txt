import sys
import json
import numpy as np
import cv2
import os

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片文件",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is provided externally (empty in this case)
        ocr_text = ""
        
        # Since no OCR text, use visual features and filename context
        # Get image basic properties
        height, width = img.shape[:2]
        
        # Check if image is mostly grayscale (common for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        color_std = np.std(img)
        
        # Check color distribution - architecture diagrams often have limited colors
        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        saturation_mean = np.mean(hsv[:,:,1])
        
        # Analyze image content based on visual characteristics
        # Architecture diagrams typically have:
        # - High contrast between elements
        # - Multiple distinct regions
        # - Often grayscale or limited color palette
        
        # Check if image has characteristics of architecture diagram
        # (block-like structures, flow patterns)
        edges = cv2.Canny(gray, 50, 150)
        edge_density = np.sum(edges > 0) / (height * width)
        
        # Check for text-like regions (common in all paper figures)
        # Use simple thresholding to detect text areas
        
        # Since no OCR text is available, rely on visual characteristics
        # and filename context
        
        # Default to other if cannot determine
        result = {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
        
        # Visual heuristics for architecture diagrams:
        # - High edge density (many boxes/lines)
        # - Low color variance (often grayscale)
        # - Multiple distinct regions
        
        if edge_density > 0.15 and color_std < 80:
            result = {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": [],
                "code_relevant_information": "需要实现模型网络结构，包含特征提取和分类模块"
            }
        elif edge_density > 0.08 and color_std < 100:
            # Could be experiment result with plots
            result = {
                "type": "experiment_result",
                "description": "展示实验结果或性能对比",
                "important_components": [],
                "code_relevant_information": "需要实现实验评估和结果可视化"
            }
        elif color_std > 100 and saturation_mean > 50:
            # Colorful image, likely dataset samples
            result = {
                "type": "dataset",
                "description": "展示数据集样本或输入示例",
                "important_components": [],
                "code_relevant_information": "需要准备和处理数据集"
            }
        
        return result
        
    except Exception as e:
        return {
            "type": "other",
            "description": "图片分析过程中出现错误",
            "important_components": [],
            "code_relevant_information": ""
        }

def main():
    if len(sys.argv) < 2:
        print(json.dumps({
            "type": "other",
            "description": "未提供图片路径",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
        return
    
    image_path = sys.argv[1]
    
    # Check if file exists
    if not os.path.exists(image_path):
        print(json.dumps({
            "type": "other",
            "description": "图片文件不存在",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
        return
    
    result = analyze_image(image_path)
    
    # Output only JSON
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
