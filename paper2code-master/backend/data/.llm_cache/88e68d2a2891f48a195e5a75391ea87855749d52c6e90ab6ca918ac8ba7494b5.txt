import sys
import json
import numpy as np
import cv2

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty in this case
        ocr_text = ""
        
        # Since OCR is empty, use visual features and heuristics
        # Get basic image properties
        height, width = img.shape[:2]
        
        # Check if image is mostly white/light (typical for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        mean_brightness = np.mean(gray)
        
        # Check color variance (experiment plots often have colored lines)
        color_variance = np.std(img.reshape(-1, 3), axis=0).mean()
        
        # Check aspect ratio (architecture diagrams often wider)
        aspect_ratio = width / height if height > 0 else 0
        
        # Heuristic classification based on visual properties
        # Since OCR is empty, we need to rely on visual cues
        
        # Architecture diagrams typically:
        # - High brightness (white background)
        # - Low color variance (mostly black/white with some colors)
        # - Often wider than tall
        
        # Experiment results typically:
        # - Have colored lines/points
        # - Higher color variance
        # - Often have grid-like patterns
        
        # Dataset images typically:
        # - Show natural images or data samples
        # - Higher color variance
        # - Various aspect ratios
        
        if mean_brightness > 200 and color_variance < 50:
            # Likely architecture diagram
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": [],
                "code_relevant_information": "需要实现模型网络结构"
            }
        elif color_variance > 80 and aspect_ratio > 1.2:
            # Likely experiment results with colored plots
            return {
                "type": "experiment_result",
                "description": "展示实验结果对比或性能变化",
                "important_components": [],
                "code_relevant_information": "需要实现实验评估和结果可视化"
            }
        elif color_variance > 60:
            # Could be dataset samples
            return {
                "type": "dataset",
                "description": "展示数据集样本或输入示例",
                "important_components": [],
                "code_relevant_information": "需要实现数据加载和预处理"
            }
        else:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
            
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
