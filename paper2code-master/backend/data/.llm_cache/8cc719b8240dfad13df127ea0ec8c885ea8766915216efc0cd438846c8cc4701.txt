import sys
import json
import numpy as np
import cv2

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty, so rely on visual features
        # Since we cannot use complex CV methods, use basic image properties
        
        height, width = img.shape[:2]
        
        # Check if image is mostly grayscale (common in architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        color_std = np.std(gray)
        
        # Check color variance - architecture diagrams often have low color variance
        # Experiment results often have colored lines/curves
        # Dataset images often have high color variance
        
        # Simple heuristic based on color statistics
        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        saturation_mean = np.mean(hsv[:,:,1])
        
        # Architecture diagrams: often low saturation, structured
        # Experiment results: medium saturation, have curves
        # Dataset: high saturation, natural images
        
        if saturation_mean < 30:
            # Likely architecture diagram (mostly black/white/gray)
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": [],
                "code_relevant_information": "需要实现模型网络结构"
            }
        elif saturation_mean < 80:
            # Could be experiment results with colored curves
            return {
                "type": "experiment_result",
                "description": "展示实验结果对比或性能变化",
                "important_components": [],
                "code_relevant_information": "需要实现实验评估和结果可视化"
            }
        else:
            # High saturation - likely dataset samples
            return {
                "type": "dataset",
                "description": "展示数据集样本或输入示例",
                "important_components": [],
                "code_relevant_information": "需要实现数据加载和预处理"
            }
            
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

def main():
    if len(sys.argv) < 2:
        print(json.dumps({
            "type": "other",
            "description": "未提供图片路径",
            "important_components": [],
            "code_relevant_information": ""
        }))
        return
    
    image_path = sys.argv[1]
    result = analyze_image(image_path)
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
