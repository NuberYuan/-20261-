import os
import sys
import json
import numpy as np
import cv2


def fallback_result():
    return {
        "type": "other",
        "description": "无法确定图片类型",
        "important_components": [],
        "code_relevant_information": ""
    }


def make_dataset_result():
    return {
        "type": "dataset",
        "description": "展示数据样本或类别示例",
        "important_components": [],
        "code_relevant_information": "需要实现数据预处理与加载代码"
    }


def make_experiment_result(ocr_text):
    ocr = ocr_text.lower()
    components = []
    seen = set()

    terms = [
        ("alexnet", "AlexNet"),
        ("resnet", "ResNet"),
        ("vgg", "VGG"),
        ("googlenet", "GoogLeNet"),
        ("transformer", "Transformer"),
        ("cnn", "CNN"),
        ("accuracy", "Accuracy"),
        ("loss", "Loss"),
        ("epoch", "Epoch"),
        ("precision", "Precision"),
        ("recall", "Recall"),
        ("f1-score", "F1-score"),
        ("f1", "F1-score"),
        ("map", "mAP"),
        ("准确率", "Accuracy"),
        ("损失", "Loss"),
    ]

    for low, canonical in terms:
        if low in ocr and canonical.lower() not in seen:
            components.append(canonical)
            seen.add(canonical.lower())

    return {
        "type": "experiment_result",
        "description": "展示实验性能对比曲线或柱状图，包含准确率、Loss等指标变化",
        "important_components": components,
        "code_relevant_information": "需要实现模型训练与评估代码，输出准确率、损失等指标"
    }


def make_architecture_result(ocr_text):
    ocr = ocr_text.lower()
    components = []
    seen = set()

    terms = [
        ("alexnet", "AlexNet"),
        ("resnet", "ResNet"),
        ("vgg", "VGG"),
        ("googlenet", "GoogLeNet"),
        ("transformer encoder", "Transformer Encoder"),
        ("transformer decoder", "Transformer Decoder"),
        ("attention module", "Attention Module"),
        ("feature extractor", "Feature Extractor"),
        ("embedding", "Embedding"),
        ("classifier", "Classifier"),
        ("encoder", "Encoder"),
        ("decoder", "Decoder"),
        ("attention", "Attention"),
        ("cnn", "CNN"),
        ("transformer", "Transformer"),
        ("block", "Block"),
    ]

    for low, canonical in terms:
        if low in ocr and canonical.lower() not in seen:
            components.append(canonical)
            seen.add(canonical.lower())

    if "transformer" in ocr:
        code_info = "需要实现Transformer编码器、注意力模块和前馈网络"
    elif "cnn" in ocr or "feature extractor" in ocr or "alexnet" in ocr or "resnet" in ocr or "vgg" in ocr or "googlenet" in ocr:
        code_info = "需要实现CNN特征提取模块和分类头"
    else:
        code_info = "需要实现模型结构相关模块"

    return {
        "type": "architecture",
        "description": "展示模型网络结构和数据流关系",
        "important_components": components,
        "code_relevant_information": code_info
    }


def classify(ocr_text, mean_sat, std_sat, unique_colors, white_ratio):
    ocr = ocr_text.lower()

    experiment_keywords = [
        "accuracy", "loss", "epoch", "precision", "recall", "f1",
        "f1-score", "实验", "experiment", "曲线", "准确率", "损失",
        "性能对比", "柱状图", "柱状"
    ]
    architecture_keywords = [
        "cnn", "transformer", "attention", "encoder", "decoder",
        "block", "embedding", "classifier", "feature extractor",
        "alexnet", "resnet", "vgg", "googlenet",
        "网络结构", "模型结构", "结构图", "特征提取", "模块"
    ]
    dataset_keywords = [
        "dataset", "data set", "sample", "example", "input image",
        "输入图像", "样本", "示例", "类别", "数据样本", "训练样本", "测试样本"
    ]

    if any(keyword in ocr for keyword in experiment_keywords):
        return make_experiment_result(ocr_text)

    if any(keyword in ocr for keyword in architecture_keywords):
        return make_architecture_result(ocr_text)

    if any(keyword in ocr for keyword in dataset_keywords):
        return make_dataset_result()

    if mean_sat > 50 and std_sat > 40 and unique_colors > 1500:
        return make_dataset_result()

    if white_ratio > 0.5 and unique_colors < 2500:
        return make_architecture_result(ocr_text)

    return fallback_result()


def main():
    try:
        if len(sys.argv) < 2:
            print(json.dumps(fallback_result(), ensure_ascii=False))
            return

        image_path = sys.argv[1]

        ocr_text = ""
        if len(sys.argv) > 2:
            ocr_text = sys.argv[2].strip()
        if not ocr_text:
            ocr_text = os.environ.get("OCR_TEXT", "").strip()

        if ocr_text in ["（无）", "无", "(无)", "None", "none", ""]:
            ocr_text = ""

        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)

        if img is None:
            print(json.dumps(fallback_result(), ensure_ascii=False))
            return

        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        saturation = hsv[:, :, 1]
        mean_sat = float(np.mean(saturation))
        std_sat = float(np.std(saturation))

        small = cv2.resize(img, (64, 64), interpolation=cv2.INTER_AREA)
        unique_colors = int(len(np.unique(small.reshape(-1, 3), axis=0)))

        lower_white = np.array([200, 200, 200], dtype=np.uint8)
        upper_white = np.array([255, 255, 255], dtype=np.uint8)
        white_mask = cv2.inRange(img, lower_white, upper_white)
        white_ratio = float(np.count_nonzero(white_mask)) / (img.shape[0] * img.shape[1])

        result = classify(ocr_text, mean_sat, std_sat, unique_colors, white_ratio)
        print(json.dumps(result, ensure_ascii=False))

    except Exception:
        print(json.dumps(fallback_result(), ensure_ascii=False))


if __name__ == "__main__":
    main()
