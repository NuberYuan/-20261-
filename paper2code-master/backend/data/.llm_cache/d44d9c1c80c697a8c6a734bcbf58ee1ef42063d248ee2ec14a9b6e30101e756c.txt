import sys
import json
import numpy as np
import cv2
from PIL import Image
import os

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty in this case
        ocr_text = ""
        
        # Since OCR is empty, use visual information
        # Get basic image properties
        height, width = img.shape[:2]
        
        # Check if image is mostly white/light (typical for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        mean_brightness = np.mean(gray)
        
        # Check color variance (architecture diagrams often have low color variance)
        color_variance = np.std(img)
        
        # Check if image has text-like features (high contrast areas)
        _, binary = cv2.threshold(gray, 200, 255, cv2.THRESH_BINARY)
        white_ratio = np.sum(binary == 255) / (height * width)
        
        # Check aspect ratio (architecture diagrams often wider than tall)
        aspect_ratio = width / height if height > 0 else 0
        
        # Since OCR is empty and we can't use complex CV,
        # use simple heuristics based on image properties
        
        # Architecture diagrams typically:
        # - Have high white ratio (clean background)
        # - Have moderate aspect ratio
        # - Have low color variance (mostly black and white)
        
        if white_ratio > 0.7 and color_variance < 80:
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": [],
                "code_relevant_information": "需要实现模型网络结构"
            }
        elif white_ratio > 0.5 and aspect_ratio > 1.2:
            # Could be experiment result or dataset
            return {
                "type": "experiment_result",
                "description": "展示实验结果数据",
                "important_components": [],
                "code_relevant_information": "需要实现实验评估和结果可视化"
            }
        else:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
            
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

def main():
    if len(sys.argv) < 2:
        result = {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
    else:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
    
    # Output only JSON
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
