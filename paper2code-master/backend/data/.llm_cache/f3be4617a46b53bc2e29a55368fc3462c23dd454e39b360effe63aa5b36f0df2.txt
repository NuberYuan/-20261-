import sys
import json
import numpy as np
import cv2
import os

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty as provided
        ocr_text = ""
        
        # Since OCR is empty, use visual information
        # Get basic image properties
        height, width = img.shape[:2]
        
        # Check if image is mostly white/light background (typical for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        mean_brightness = np.mean(gray)
        
        # Check color variance (architecture diagrams often have low color variance)
        color_variance = np.std(img)
        
        # Check if image has text-like features (high contrast areas)
        _, binary = cv2.threshold(gray, 200, 255, cv2.THRESH_BINARY)
        white_ratio = np.sum(binary == 255) / (height * width)
        
        # Check aspect ratio (architecture diagrams often wider than tall)
        aspect_ratio = width / height if height > 0 else 1
        
        # Since OCR is empty, we need to rely on visual cues
        # Architecture diagrams typically have:
        # - Light background
        # - Boxes/shapes connected by lines
        # - Moderate complexity
        
        # Experiment results typically have:
        # - Lines/curves
        # - Axis labels
        # - Multiple colors
        
        # Dataset images typically have:
        # - Grid of images
        # - High color variance
        # - Natural images
        
        # Default to other if cannot determine
        result = {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
        
        # Visual heuristics (simple, not relying on contours/edges)
        if mean_brightness > 200 and white_ratio > 0.7:
            # Likely a diagram or chart on white background
            if color_variance < 80:
                # Low color variance suggests architecture diagram
                result = {
                    "type": "architecture",
                    "description": "展示模型网络结构和数据流关系",
                    "important_components": [],
                    "code_relevant_information": "需要实现模型网络结构"
                }
            else:
                # Higher color variance might be experiment results
                result = {
                    "type": "experiment_result",
                    "description": "展示实验结果对比或性能指标",
                    "important_components": [],
                    "code_relevant_information": "需要实现实验评估和结果可视化"
                }
        elif color_variance > 100:
            # High color variance suggests dataset images
            result = {
                "type": "dataset",
                "description": "展示数据样本或输入示例",
                "important_components": [],
                "code_relevant_information": "需要实现数据加载和预处理"
            }
        
        return result
        
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

def main():
    if len(sys.argv) < 2:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }))
        return
    
    image_path = sys.argv[1]
    result = analyze_image(image_path)
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
