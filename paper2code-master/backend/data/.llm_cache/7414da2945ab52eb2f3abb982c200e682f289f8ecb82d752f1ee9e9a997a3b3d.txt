import sys
import json
import os
import numpy as np
import cv2

def analyze_image(image_path):
    # Default result
    result = {
        "type": "other",
        "description": "无法确定图片类型",
        "important_components": [],
        "code_relevant_information": ""
    }
    
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return result
        
        # OCR text provided externally (simulated here)
        ocr_text = """Feature: McCabe_Cyclomatic_Complexity
Feature; Lines_of_Code
0.06
0.25
0.20
0.04

1.4
McCabe_Cyclomatic_Complexity
Lines of Code
Feature: Number_of Methods
Feature: Number_of_Comments
0.03
0.01
~0.05
~0.10"""
        
        # Convert OCR text to lowercase for matching
        ocr_lower = ocr_text.lower()
        
        # Check for experiment_result indicators
        experiment_keywords = [
            'accuracy', 'loss', 'epoch', 'precision', 'recall', 'f1',
            'curve', 'performance', 'comparison', 'metric', 'feature importance',
            'correlation', 'coefficient'
        ]
        
        # Check for architecture indicators
        architecture_keywords = [
            'cnn', 'transformer', 'attention', 'encoder', 'decoder',
            'block', 'embedding', 'classifier', 'feature extractor',
            'alexnet', 'resnet', 'vgg', 'googlenet'
        ]
        
        # Check for dataset indicators
        dataset_keywords = [
            'sample', 'example', 'dataset', 'class', 'category',
            'input image', 'training data'
        ]
        
        # Determine type based on priority
        is_experiment = any(kw in ocr_lower for kw in experiment_keywords)
        is_architecture = any(kw in ocr_lower for kw in architecture_keywords)
        is_dataset = any(kw in ocr_lower for kw in dataset_keywords)
        
        # Priority: experiment_result > architecture > dataset > other
        if is_experiment:
            result["type"] = "experiment_result"
            result["description"] = "展示实验指标或性能对比结果"
            
            # Extract important components
            components = []
            if 'mccabe_cyclomatic_complexity' in ocr_lower:
                components.append("McCabe Cyclomatic Complexity")
            if 'lines_of_code' in ocr_lower:
                components.append("Lines of Code")
            if 'number_of_methods' in ocr_lower:
                components.append("Number of Methods")
            if 'number_of_comments' in ocr_lower:
                components.append("Number of Comments")
            
            result["important_components"] = components
            result["code_relevant_information"] = "需要实现代码复杂度指标计算和相关性分析功能"
            
        elif is_architecture:
            result["type"] = "architecture"
            result["description"] = "展示模型网络结构和数据流关系"
            
            # Extract model components
            components = []
            architecture_components = {
                'cnn': 'CNN',
                'transformer': 'Transformer',
                'attention': 'Attention Module',
                'encoder': 'Encoder',
                'decoder': 'Decoder',
                'block': 'Block',
                'embedding': 'Embedding',
                'classifier': 'Classifier',
                'feature extractor': 'Feature Extractor',
                'alexnet': 'AlexNet',
                'resnet': 'ResNet',
                'vgg': 'VGG',
                'googlenet': 'GoogLeNet'
            }
            
            for key, value in architecture_components.items():
                if key in ocr_lower:
                    components.append(value)
            
            result["important_components"] = components
            result["code_relevant_information"] = "需要实现模型网络结构定义和特征提取模块"
            
        elif is_dataset:
            result["type"] = "dataset"
            result["description"] = "展示数据集样本或类别示例"
            result["important_components"] = []
            result["code_relevant_information"] = "需要实现数据加载和预处理功能"
        
        # If no specific type detected, check visual features
        if result["type"] == "other":
            # Get basic image info
            height, width = img.shape[:2]
            
            # Check if image looks like a plot/chart (experiment result)
            # Simple heuristic: check if image has grid-like patterns or axis labels
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
            
            # Check for text-like content (high contrast areas)
            _, binary = cv2.threshold(gray, 200, 255, cv2.THRESH_BINARY)
            text_ratio = np.sum(binary == 255) / (height * width)
            
            # Check for colored elements (common in plots)
            hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
            colored_ratio = np.sum(hsv[:,:,1] > 50) / (height * width)
            
            if text_ratio > 0.05 and colored_ratio > 0.1:
                result["type"] = "experiment_result"
                result["description"] = "包含图表元素，可能为实验结果展示"
                result["code_relevant_information"] = "需要实现数据可视化功能"
        
    except Exception as e:
        # Keep default result on error
        pass
    
    return result

def main():
    if len(sys.argv) < 2:
        print(json.dumps({
            "type": "other",
            "description": "未提供图片路径",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
        return
    
    image_path = sys.argv[1]
    
    # Check if file exists
    if not os.path.exists(image_path):
        print(json.dumps({
            "type": "other",
            "description": "图片文件不存在",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
        return
    
    result = analyze_image(image_path)
    
    # Output only JSON
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
