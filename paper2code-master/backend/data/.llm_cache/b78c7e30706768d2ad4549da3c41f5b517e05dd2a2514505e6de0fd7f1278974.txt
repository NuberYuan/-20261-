import sys
import json
import numpy as np
import cv2
from PIL import Image

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is empty, so rely on visual features
        # Since OCR is empty, we need to use basic visual information
        
        # Get image dimensions
        height, width = img.shape[:2]
        
        # Convert to grayscale for basic analysis
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        
        # Basic statistics
        mean_intensity = np.mean(gray)
        std_intensity = np.std(gray)
        
        # Check if image is mostly white/light (typical for architecture diagrams)
        white_ratio = np.sum(gray > 200) / (height * width)
        
        # Check if image has colorful elements (typical for experiment results)
        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
        saturation = hsv[:, :, 1]
        high_sat_ratio = np.sum(saturation > 100) / (height * width)
        
        # Check color variance
        color_variance = np.var(img.reshape(-1, 3), axis=0).mean()
        
        # Decision logic based on visual features
        # Since OCR is empty, we rely on visual characteristics
        
        # Architecture diagrams typically:
        # - High white ratio (clean background)
        # - Low color variance (mostly black/white/gray)
        # - Moderate complexity
        
        # Experiment results typically:
        # - Have colored lines/points
        # - Higher color variance
        # - May have grid patterns
        
        # Dataset images typically:
        # - High color variance
        # - Natural image characteristics
        # - Lower white ratio
        
        if white_ratio > 0.7 and color_variance < 3000:
            # Likely architecture diagram
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": [],
                "code_relevant_information": "需要实现模型网络结构，包含特征提取和分类模块"
            }
        elif high_sat_ratio > 0.2 and color_variance > 2000:
            # Likely experiment results with colored curves
            return {
                "type": "experiment_result",
                "description": "展示实验结果对比或性能指标变化",
                "important_components": [],
                "code_relevant_information": "需要实现实验评估模块，包含指标计算和结果可视化"
            }
        elif color_variance > 5000 and white_ratio < 0.5:
            # Likely dataset samples
            return {
                "type": "dataset",
                "description": "展示数据集样本或输入示例",
                "important_components": [],
                "code_relevant_information": "需要实现数据加载和预处理模块"
            }
        else:
            # Default to other if cannot determine
            return {
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }
            
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
