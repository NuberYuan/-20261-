```python
import sys
import json
import numpy as np
import cv2

def analyze_image(image_path, ocr_text):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # Get basic image info
        height, width = img.shape[:2]
        
        # Normalize OCR text for matching
        ocr_lower = ocr_text.lower()
        
        # Define keyword sets
        architecture_keywords = [
            "cnn", "transformer", "attention", "encoder", "decoder",
            "block", "embedding", "classifier", "feature extractor",
            "alexnet", "resnet", "vgg", "googlenet"
        ]
        
        experiment_keywords = [
            "accuracy", "loss", "epoch", "precision", "recall", "f1",
            "curve", "performance", "comparison", "training", "validation"
        ]
        
        dataset_keywords = [
            "sample", "example", "input", "dataset", "class", "category"
        ]
        
        # Check for experiment results first (highest priority)
        experiment_matches = [kw for kw in experiment_keywords if kw in ocr_lower]
        if experiment_matches:
            # Extract model names if present
            components = []
            for kw in architecture_keywords:
                if kw in ocr_lower:
                    components.append(kw.upper())
            
            # Add metric names
            for kw in experiment_matches:
                components.append(kw.capitalize())
            
            return {
                "type": "experiment_result",
                "description": "展示实验结果和性能指标",
                "important_components": components[:5],
                "code_relevant_information": "需要实现模型训练和评估代码，包含指标计算"
            }
        
        # Check for architecture
        architecture_matches = [kw for kw in architecture_keywords if kw in ocr_lower]
        if architecture_matches:
            # Check if it's a dataset sample instead
            dataset_matches = [kw for kw in dataset_keywords if kw in ocr_lower]
            if dataset_matches and len(dataset_matches) > len(architecture_matches):
                return {
                    "type": "dataset",
                    "description": "展示数据样本和类别示例",
                    "important_components": [],
                    "code_relevant_information": "需要实现数据加载和预处理代码"
                }
            
            # Extract specific model components
            components = []
            for kw in architecture_keywords:
                if kw in ocr_lower:
                    if kw in ["cnn", "transformer", "attention", "encoder", "decoder"]:
                        components.append(kw.upper() + " Module")
                    else:
                        components.append(kw.upper())
            
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": components[:5],
                "code_relevant_information": "需要实现模型架构和特征提取模块"
            }
        
        # Check for dataset
        dataset_matches = [kw for kw in dataset_keywords if kw in ocr_lower]
        if dataset_matches:
            return {
                "type": "dataset",
                "description": "展示数据样本和输入示例",
                "important_components": [],
                "code_relevant_information": "需要实现数据加载和预处理代码"
            }
        
        # Default case
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "图片分析失败",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    try:
        image_path = sys.argv[1]
        # OCR text provided externally
        ocr_text = "output Tape Input 152r"
        
        result = analyze_image(image_path, ocr_text)
        print(json.dumps(result, ensure_ascii=False))
    except Exception as e:
        print(json.dumps({
            "type": "other",
            "description": "程序执行失败",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
```
