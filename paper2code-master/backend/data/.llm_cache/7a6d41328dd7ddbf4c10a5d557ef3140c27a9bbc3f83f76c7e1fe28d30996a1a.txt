```python
import sys
import json
import numpy as np
import cv2


def load_image(image_path):
    data = np.fromfile(image_path, dtype=np.uint8)
    img = cv2.imdecode(data, cv2.IMREAD_COLOR)
    if img is None:
        raise ValueError("无法读取图片")
    return img


def extract_components(text, kind):
    model_keywords = [
        "AlexNet", "ResNet", "VGG", "GoogLeNet", "Inception",
        "DenseNet", "MobileNet", "EfficientNet", "Transformer",
        "CNN", "RNN", "LSTM", "Attention", "Encoder", "Decoder",
        "BERT", "U-Net", "UNet"
    ]
    metric_keywords = [
        "Accuracy", "Precision", "Recall", "F1", "F1-score",
        "Loss", "Epoch", "AUC", "mAP", "IoU"
    ]
    found = []
    if kind == "experiment_result":
        for k in metric_keywords:
            if k.lower() in text.lower():
                if k not in found:
                    found.append(k)
        for k in model_keywords:
            if k.lower() in text.lower():
                if k not in found:
                    found.append(k)
    elif kind == "architecture":
        for k in model_keywords:
            if k.lower() in text.lower():
                if k not in found:
                    found.append(k)
    return found


def classify_visual(img):
    try:
        h, w = img.shape[:2]
        max_dim = 256
        scale = max_dim / max(h, w)
        if scale < 1:
            img_small = cv2.resize(
                img,
                (max(1, int(w * scale)), max(1, int(h * scale))),
                interpolation=cv2.INTER_AREA,
            )
        else:
            img_small = img.copy()

        hsv = cv2.cvtColor(img_small, cv2.COLOR_BGR2HSV)
        gray = cv2.cvtColor(img_small, cv2.COLOR_BGR2GRAY)
        v = hsv[:, :, 2]
        s = hsv[:, :, 1]

        white_ratio = float(np.mean((v > 200) & (s < 50)))
        sat_ratio = float(np.mean(s > 80))

        quant = (img_small // 32).astype(np.int32)
        color_code = quant[:, :, 0] * 1024 + quant[:, :, 1] * 32 + quant[:, :, 2]
        unique_colors = len(np.unique(color_code))

        small_gray = cv2.resize(gray, (64, 64), interpolation=cv2.INTER_AREA).astype(np.float32)
        kernel = np.ones((3, 3), dtype=np.float32) / 9.0
        local_mean = cv2.filter2D(small_gray, -1, kernel)
        diff = small_gray - local_mean
        local_var = cv2.filter2D(diff * diff, -1, kernel)
        texture = float(np.sqrt(local_var.mean()))

        if unique_colors > 4000 and texture > 25 and white_ratio < 0.4:
            return "dataset", "展示数据集中样本或类别图片", [], "需要准备数据集并进行预处理"

        if white_ratio > 0.4 and unique_colors > 500 and sat_ratio > 0.02:
            if texture > 30 and unique_colors > 3000:
                return "dataset", "展示数据集中样本或类别图片", [], "需要准备数据集并进行预处理"
            return "experiment_result", "展示实验性能曲线或对比柱状图", [], "需要实现模型评估、训练曲线绘制相关代码"

        if white_ratio > 0.5:
            return "architecture", "展示模型网络结构和数据流关系", [], "需要实现神经网络模块结构"

        if unique_colors > 3000:
            return "dataset", "展示数据集中样本或类别图片", [], "需要准备数据集并进行预处理"

        return "experiment_result", "展示实验性能曲线或对比柱状图", [], "需要实现模型评估和训练流程"
    except Exception:
        return "other", "无法确定图片类型", [], ""


def classify_by_ocr(ocr_text):
    low = ocr_text.lower()
    metric_kw = [
        "accuracy", "loss", "epoch", "precision", "recall",
        "f1", "f1-score", "auc", "map", "iou"
    ]
    if any(k in low for k in metric_kw):
        return (
            "experiment_result",
            "展示实验性能指标或训练曲线",
            extract_components(ocr_text, "experiment_result"),
            "需要实现模型评估指标计算和训练流程",
        )

    arch_kw = [
        "cnn", "transformer", "attention", "encoder", "decoder",
        "block", "embedding", "classifier", "feature extractor",
        "alexnet", "resnet", "vgg", "googlenet"
    ]
    if any(k in low for k in arch_kw):
        return (
            "architecture",
            "展示模型网络结构和数据流关系",
            extract_components(ocr_text, "architecture"),
            "需要实现相应的神经网络模块和特征提取模块",
        )

    dataset_kw = [
        "dataset", "sample", "input image", "classes",
        "example", "training image", "test image"
    ]
    if any(k in low for k in dataset_kw):
        return (
            "dataset",
            "展示数据集中样本或类别图片",
            [],
            "需要准备相应的数据集和预处理方式",
        )

    return None


def main():
    try:
        if len(sys.argv) < 2:
            print(json.dumps({
                "type": "other",
                "description": "无法确定图片类型",
                "important_components": [],
                "code_relevant_information": ""
            }, ensure_ascii=False))
            return

        image_path = sys.argv[1]
        ocr_text = sys.argv[2] if len(sys.argv) > 2 else ""

        img = None
        try:
            img = load_image(image_path)
        except Exception:
            img = None

        if img is not None:
            visual_type, visual_desc, visual_comp, visual_code = classify_visual(img)
        else:
            visual_type, visual_desc, visual_comp, visual_code = (
                "other", "无法确定图片类型", [], ""
            )

        ocr_result = classify_by_ocr(ocr_text)

        if ocr_result is not None:
            o_type, o_desc, o_comp, o_code = ocr_result

            if o_type == "experiment_result":
                result = {
                    "type": "experiment_result",
                    "description": o_desc,
                    "important_components": o_comp,
                    "code_relevant_information": o_code,
                }
            elif o_type == "architecture":
                if visual_type == "experiment_result":
                    comps = extract_components(ocr_text, "experiment_result")
                    result = {
                        "type": "experiment_result",
                        "description": "展示实验性能曲线或对比柱状图",
                        "important_components": comps,
                        "code_relevant_information": "需要实现模型评估、训练曲线绘制相关代码",
                    }
                else:
                    result = {
                        "type": "architecture",
                        "description": o_desc,
                        "important_components": o_comp,
                        "code_relevant_information": o_code,
                    }
            else:
                result = {
                    "type": o_type,
                    "description": o_desc,
                    "important_components": o_comp,
                    "code_relevant_information": o_code,
                }
        else:
            result = {
                "type": visual_type,
                "description": visual_desc,
                "important_components": visual_comp,
                "code_relevant_information": visual_code,
            }

        print(json.dumps(result, ensure_ascii=False))
    except Exception:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))


if __name__ == "__main__":
    main()
```
