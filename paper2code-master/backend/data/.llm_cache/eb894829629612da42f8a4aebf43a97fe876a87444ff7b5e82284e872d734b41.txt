        # Convert OCR text to lowercase for matching

            "block", "embedding", "classifier", "feature extractor",

        ]
        
        # Dataset keywords
        dataset_keywords = [
            "sample", "example", "dataset", "input image"
        ]
        
        # Check for experiment result first (highest priority)
        exp_matches = [kw for kw in exp_keywords if kw in ocr_lower]
        if exp_matches:
            # Extract model names and metrics
            components = []
            for kw in ["alexnet", "resnet", "vgg", "googlenet", "transformer", "cnn"]:
                if kw in ocr_lower:
                    components.append(kw.capitalize())
            for kw in ["accuracy", "loss", "precision", "recall", "f1"]:
                if kw in ocr_lower:
                    components.append(kw.capitalize())
            
            return {
                "type": "experiment_result",

                "important_components": components if components else [],
                "code_relevant_information": "需要实现模型训练和评估代码，包含指标计算"
            }
        
        # Check for architecture
        arch_matches = [kw for kw in arch_keywords if kw in ocr_lower]
        if arch_matches:
            # Extract specific components
            components = []
            if "transformer" in ocr_lower:
                components.append("Transformer")
            if "attention" in ocr_lower:
                components.append("Attention Module")
            if "encoder" in ocr_lower:
                components.append("Encoder")
            if "decoder" in ocr_lower:
                components.append("Decoder")

                components.append("Gated Residual Network")
            if "self-attention" in ocr_lower:
                components.append("Self-Attention")
            if "multi-head" in ocr_lower:
                components.append("Multi-head Attention")
            
            return {
                "type": "architecture",
                "description": "展示模型网络结构和数据流关系",
                "important_components": components if components else [],
                "code_relevant_information": "需要实现包含注意力机制和门控残差网络的深度学习模型"
            }
        
        # Check for dataset
        dataset_matches = [kw for kw in dataset_keywords if kw in ocr_lower]
