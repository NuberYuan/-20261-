import sys
import json
import numpy as np
import cv2
import os

def analyze_image(image_path):
    try:
        # Read image with Chinese path support
        data = np.fromfile(image_path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        
        if img is None:
            return {
                "type": "other",
                "description": "无法读取图片",
                "important_components": [],
                "code_relevant_information": ""
            }
        
        # OCR text is provided externally (empty in this case)
        ocr_text = ""
        
        # Since OCR is empty, use visual features and filename hints
        # Get image basic properties
        height, width = img.shape[:2]
        
        # Check if image is mostly white/light (common for architecture diagrams)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        mean_brightness = np.mean(gray)
        
        # Check color variance (experiment results often have colored lines)
        color_variance = np.std(img.reshape(-1, 3), axis=0).mean()
        
        # Check aspect ratio (architecture diagrams often wider)
        aspect_ratio = width / height if height > 0 else 0
        
        # Default classification
        result_type = "other"
        description = "无法确定图片类型"
        components = []
        code_info = ""
        
        # Since OCR is empty, use visual heuristics
        # Architecture diagrams: usually white background, box-like structures, moderate complexity
        if mean_brightness > 200 and aspect_ratio > 1.2:
            result_type = "architecture"
            description = "展示模型网络结构和数据流关系"
            components = []
            code_info = "需要实现模型网络结构"
        
        # Experiment results: often have colored elements, varying brightness
        elif color_variance > 30 and mean_brightness < 200:
            result_type = "experiment_result"
            description = "展示实验结果对比或性能指标"
            components = []
            code_info = "需要实现实验评估和结果可视化"
        
        # Dataset images: usually contain natural images or samples
        elif mean_brightness < 180:
            result_type = "dataset"
            description = "展示数据集样本或输入示例"
            components = []
            code_info = "需要实现数据加载和预处理"
        
        return {
            "type": result_type,
            "description": description,
            "important_components": components,
            "code_relevant_information": code_info
        }
        
    except Exception as e:
        return {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }

if __name__ == "__main__":
    if len(sys.argv) > 1:
        image_path = sys.argv[1]
        result = analyze_image(image_path)
        print(json.dumps(result, ensure_ascii=False))
    else:
        print(json.dumps({
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }, ensure_ascii=False))
