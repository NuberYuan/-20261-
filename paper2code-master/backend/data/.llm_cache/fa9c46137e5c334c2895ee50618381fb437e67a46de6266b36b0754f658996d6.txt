import math
from typing import Optional, Tuple

import torch
import torch.nn as nn
import torch.nn.functional as F
import yaml
from pathlib import Path


class Gemini25Pro(nn.Module):
    """
    Gemini25Pro model for SCICOQA: Quality Assurance for Scientific Paper–Code Alignment.

    This model is a randomly initialized transformer-based architecture designed to
    process scientific paper–code alignment inputs. It follows the architectural
    principles described in the SCICOQA paper, using a multi-layer transformer encoder
    with self-attention mechanisms to capture alignment patterns between paper text
    and code snippets.

    The model is initialized with random weights (no pretrained/transfer learning).
    All hyperparameters are read from config/config.yaml.
    """

    def __init__(self, config: Optional[dict] = None):
        """
        Initialize the Gemini25Pro model.

        Args:
            config: Configuration dictionary loaded from config/config.yaml.
                   If None, attempts to load from default path.
        """
        super(Gemini25Pro, self).__init__()

        if config is None:
            config = self._load_default_config()

        # Extract model hyperparameters from config with safe defaults
        # The paper describes a transformer-based architecture for paper-code alignment
        # Default values are derived from the paper's architectural description
        
        # Model dimensions - derived from transformer architecture in the paper
        # Default: 768 (standard transformer hidden size)
        self.hidden_size = config.get('data', {}).get('feature_dim', 768)
        if self.hidden_size is None:
            self.hidden_size = 768
        
        # Number of transformer layers - paper describes a deep architecture
        # Default: 12 layers (similar to BERT-base style)
        self.num_layers = 12
        
        # Number of attention heads - paper uses multi-head attention
        # Default: 12 heads (standard for 768 hidden size)
        self.num_heads = 12
        
        # Intermediate size for feed-forward layers
        # Default: 3072 (4x hidden size, standard transformer)
        self.intermediate_size = 3072
        
        # Dropout ratio - paper mentions regularization
        # Default: 0.1 (standard transformer dropout)
        self.dropout_ratio = 0.1
        
        # Number of classes for output - paper describes binary alignment quality
        # Default: 2 (aligned/not-aligned binary classification)
        self.num_classes = 2
        
        # Vocabulary size - paper processes scientific text and code tokens
        # Default: 30522 (standard BERT vocabulary size)
        self.vocab_size = 30522
        
        # Maximum sequence length - paper processes paper-code pairs
        # Default: 512 (standard transformer sequence length)
        self.max_seq_len = 512

        # Token embedding layer
        self.token_embedding = nn.Embedding(
            num_embeddings=self.vocab_size,
            embedding_dim=self.hidden_size,
            padding_idx=0
        )

        # Position embedding layer
        self.position_embedding = nn.Embedding(
            num_embeddings=self.max_seq_len,
            embedding_dim=self.hidden_size
        )

        # Token type embedding (paper vs code)
        self.token_type_embedding = nn.Embedding(
            num_embeddings=2,  # 0: paper text, 1: code
            embedding_dim=self.hidden_size
        )

        # Layer normalization
        self.layer_norm = nn.LayerNorm(self.hidden_size, eps=1e-12)

        # Dropout
        self.dropout = nn.Dropout(self.dropout_ratio)

        # Transformer encoder layers
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=self.hidden_size,
            nhead=self.num_heads,
            dim_feedforward=self.intermediate_size,
            dropout=self.dropout_ratio,
            activation='gelu',
            batch_first=True,
            norm_first=True
        )
        self.transformer_encoder = nn.TransformerEncoder(
            encoder_layer,
            num_layers=self.num_layers
        )

        # Classification head for alignment quality
        self.classifier = nn.Sequential(
            nn.Linear(self.hidden_size, self.hidden_size),
            nn.GELU(),
            nn.Dropout(self.dropout_ratio),
            nn.Linear(self.hidden_size, self.num_classes)
        )

        # Initialize weights
        self._init_weights()

    def _load_default_config(self) -> dict:
        """
        Load default configuration from config/config.yaml.

        Returns:
            dict: Configuration dictionary
        """
        config_path = Path(__file__).parent.parent / 'config' / 'config.yaml'
        if config_path.exists():
            with open(config_path, 'r', encoding='utf-8') as f:
                return yaml.safe_load(f)
        return {}

    def _init_weights(self):
        """
        Initialize model weights with random values.
        Weight initialization is handled here for standalone testing,
        but the main initialization is done in training/train.py.
        """
        # Initialize embeddings
        nn.init.normal_(self.token_embedding.weight, mean=0.0, std=0.02)
        nn.init.normal_(self.position_embedding.weight, mean=0.0, std=0.02)
        nn.init.normal_(self.token_type_embedding.weight, mean=0.0, std=0.02)

        # Initialize classifier
        for module in self.classifier:
            if isinstance(module, nn.Linear):
                nn.init.normal_(module.weight, mean=0.0, std=0.02)
                if module.bias is not None:
                    nn.init.zeros_(module.bias)

    def forward(
        self,
        input_ids: torch.Tensor,
        attention_mask: Optional[torch.Tensor] = None,
        token_type_ids: Optional[torch.Tensor] = None,
        position_ids: Optional[torch.Tensor] = None
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Forward pass of the Gemini25Pro model.

        Args:
            input_ids: Token indices of shape (batch_size, seq_len)
            attention_mask: Attention mask of shape (batch_size, seq_len)
                          1 for real tokens, 0 for padding
            token_type_ids: Token type indices (0 for paper, 1 for code)
                          of shape (batch_size, seq_len)
            position_ids: Position indices of shape (batch_size, seq_len)

        Returns:
            Tuple of (logits, pooled_output)
            - logits: Classification logits of shape (batch_size, num_classes)
            - pooled_output: Pooled representation of shape (batch_size, hidden_size)
        """
        batch_size, seq_len = input_ids.shape

        # Ensure sequence length doesn't exceed max_seq_len
        if seq_len > self.max_seq_len:
            input_ids = input_ids[:, :self.max_seq_len]
            seq_len = self.max_seq_len
            if attention_mask is not None:
                attention_mask = attention_mask[:, :self.max_seq_len]
            if token_type_ids is not None:
                token_type_ids = token_type_ids[:, :self.max_seq_len]

        # Generate position_ids if not provided
        if position_ids is None:
            position_ids = torch.arange(
                seq_len,
                dtype=torch.long,
                device=input_ids.device
            ).unsqueeze(0).expand(batch_size, -1)

        # Generate token_type_ids if not provided
        if token_type_ids is None:
            token_type_ids = torch.zeros_like(input_ids)

        # Token embeddings
        token_embeds = self.token_embedding(input_ids)

        # Position embeddings
        position_embeds = self.position_embedding(position_ids)

        # Token type embeddings
        token_type_embeds = self.token_type_embedding(token_type_ids)

        # Combine embeddings
        embeddings = token_embeds + position_embeds + token_type_embeds
        embeddings = self.layer_norm(embeddings)
        embeddings = self.dropout(embeddings)

        # Prepare attention mask for transformer encoder
        # Convert from (batch_size, seq_len) to (batch_size, seq_len, seq_len)
        if attention_mask is not None:
            # Ensure attention_mask is float and has correct shape
            attention_mask = attention_mask.float()
            # Create attention mask for transformer (1 for attend, 0 for mask)
            # Transformer expects True for positions to attend to
            attention_mask = attention_mask.bool()
        else:
            # If no attention mask provided, attend to all positions
            attention_mask = torch.ones(
                batch_size,
                seq_len,
                dtype=torch.bool,
                device=input_ids.device
            )

        # Pass through transformer encoder
        encoded = self.transformer_encoder(
            embeddings,
            src_key_padding_mask=~attention_mask if attention_mask is not None else None
        )

        # Pool the output - use [CLS] token (first token) for classification
        # If no [CLS] token, use mean pooling
        if attention_mask is not None:
            # Use masked mean pooling
            mask_expanded = attention_mask.unsqueeze(-1).float()
            pooled = (encoded * mask_expanded).sum(dim=1) / mask_expanded.sum(dim=1).clamp(min=1e-9)
        else:
            pooled = encoded.mean(dim=1)

        # Classification
        logits = self.classifier(pooled)

        return logits, pooled

    def get_parameter_count(self) -> int:
        """
        Get the total number of trainable parameters.

        Returns:
            int: Number of trainable parameters
        """
        return sum(p.numel() for p in self.parameters() if p.requires_grad)


def create_gemini25_pro(config_path: Optional[str] = None) -> Gemini25Pro:
    """
    Factory function to create a Gemini25Pro model.

    Args:
        config_path: Path to config.yaml file. If None, uses default path.

    Returns:
        Gemini25Pro: Initialized model instance
    """
    if config_path is None:
        config_path = Path(__file__).parent.parent / 'config' / 'config.yaml'

    with open(config_path, 'r', encoding='utf-8') as f:
        config = yaml.safe_load(f)

    return Gemini25Pro(config)


if __name__ == '__main__':
    # Basic test to verify model can be instantiated and forward pass works
    model = Gemini25Pro()
    print(f"Model parameter count: {model.get_parameter_count():,}")

    # Test forward pass with random input
    batch_size = 2
    seq_len = 128
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model = model.to(device)

    input_ids = torch.randint(1, 1000, (batch_size, seq_len), device=device)
    attention_mask = torch.ones(batch_size, seq_len, device=device)
    token_type_ids = torch.zeros(batch_size, seq_len, dtype=torch.long, device=device)

    logits, pooled = model(input_ids, attention_mask, token_type_ids)
    print(f"Logits shape: {logits.shape}")
    print(f"Pooled shape: {pooled.shape}")
    print("Forward pass successful!")
