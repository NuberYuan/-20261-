def load_image(image_path):
    data = np.fromfile(image_path, dtype=np.uint8)
    img = cv2.imdecode(data, cv2.IMREAD_COLOR)
    if img is None:
        raise ValueError("cannot decode image")
    return img


def read_external_ocr_text(image_path):
    if len(sys.argv) > 2:
        return sys.argv[2].strip()

    base = os.path.splitext(image_path)[0]
    for ext in (".txt", ".ocr.txt", ".ocr.json"):
        path = base + ext
        if not os.path.isfile(path):
            continue
        try:
            with open(path, "r", encoding="utf-8") as f:
                content = f.read().strip()
            if ext.endswith(".json"):
                try:
                    obj = json.loads(content)
                    if isinstance(obj, dict):
                        content = obj.get("text") or obj.get("ocr_text") or content
                except Exception:
                    pass
            if content:
                return content
        except Exception:
            continue
    return ""


def compute_visual_stats(img):
    h, w = img.shape[:2]
    rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)

    white_mask = np.all(rgb >= 225, axis=2)
    black_mask = np.all(rgb <= 60, axis=2)
    white_ratio = float(np.mean(white_mask))
    black_ratio = float(np.mean(black_mask))

    sat = hsv[:, :, 1].astype(np.float32)
    val = hsv[:, :, 2].astype(np.float32)

    low_sat_ratio = float(np.mean(sat < 25))
    colorful_mask = (sat > 70) & (val > 50) & (val < 235)
    colorful_pixel_ratio = float(np.mean(colorful_mask))

    R = rgb[:, :, 0].astype(np.float32)
    G = rgb[:, :, 1].astype(np.float32)
    B = rgb[:, :, 2].astype(np.float32)
    rg = R - G
    yb = 0.5 * (R + G) - B
    colorfulness = float(
        np.sqrt(np.mean(rg ** 2) + np.mean(yb ** 2))
        + 0.3 * np.sqrt(np.var(rg) + np.var(yb))
    )

    scale = max(1, min(w, h) // 4)
    small = cv2.resize(
        rgb,
        (max(1, w // scale), max(1, h // scale)),
        interpolation=cv2.INTER_AREA
    )
    quant = (small // 32).astype(np.uint8)
    unique_colors = int(len(np.unique(quant.reshape(-1, 3), axis=0)))

    return {
        "white_ratio": white_ratio,
        "black_ratio": black_ratio,
        "low_sat_ratio": low_sat_ratio,
        "colorful_pixel_ratio": colorful_pixel_ratio,
        "colorfulness": colorfulness,
        "unique_colors": unique_colors
    }


def count_hits(text, keywords):
    t = text.lower()
    return sum(1 for kw in keywords if kw in t)


def classify_image_type(ocr_text, image_path, stats):
    text = ocr_text.lower()
    path_text = image_path.lower()

    exp_text_hits = count_hits(text, EXP_KEYWORDS)
    arch_text_hits = count_hits(text, ARCH_KEYWORDS)
    data_text_hits = count_hits(text, DATA_KEYWORDS)

    path_exp_hits = count_hits(path_text, PATH_EXP_KEYWORDS)
    path_arch_hits = count_hits(path_text, PATH_ARCH_KEYWORDS)
    path_data_hits = count_hits(path_text, PATH_DATA_KEYWORDS)

    is_natural = (
        stats["white_ratio"] < 0.28
        and stats["colorfulness"] > 22
        and stats["unique_colors"] > 10
    )

    if exp_text_hits > 0:
        return "experiment_result"

    if arch_text_hits > 0:
        return "architecture"

    if is_natural:
        return "dataset"

    if data_text_hits > 0 and path_data_hits > 0:
        return "dataset"

    if path_exp_hits > 0:
        return "experiment_result"

    if path_arch_hits > 0:
        return "architecture"

    if (
        stats["white_ratio"] > 0.35
        and stats["colorful_pixel_ratio"] > 0.015
        and stats["black_ratio"] < 0.15
    ):
        return "experiment_result"

    if stats["white_ratio"] > 0.35:
        return "architecture"

    return "other"


def extract_architecture_components(text):
    text_l = text.lower()
    comps = []
    mapping = [
        ("transformer encoder", "Transformer Encoder"),
        ("attention module", "Attention Module"),
        ("transformer", "Transformer"),
        ("attention", "Attention"),
        ("encoder", "Encoder"),
        ("decoder", "Decoder"),
        ("cnn", "CNN"),
        ("alexnet", "AlexNet"),
        ("resnet", "ResNet"),
        ("vgg", "VGG"),
        ("googlenet", "GoogLeNet"),
        ("feature extractor", "Feature Extractor"),
        ("classifier", "Classifier"),
        ("embedding", "Embedding"),
        ("block", "Block"),
    ]
    for key, name in mapping:
        if key in text_l and name not in comps:
            comps.append(name)
    return comps


def extract_experiment_components(text):
    text_l = text.lower()
    comps = []

    model_map = {
        "alexnet": "AlexNet",
        "resnet": "ResNet",
        "vgg": "VGG",
        "googlenet": "GoogLeNet",
        "cnn": "CNN",
        "transformer": "Transformer",
        "attention": "Attention",
        "encoder": "Encoder",
        "decoder": "Decoder",
    }
    for key, name in model_map.items():
        if key in text_l and name not in comps:
            comps.append(name)

    metric_map = {
        "accuracy": "Accuracy",
        "loss": "Loss",
        "epoch": "Epoch",
        "precision": "Precision",
        "recall": "Recall",
        "f1": "F1-score",
        "准确率": "Accuracy",
        "损失": "Loss",
        "精度": "Precision",
        "召回": "Recall",
    }
    for key, name in metric_map.items():
        if key in text_l and name not in comps:
            comps.append(name)

    return comps


def build_description(image_type):
    if image_type == "architecture":
        return "展示模型网络结构和数据流关系"
    if image_type == "experiment_result":
        return "展示实验性能指标或训练过程变化"
    if image_type == "dataset":
        return "展示数据样本或类别示例"
    return "无法确定图片类型"


def build_code_relevant_information(image_type, comps):
    if image_type == "architecture":
        if comps:
            return "需要实现" + "、".join(comps) + "等相关模块"
        return "需要实现CNN特征提取模块和分类头"
    if image_type == "experiment_result":
        return "需要实现模型训练与评估流程，并记录指标变化"
    if image_type == "dataset":
        return "需要准备图像数据并划分训练集与测试集"
    return ""


def main():
    try:
        try:
            sys.stdout.reconfigure(encoding="utf-8")
        except Exception:
            pass

        if len(sys.argv) < 2:
            raise RuntimeError("missing image path")

        image_path = sys.argv[1]
        ocr_text = read_external_ocr_text(image_path)

        img = load_image(image_path)
        stats = compute_visual_stats(img)

        image_type = classify_image_type(ocr_text, image_path, stats)

        if image_type == "architecture":
            comps = extract_architecture_components(ocr_text)
        elif image_type == "experiment_result":
            comps = extract_experiment_components(ocr_text)
        else:
            comps = []

        result = {
            "type": image_type,
            "description": build_description(image_type),
            "important_components": comps,
            "code_relevant_information": build_code_relevant_information(image_type, comps)
        }

        print(json.dumps(result, ensure_ascii=False))

    except Exception:
        fallback = {
            "type": "other",
            "description": "无法确定图片类型",
            "important_components": [],
            "code_relevant_information": ""
        }
        print(json.dumps(fallback, ensure_ascii=False))


if __name__ == "__main__":
    main()
